#pragma once

#include <string>
#include "SegmentTree.h"

// Quản lý thông tin và trạng thái đặt chỗ của một toa tàu
struct Carriage {
    int carriageNo;                         // Số hiệu toa
    std::string type;                       // Loại toa (ghế ngồi, giường nằm...)
    int totalSeats;                         // Tổng số ghế trong toa
    SegmentTree<int>* seatSegments;         // Mảng Segment Tree quản lý tình trạng từng ghế theo chặng
    int configuredSegments;                 // Số lượng chặng đường hiện tại

public:
    // Khởi tạo thông tin cơ bản của toa
    Carriage(int cNo, const std::string& cType, int seats)
        : carriageNo(cNo), type(cType), totalSeats(seats),
          seatSegments(nullptr), configuredSegments(0) {}

    // Giải phóng mảng Segment Tree trên Heap
    ~Carriage() {
        delete[] seatSegments;
    }

    // Chặn sao chép để tránh lỗi double-free con trỏ seatSegments
    Carriage(const Carriage&) = delete;
    Carriage& operator=(const Carriage&) = delete;

    // Lấy thông tin toa
    int getCarriageNo() const { return carriageNo; }
    std::string getType() const { return type; }
    int getTotalSeats() const { return totalSeats; }

    // Khởi tạo/cấp phát lại Segment Tree cho từng ghế theo số lượng chặng ga
    void configureSeatSegments(int segmentCount) {
        if (segmentCount <= 0) {
            delete[] seatSegments;
            seatSegments = nullptr;
            configuredSegments = 0;
            return;
        }
        // Giữ nguyên bộ nhớ, chỉ reset trạng thái nếu số chặng không đổi
        if (segmentCount == configuredSegments) {
            for (int seat = 0; seat < totalSeats; ++seat) {
                seatSegments[seat].initialize(segmentCount, 1);
            }
            return;
        }

        // Cấp phát mảng Segment Tree mới cho tất cả các ghế
        delete[] seatSegments;
        seatSegments = nullptr;
        configuredSegments = segmentCount;
        if (totalSeats <= 0) {
            return;
        }
        seatSegments = new SegmentTree<int>[totalSeats];
        for (int seat = 0; seat < totalSeats; ++seat) {
            seatSegments[seat].initialize(configuredSegments, 1);
        }
    }

    // Kiểm tra ghế có còn trống trên toàn bộ đoạn chặng [fromSegment, toSegment] hay không
    bool isSeatAvailableOnSegments(int seatNo, int fromSegment, int toSegment) {
        if (!seatSegments || seatNo < 1 || seatNo > totalSeats) {
            return false;
        }
        return seatSegments[seatNo - 1].queryMin(fromSegment, toSegment) > 0;
    }

    // Cập nhật trạng thái ghế trên đoạn chặng (delta = -1 khi đặt vé, +1 khi hủy vé)
    void updateSeatSegments(int seatNo, int fromSegment, int toSegment, int delta) {
        if (seatSegments && seatNo >= 1 && seatNo <= totalSeats) {
            seatSegments[seatNo - 1].updateRange(fromSegment, toSegment, delta);
        }
    }
};

// ==================== STATIONS ====================
void FileManager::loadStations(const string& filename, StationList& stationList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string strID, name, strKm, strTracks;

        if (getline(ss, strID, '|') && getline(ss, name, '|') &&
            getline(ss, strKm, '|') && getline(ss, strTracks, '|')) {
            stationList.addStation(stoi(strID), name, stod(strKm), stoi(strTracks));
        }
    }
    inFile.close();
}

void FileManager::saveStations(const string& filename, const StationList& stationList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    for (LinkedList<StationNode*>::Node* station = stationList.getHead();
         station != nullptr; station = station->next) {
        StationNode* st = station->data;
        outFile << st->getStationID() << "|"
                << st->getName() << "|"
                << st->getKmMarker() << "|"
                << st->getMaxTracks() << "\n";
    }
    outFile.close();
}

// ==================== TRAINS ====================
void FileManager::loadTrains(const string& filename, TrainList& trainList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    TrainNode* currentTrain = nullptr;

    while (getline(inFile, line)) {
        if (line.empty()) continue;

        stringstream ss(line);
        string token;
        getline(ss, token, '|');

        if (token == "CARRIAGE") {
            if (currentTrain) {
                string strNo, type, strSeats;
                if (getline(ss, strNo, '|') && getline(ss, type, '|') && getline(ss, strSeats, '|')) {
                    currentTrain->addCarriage(stoi(strNo), type, stoi(strSeats));
                }
            }
        } 
        else if (token == "STOP") {
            if (currentTrain) {
                string strStID, strArr, strDep, strTrack;
                if (getline(ss, strStID, '|') && getline(ss, strArr, '|') &&
                    getline(ss, strDep, '|') && getline(ss, strTrack, '|')) {
                    currentTrain->addSchedule(stoi(strStID), stoi(strArr), stoi(strDep), stoi(strTrack));
                }
            }
        } 
        else {
            // Định dạng dòng đầu: Mã Tàu | Hướng chạy [ | Độ ưu tiên ]
            string trainCode = token;
            string strDir, strPrio;

            if (getline(ss, strDir, '|')) {
                int prio = -1;
                if (getline(ss, strPrio, '|') && !strPrio.empty()) {
                    try {
                        prio = stoi(strPrio);
                    } catch (...) {
                        prio = -1;
                    }
                }

                trainList.addTrain(trainCode, stoi(strDir), prio);
                currentTrain = trainList.findTrainByCode(trainCode);
            }
        }
    }
    inFile.close();
}

void FileManager::saveTrains(const string& filename, const TrainList& trainList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    for (LinkedList<TrainNode*>::Node* trainNode = trainList.getHead();
         trainNode != nullptr; trainNode = trainNode->next) {
        TrainNode* tr = trainNode->data;
        // Lưu kèm độ ưu tiên vào cột thứ 3: Mã Tàu | Hướng chạy | Độ ưu tiên
        outFile << tr->getTrainCode() << "|" 
                << tr->getDirection() << "|" 
                << tr->getPriority() << "\n";

        for (LinkedList<CarriageNode*>::Node* carriage = tr->getHeadCarriage();
             carriage != nullptr; carriage = carriage->next) {
            CarriageNode* c = carriage->data;
            outFile << "CARRIAGE|" << c->getCarriageNo() << "|" << c->getType() << "|" << c->getTotalSeats() << "\n";
        }

        for (LinkedList<StopScheduleNode*>::Node* stop = tr->getHeadSchedule();
             stop != nullptr; stop = stop->next) {
            StopScheduleNode* s = stop->data;
            outFile << "STOP|" << s->getStationID() << "|" << s->getArrivalMinute() << "|" << s->getDepartureMinute() << "|" << s->getTrackUsed() << "\n";
        }
    }
    outFile.close();
}

// Tạo hình chữ nhật bo góc bằng sf::ConvexShape
static sf::ConvexShape createRoundedRect(float width, float height, float radius, unsigned int cornerPoints = 6) {
    radius = std::min(radius, std::min(width, height) / 2.0f);
    sf::ConvexShape shape;
    shape.setPointCount(cornerPoints * 4);

    const float PI = 3.14159265359f;
    unsigned int idx = 0;

    sf::Vector2f centers[4] = {
        {width - radius, height - radius},
        {radius, height - radius},
        {radius, radius},
        {width - radius, radius}
    };

    for (int c = 0; c < 4; ++c) {
        float startAngle = c * (PI / 2.0f);
        for (unsigned int i = 0; i < cornerPoints; ++i) {
            float angle = startAngle + (static_cast<float>(i) / (cornerPoints - 1)) * (PI / 2.0f);
            shape.setPoint(idx++, {
                centers[c].x + radius * std::cos(angle),
                centers[c].y + radius * std::sin(angle)
            });
        }
    }
    shape.setOrigin({width / 2.0f, height / 2.0f});
    return shape;
}

// Thêm một hình chữ nhật (ghép từ 2 tam giác) vào VertexArray trong SFML 3.0
static void appendQuadAsTriangles(sf::VertexArray& va, float x, float y, float w, float h, const sf::Color& color) {
    sf::Vector2f p0(x, y), p1(x + w, y), p2(x + w, y + h), p3(x, y + h);
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p1, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p3, color});
}

// Vẽ đoạn thẳng có độ dày tùy chỉnh (dùng cho ghi rẽ nhánh đường ray phụ)
static void drawThickLine(sf::RenderWindow& window, sf::Vector2f p1, sf::Vector2f p2, float thickness, sf::Color color) {
    sf::Vector2f dir = p2 - p1;
    float length = std::sqrt(dir.x * dir.x + dir.y * dir.y);
    if (length <= 0.001f) return;

    sf::RectangleShape line({length, thickness});
    line.setOrigin({0.0f, thickness / 2.0f});
    line.setPosition(p1);
    float angleRad = std::atan2(dir.y, dir.x);
    line.setRotation(sf::radians(angleRad));
    line.setFillColor(color);
    window.draw(line);
}

// ==================== 3. HÀM TÍNH TỌA ĐỘ KM MƯỢT THEO THỜI GIAN THỰC ====================
static double calculateTrainKmSmooth(TrainNode* train, StationList* stationList, float currentMinute, int& trackUsedOut) {
    trackUsedOut = 0;
    LinkedList<StopScheduleNode*>::Node* firstStopNode = train->getHeadSchedule();
    StopScheduleNode* firstStop = firstStopNode ? firstStopNode->data : nullptr;
    if (!firstStop) return -1.0;

    LinkedList<StopScheduleNode*>::Node* lastStopNode = train->getTailSchedule();
    StopScheduleNode* lastStop = lastStopNode ? lastStopNode->data : nullptr;

    // 1. Chưa tới giờ xuất phát -> Tàu chờ sẵn tại Ga Đầu
    if (currentMinute < static_cast<float>(firstStop->getArrivalMinute())) {
        StationNode* st = stationList->findStationByID(firstStop->getStationID());
        if (st) {
            trackUsedOut = firstStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 2. Đã qua giờ về ga cuối -> Tàu đỗ tại Ga Cuối
    if (currentMinute > static_cast<float>(lastStop->getDepartureMinute())) {
        StationNode* st = stationList->findStationByID(lastStop->getStationID());
        if (st) {
            trackUsedOut = lastStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 3. Kiểm tra tàu đang đỗ tại ga hoặc đang di chuyển giữa 2 ga (Khu gian)
    for (LinkedList<StopScheduleNode*>::Node* stopNode = firstStopNode;
         stopNode != nullptr; stopNode = stopNode->next) {
        StopScheduleNode* curr = stopNode->data;
        float arr = static_cast<float>(curr->getArrivalMinute());
        float dep = static_cast<float>(curr->getDepartureMinute());

        if (currentMinute >= arr && currentMinute <= dep) {
            StationNode* st = stationList->findStationByID(curr->getStationID());
            if (st) {
                trackUsedOut = curr->getTrackUsed();
                return st->getKmMarker();
            }
        }

        StopScheduleNode* nextStop =
            stopNode->next ? stopNode->next->data : nullptr;
        if (nextStop != nullptr) {
            float nextArr = static_cast<float>(nextStop->getArrivalMinute());
            if (currentMinute > dep && currentMinute < nextArr) {
                StationNode* st1 = stationList->findStationByID(curr->getStationID());
                StationNode* st2 = stationList->findStationByID(nextStop->getStationID());
                if (st1 && st2 && nextArr > dep) {
                    trackUsedOut = 0; // 0: Đang chạy trên khu gian đơn
                    double ratio = static_cast<double>(currentMinute - dep) / static_cast<double>(nextArr - dep);
                    return st1->getKmMarker() + ratio * (st2->getKmMarker() - st1->getKmMarker());
                }
            }
        }
    }

    return -1.0;
}

// ==================== 4. VÒNG LẶP ĐỒ HỌA CHÍNH (GRAPHICAL SIMULATION ENGINE) ====================
void GraphicalSimulationEngine::runSFMLSimulation(StationList* stationList, TrainList* trainList) {
    if (!stationList || !trainList) return;

    sf::ContextSettings settings;
    settings.antiAliasingLevel = 8;
    sf::RenderWindow window(sf::VideoMode({WINDOW_WIDTH, WINDOW_HEIGHT}),
                            "PBL2: MO PHONG DIEU PHOI DUONG RAY & CHAY TAU (SFML GRAPHICS)",
                            sf::Style::Default, sf::State::Windowed, settings);
    window.setFramerateLimit(60);

    sf::Font font;
    if (!font.openFromFile("arial.ttf")) {
        cout << "[Canh bao] Khong tim thay font 'arial.ttf'!\n";
    }

    // Tìm mốc Km lớn nhất trên tuyến để chuẩn hóa tỷ lệ vẽ
    double maxKm = 100.0;
    for (LinkedList<StationNode*>::Node* station = stationList->getHead();
         station != nullptr; station = station->next) {
        StationNode* st = station->data;
        if (st->getKmMarker() > maxKm) maxKm = st->getKmMarker();
    }
    if (maxKm <= 0.0) maxKm = 100.0;

    const float marginX = 95.0f;
    const float trackY = 245.0f;
    const float trackWidth = WINDOW_WIDTH - 2.0f * marginX;
    const float sidingSpacingY = 44.0f;

    // Khởi tạo VertexArray cho Đường ray chính & Tà vẹt (Chỉ tốn 1 Draw Call mỗi khung hình)
    sf::VertexArray mainTrackVA(sf::PrimitiveType::Triangles);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 8.0f, trackWidth + 40.0f, 16.0f, Theme::BallastBed);
    for (float x = marginX - 10.0f; x <= marginX + trackWidth + 10.0f; x += 12.0f) {
        appendQuadAsTriangles(mainTrackVA, x, trackY - 7.0f, 4.0f, 14.0f, Theme::SleeperWood);
    }
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 3.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY + 1.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);

    // Mảng tĩnh quản lý hiệu ứng chuyển làn mượt (Smooth Y-Lerp) không dùng STL
    static const int MAX_SMOOTH_TRAINS = 64;
    TrainSmoothState smoothStates[MAX_SMOOTH_TRAINS];
    int smoothCount = 0;

    auto getSmoothY = [&](const string& code, float targetY, float dt) -> float {
        for (int i = 0; i < smoothCount; ++i) {
            if (smoothStates[i].code == code) {
                float lerpFactor = std::min(1.0f, dt * 8.5f);
                smoothStates[i].currentY += (targetY - smoothStates[i].currentY) * lerpFactor;
                return smoothStates[i].currentY;
            }
        }
        if (smoothCount < MAX_SMOOTH_TRAINS) {
            smoothStates[smoothCount].code = code;
            smoothStates[smoothCount].currentY = targetY;
            smoothCount++;
        }
        return targetY;
    };

    // Tự động lấy mốc thời gian bắt đầu sớm nhất của các đoàn tàu (trừ hao 15 phút chuẩn bị)
    int earliestMinute = 1440;
    for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
         trainNode != nullptr; trainNode = trainNode->next) {
        TrainNode* tr = trainNode->data;
        if (tr->getHeadSchedule()) {
            earliestMinute = std::min(earliestMinute,
                tr->getHeadSchedule()->data->getArrivalMinute());
        }
    }
    float startMinute = (earliestMinute < 1440) ? static_cast<float>(std::max(0, earliestMinute - 15)) : 0.0f;
    float currentMinute = startMinute;

    float timeSpeed = 30.0f;
    bool isPaused = false;
    sf::Clock deltaClock;
    sf::Clock animClock;

    while (window.isOpen()) {
        while (const auto event = window.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                window.close();
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Space) {
                    isPaused = !isPaused;
                } else if (keyPressed->code == sf::Keyboard::Key::Up) {
                    timeSpeed = std::min(240.0f, timeSpeed + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Down) {
                    timeSpeed = std::max(5.0f, timeSpeed - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Right) {
                    currentMinute = std::min(1439.0f, currentMinute + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Left) {
                    currentMinute = std::max(0.0f, currentMinute - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::R) {
                    currentMinute = startMinute;
                } else if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    window.close();
                }
            }
        }

        float dt = deltaClock.restart().asSeconds();
        float animTime = animClock.getElapsedTime().asSeconds();

        if (!isPaused) {
            currentMinute += timeSpeed * dt;
            if (currentMinute >= 1440.0f) currentMinute = 0.0f;
        }

        window.clear(Theme::BgCanvas);

        // ==================== A. VẼ LƯỚI TỌA ĐỘ NỀN (SUBTLE GRID) ====================
        for (float gx = marginX; gx <= marginX + trackWidth; gx += 120.0f) {
            sf::RectangleShape vLine({1.0f, 390.0f});
            vLine.setPosition({gx, 105.0f});
            vLine.setFillColor(Theme::BgGridLine);
            window.draw(vLine);
        }

        // ==================== B. VẼ NHÀ GA, ĐƯỜNG RAY PHỤ & ĐÈN TÍN HIỆU ====================
        for (LinkedList<StationNode*>::Node* station = stationList->getHead();
             station != nullptr; station = station->next) {
            StationNode* st = station->data;
            float posX = marginX + static_cast<float>(st->getKmMarker() / maxKm) * trackWidth;
            int maxTracks = st->getMaxTracks();

            // Kiểm tra có đoàn tàu nào đang dừng đỗ tại ga này không để bật đèn tín hiệu Đỏ
            bool stationOccupied = false;
            for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
                 trainNode != nullptr; trainNode = trainNode->next) {
                TrainNode* tr = trainNode->data;
                for (LinkedList<StopScheduleNode*>::Node* stopNode =
                         tr->getHeadSchedule(); stopNode != nullptr;
                     stopNode = stopNode->next) {
                    StopScheduleNode* sc = stopNode->data;
                    if (sc->getStationID() == st->getStationID()) {
                        if (currentMinute >= static_cast<float>(sc->getArrivalMinute()) &&
                            currentMinute <= static_cast<float>(sc->getDepartureMinute())) {
                            stationOccupied = true;
                            break;
                        }
                    }
                }
                if (stationOccupied) break;
            }

            // 1. Trục gióng dọc tại tâm nhà ga
            float totalStationHeight = (maxTracks - 1) * sidingSpacingY + 75.0f;
            sf::RectangleShape guideLine({2.0f, totalStationHeight});
            guideLine.setPosition({posX - 1.0f, trackY - 35.0f});
            guideLine.setFillColor(sf::Color(62, 70, 96, 110));
            window.draw(guideLine);

            // 2. Các đường ray phụ (Siding Tracks: từ Ray 2 đến maxTracks)
            for (int t = 2; t <= maxTracks; ++t) {
                float sidY = trackY + (t - 1) * sidingSpacingY;
                float prevY = trackY + (t - 2) * sidingSpacingY;
                float halfLen = 46.0f;

                // Ghi rẽ nhánh trái & phải nối chéo lên ray phía trên
                drawThickLine(window, {posX - halfLen - 18.0f, prevY}, {posX - halfLen, sidY}, 3.0f, Theme::SidingRail);
                drawThickLine(window, {posX + halfLen, sidY}, {posX + halfLen + 18.0f, prevY}, 3.0f, Theme::SidingRail);

                // Tà vẹt trên ray phụ
                for (float sx = posX - halfLen; sx <= posX + halfLen; sx += 10.0f) {
                    sf::RectangleShape slp({3.0f, 10.0f});
                    slp.setOrigin({1.5f, 5.0f});
                    slp.setPosition({sx, sidY});
                    slp.setFillColor(Theme::SleeperWood);
                    window.draw(slp);
                }

                // Thanh ray phụ
                sf::RectangleShape sidingBar({halfLen * 2.0f, 3.5f});
                sidingBar.setOrigin({halfLen, 1.75f});
                sidingBar.setPosition({posX, sidY});
                sidingBar.setFillColor(Theme::SidingRail);
                window.draw(sidingBar);

                // Nhãn ký hiệu ray phụ (R2, R3...)
                sf::Text trkLabel(font, "R" + to_string(t), 10);
                trkLabel.setFillColor(Theme::TextSecondary);
                trkLabel.setPosition({posX - halfLen - 18.0f, sidY - 6.0f});
                window.draw(trkLabel);
            }

            // 3. Vùng Sân ga (Station Platform Badge)
            sf::ConvexShape platform = createRoundedRect(92.0f, 24.0f, 6.0f);
            platform.setPosition({posX, trackY - 58.0f});
            platform.setFillColor(Theme::StationPlatform);
            platform.setOutlineThickness(1.5f);
            platform.setOutlineColor(Theme::HudBorder);
            window.draw(platform);

            sf::Text stTitle(font, st->getName(), 12);
            stTitle.setFillColor(Theme::AccentGold);
            sf::FloatRect tb = stTitle.getLocalBounds();
            stTitle.setPosition({posX - tb.size.x / 2.0f, trackY - 66.0f});
            window.draw(stTitle);

            sf::Text stSub(font, "ID:" + to_string(st->getStationID()) + " | " +
                           to_string(static_cast<int>(st->getKmMarker())) + "km (" + to_string(maxTracks) + "R)", 10);
            stSub.setFillColor(Theme::TextSecondary);
            sf::FloatRect sb = stSub.getLocalBounds();
            stSub.setPosition({posX - sb.size.x / 2.0f, trackY - 42.0f});
            window.draw(stSub);

            // 4. Đèn tín hiệu nhà ga (Xanh / Đỏ nhấp nháy)
            sf::Color sigColor = stationOccupied ? Theme::SignalRed : Theme::SignalGreen;
            float pulse = stationOccupied ? (0.55f + 0.45f * std::sin(animTime * 7.0f)) : 1.0f;

            sf::CircleShape sigGlow(10.0f);
            sigGlow.setOrigin({10.0f, 10.0f});
            sigGlow.setPosition({posX + 38.0f, trackY - 22.0f});
            sigGlow.setFillColor(sf::Color(sigColor.r, sigColor.g, sigColor.b, static_cast<uint8_t>(90 * pulse)));
            window.draw(sigGlow);

            sf::CircleShape sigLamp(4.5f);
            sigLamp.setOrigin({4.5f, 4.5f});
            sigLamp.setPosition({posX + 38.0f, trackY - 22.0f});
            sigLamp.setFillColor(sf::Color(sigColor.r, sigColor.g, sigColor.b, static_cast<uint8_t>(255 * pulse)));
            sigLamp.setOutlineThickness(1.5f);
            sigLamp.setOutlineColor(sf::Color(20, 20, 25));
            window.draw(sigLamp);
        }

        // ==================== C. VẼ ĐƯỜNG RAY CHÍNH & CÁC TÂM GA ====================
        window.draw(mainTrackVA);

        for (LinkedList<StationNode*>::Node* station = stationList->getHead();
             station != nullptr; station = station->next) {
            StationNode* st = station->data;
            float posX = marginX + static_cast<float>(st->getKmMarker() / maxKm) * trackWidth;

            sf::CircleShape outerNode(9.0f);
            outerNode.setOrigin({9.0f, 9.0f});
            outerNode.setPosition({posX, trackY});
            outerNode.setFillColor(Theme::BgCanvas);
            outerNode.setOutlineThickness(2.5f);
            outerNode.setOutlineColor(Theme::StationCore);
            window.draw(outerNode);

            sf::CircleShape innerNode(4.0f);
            innerNode.setOrigin({4.0f, 4.0f});
            innerNode.setPosition({posX, trackY});
            innerNode.setFillColor(Theme::StationCore);
            window.draw(innerNode);
        }

             trainNode != nullptr; trainNode = trainNode->next) {

            double currentKm = calculateTrainKmSmooth(train, stationList, currentMinute, trackUsed);

            if (currentKm >= 0.0) {

                float posY = getSmoothY(train->getTrainCode(), targetY, dt);

                if (trackUsed == 1) bodyColor = Theme::TrainMainStop;
                else if (trackUsed > 1) bodyColor = Theme::TrainSidingWait;

                if (trackUsed == 0) {

                    sf::ConvexShape beam;
                    beam.setPointCount(3);
                    beam.setPoint(0, {posX + dirSign * 26.0f, posY});
                    beam.setPoint(1, {posX + dirSign * 72.0f, posY - 12.0f});

                shadowShape.setFillColor(Theme::TrainShadow);

                trainLabel.setFillColor(sf::Color(15, 18, 25));
                sf::FloatRect lb = trainLabel.getLocalBounds();
                trainLabel.setPosition({posX - lb.size.x / 2.0f, posY - 0.5f});
                window.draw(trainLabel);

                // Nhãn hướng chạy nổi phía trên đoàn tàu
                sf::Text dirTag(font, (train->getDirection() == 1 ? code + " ->" : "<- " + code), 11);
                dirTag.setFillColor(Theme::TextPrimary);
                sf::FloatRect db = dirTag.getLocalBounds();
                dirTag.setPosition({posX - db.size.x / 2.0f, posY - 26.0f});
                window.draw(dirTag);

        topHud.setPosition({WINDOW_WIDTH / 2.0f, 38.0f});
        topHud.setFillColor(Theme::HudPanelBg);
        topHud.setOutlineThickness(1.5f);
        topHud.setOutlineColor(Theme::HudBorder);
        window.draw(topHud);

        int hours = (static_cast<int>(currentMinute) / 60) % 24;
        int mins  = static_cast<int>(currentMinute) % 60;
        int secs  = static_cast<int>((currentMinute - std::floor(currentMinute)) * 60.0f);

        stringstream clockSS;
        clockSS << setfill('0') << setw(2) << hours << ":"
                << setfill('0') << setw(2) << mins << ":"
                << setfill('0') << setw(2) << secs;

        sf::Text clockText(font, "TIME: " + clockSS.str(), 20);
        clockText.setFillColor(Theme::AccentGold);
        clockText.setPosition({45.0f, 24.0f});
        window.draw(clockText);

        stringstream ctrlSS;
        ctrlSS << (isPaused ? "[TAM DUNG]" : "[DANG CHAY]")
               << "  |  TOC DO: " << static_cast<int>(timeSpeed) << "x"
               << "  |  [SPACE] Pause  [UP/DOWN] Toc do  [LEFT/RIGHT] Tua +/-15p  [R] Reset  [ESC] Thoat";
        sf::Text ctrlText(font, ctrlSS.str(), 13);
        ctrlText.setFillColor(Theme::TextPrimary);
        ctrlText.setPosition({255.0f, 28.0f});
        window.draw(ctrlText);

        // Chú giải màu sắc (Legend)
        auto drawLegendItem = [&](float lx, float ly, sf::Color col, const string& textStr) {

            dot.setPosition({lx, ly + 2.0f});

            sf::Text lbl(font, textStr, 12);
            lbl.setFillColor(Theme::TextSecondary);

        };
        drawLegendItem(45.0f, 76.0f, Theme::TrainRunning, "Chay khu gian (Cyan)");

        logPanel.setOutlineThickness(1.5f);
        logPanel.setOutlineColor(Theme::HudBorder);

        sf::Text logTitle(font, "NHAT KY DIEU PHOI & GIAI QUYET XUNG DOT (MEET & PASS - CIRCULAR BUFFER LOGS)", 13);

        logTitle.setPosition({38.0f, WINDOW_HEIGHT - logH - 8.0f});
        window.draw(logTitle);

        int totalLogs = trainList->getLogCount();
        if (totalLogs == 0) {
            sf::Text emptyLog(font, "-> He thong dang van hanh on dinh. Chua phat sinh xung dot moi hoac da giai toa xong.", 12);
            emptyLog.setFillColor(Theme::TextSecondary);

        // 3. Bảng Giám sát Trạng thái Đoàn tàu Thời gian thực (Bottom-Right Telemetry Panel)

        for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();

            TrainNode* tr = trainNode->data;

            double km = calculateTrainKmSmooth(tr, stationList, currentMinute, trk);
            string statusStr = (trk == 0) ? "Chay khu gian" : (trk == 1 ? "Do Ray Chinh (R1)" : "Tranh Ray Phu (R" + to_string(trk) + ")");

            stringstream rowSS;
            rowSS << left << setw(6) << tr->getTrainCode()
                  << " | Uu tien P" << tr->getPriority()
                  << " | " << (tr->getDirection() == 1 ? "Nam" : "Bac")
                  << " | Km " << setw(5) << static_cast<int>(std::max(0.0, km))
                  << " | " << statusStr;

            sf::Text rowTxt(font, rowSS.str(), 12);
            rowTxt.setFillColor(trk > 1 ? Theme::TrainSidingWait : (trk == 1 ? Theme::TrainMainStop : Theme::TrainRunning));
            rowTxt.setPosition({teleLeft, ty});
            window.draw(rowTxt);
            ty += 22.0f;
        }

        window.display();
    }
}

// ==================== 1. HỘP THOẠI NHẬP LIỆU HIỆN ĐẠI (MODAL INPUT DIALOG) ====================
bool GuiMenu::showInputDialog(const string& title, const string& promptLabel, string& outputValue, sf::Font& font) {
    sf::RenderWindow dialog(sf::VideoMode({520, 230}), title, sf::Style::Titlebar | sf::Style::Close);
    dialog.setFramerateLimit(60);

    string inputText = "";

    sf::RectangleShape topBanner(sf::Vector2f(520.0f, 38.0f));
    topBanner.setFillColor(GuiTheme::CardBg);
    sf::Text bannerTitle(font, "NHAP THONG TIN HE THONG", 13);
    bannerTitle.setFillColor(GuiTheme::AccentGold);
    bannerTitle.setPosition(sf::Vector2f(20.0f, 10.0f));

    sf::Text promptText(font, promptLabel, 14);
    promptText.setFillColor(GuiTheme::TextLight);
    promptText.setPosition(sf::Vector2f(25.0f, 52.0f));

    sf::RectangleShape inputBox(sf::Vector2f(470.0f, 42.0f));
    inputBox.setPosition(sf::Vector2f(25.0f, 82.0f));
    inputBox.setFillColor(GuiTheme::BgMain);
    inputBox.setOutlineThickness(2.0f);
    inputBox.setOutlineColor(GuiTheme::AccentCyan);

    sf::Text typedText(font, "", 16);
    typedText.setFillColor(GuiTheme::AccentGold);
    typedText.setPosition(sf::Vector2f(35.0f, 92.0f));

    GuiButton btnConfirm;
    btnConfirm.init(75.0f, 150.0f, 175.0f, 40.0f, "XAC NHAN (ENTER)", font, 99, false);
    GuiButton btnCancel;
    btnCancel.init(270.0f, 150.0f, 175.0f, 40.0f, "HUY BO (ESC)", font, 98, true);

    sf::Clock blinkClock;

    while (dialog.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(dialog);
        bool mouseClicked = false;

        while (const auto event = dialog.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                dialog.close();
                return false;
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    dialog.close();
                    return false;
                }
            }
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
            if (const auto* textEntered = event->getIf<sf::Event::TextEntered>()) {
                uint32_t unicode = textEntered->unicode;
                if (unicode == 8) { // Backspace
                    if (!inputText.empty()) inputText.pop_back();
                } else if (unicode == 13) { // Enter
                    if (!inputText.empty()) {
                        outputValue = inputText;
                        dialog.close();
                        return true;
                    }
                } else if (unicode >= 32 && unicode <= 126) {
                    inputText += static_cast<char>(unicode);
                }
            }
        }

        btnConfirm.updateHover(mousePos);
        btnCancel.updateHover(mousePos);

        if (btnConfirm.isClicked(mouseClicked) && !inputText.empty()) {
            outputValue = inputText;
            dialog.close();
            return true;
        }
        if (btnCancel.isClicked(mouseClicked)) {
            dialog.close();
            return false;
        }

        bool showCursor = (static_cast<int>(blinkClock.getElapsedTime().asSeconds() * 2.0f) % 2 == 0);
        typedText.setString(inputText + (showCursor ? "_" : ""));

        dialog.clear(GuiTheme::PanelBg);
        dialog.draw(topBanner);
        dialog.draw(bannerTitle);
        dialog.draw(promptText);
        dialog.draw(inputBox);
        dialog.draw(typedText);
        btnConfirm.draw(dialog);
        btnCancel.draw(dialog);
        dialog.display();
    }
    return false;
}

// ==================== 2. SƠ ĐỒ GHẾ 2D TƯƠNG TÁC (CLICK ĐỂ ĐẶT VÉ TRỰC TIẾP) ====================
void GuiMenu::showSeatMapUI(StationList& stationList, TrainList& trainList, TicketList& ticketList, sf::Font& font, string& outputLog) {
    string trainCode, strCarriage, strFrom, strTo;
    if (!showInputDialog("SEAT MAP 2D", "1/4. Nhap Ma Tau (Vi du: SE1):", trainCode, font) ||
        !showInputDialog("SEAT MAP 2D", "2/4. Nhap So Toa (Vi du: 1):", strCarriage, font) ||
        !showInputDialog("SEAT MAP 2D", "3/4. Nhap ID Ga Di (Vi du: 101):", strFrom, font) ||
        !showInputDialog("SEAT MAP 2D", "4/4. Nhap ID Ga Den (Vi du: 105):", strTo, font)) return;

    int carriageNo = 0, fromSt = 0, toSt = 0;
    try {
        carriageNo = stoi(strCarriage);
        fromSt = stoi(strFrom);
        toSt = stoi(strTo);
    } catch (...) {
        outputLog = "-> [LOI NHAP LIEU] So toa va ID ga phai la so nguyen hop le!";
        return;
    }

    TrainNode* train = trainList.findTrainByCode(trainCode);
    if (!train) {
        outputLog = "-> [LOI TRA CUU] Khong tim thay doan tau co ma: " + trainCode;
        return;
    }

    int totalSeats = train->getCarriageSeats(carriageNo);
    if (totalSeats <= 0 || totalSeats > 128) {
        outputLog = "-> [LOI TOA XE] Toa so " + to_string(carriageNo) + " khong ton tai tren tau " + trainCode + "!";
        return;
    }

    // Kiểm tra sơ đồ ghế bằng truy vấn cây phân đoạn theo từng ghế.
    bool seatFree[129];
    if (!ticketList.checkCarriageSeats(&stationList, &trainList, trainCode, carriageNo, fromSt, toSt, seatFree, totalSeats)) {
        outputLog = "-> [LOI HANH TRINH] Tau " + trainCode + " KHONG phuc vu chang Ga " + to_string(fromSt) +
                    " -> Ga " + to_string(toSt) + "\n   (Nguyen nhan: Nhap nguoc chieu chay cua tau hoac tau khong dung tai ga nay).";
        return;
    }

    StationNode* stFromNode = stationList.findStationByID(fromSt);
    StationNode* stToNode   = stationList.findStationByID(toSt);
    string routeDesc = (stFromNode ? stFromNode->getName() : to_string(fromSt)) + " -> " +
                       (stToNode ? stToNode->getName() : to_string(toSt));

    sf::RenderWindow mapWin(sf::VideoMode({880, 520}), "PBL2: SO DO GHE NGOI TUONG TAC (INTERACTIVE SEAT MAP 2D)", sf::Style::Titlebar | sf::Style::Close);
    mapWin.setFramerateLimit(60);

    string statusBanner = "MEO: Click chuot trai truc tiep vao o GHE XANH de dat ve nhanh cho ghe do!";

    while (mapWin.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(mapWin);
        bool mouseClicked = false;

        while (const auto event = mapWin.pollEvent()) {
            if (event->is<sf::Event::Closed>()) mapWin.close();
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
        }

        mapWin.clear(GuiTheme::BgMain);

        // Header thông tin toa tàu
        sf::Text header(font, "SO DO GHE TOA " + to_string(carriageNo) + " | TAU " + trainCode +
                        " (P" + to_string(train->getPriority()) + ") | CHANG: " + routeDesc, 16);
        header.setFillColor(GuiTheme::AccentGold);
        header.setPosition(sf::Vector2f(35.0f, 20.0f));
        mapWin.draw(header);

        // Chú giải (Legend)
        int freeCount = 0;
        for (int i = 1; i <= totalSeats; ++i) if (seatFree[i]) freeCount++;

        sf::Text subHeader(font, "Tong so ghe: " + to_string(totalSeats) +
                           "  |  Con trong (Xanh): " + to_string(freeCount) +
                           "  |  Da dat (Do): " + to_string(totalSeats - freeCount), 13);
        subHeader.setFillColor(GuiTheme::AccentCyan);
        subHeader.setPosition(sf::Vector2f(35.0f, 48.0f));
        mapWin.draw(subHeader);

        // Khung thân toa tàu (Coach Hull)
        sf::RectangleShape carriageHull(sf::Vector2f(810.0f, 330.0f));
        carriageHull.setPosition(sf::Vector2f(35.0f, 85.0f));
        carriageHull.setFillColor(GuiTheme::PanelBg);
        carriageHull.setOutlineThickness(2.0f);
        carriageHull.setOutlineColor(GuiTheme::Border);
        mapWin.draw(carriageHull);

        // Lối đi giữa toa (Center Aisle)
        sf::RectangleShape aisle(sf::Vector2f(790.0f, 26.0f));
        aisle.setPosition(sf::Vector2f(45.0f, 237.0f));
        aisle.setFillColor(GuiTheme::BgMain);
        mapWin.draw(aisle);

        sf::Text aisleText(font, "--- LOI DI GIUA TOA TAU (AISLE) ---", 11);
        aisleText.setFillColor(GuiTheme::TextMuted);
        aisleText.setPosition(sf::Vector2f(330.0f, 243.0f));
        mapWin.draw(aisleText);

        // Vẽ ma trận ghế và bắt sự kiện click đặt vé trực tiếp
        int hoveredSeat = -1;
        for (int i = 1; i <= totalSeats; ++i) {
            int row = (i - 1) % 4;
            int col = (i - 1) / 4;
            float x = 55.0f + col * 48.0f;
            float y = 105.0f + row * 62.0f + (row >= 2 ? 32.0f : 0.0f);

            sf::RectangleShape seatBox(sf::Vector2f(40.0f, 46.0f));
            seatBox.setPosition(sf::Vector2f(x, y));

            bool available = seatFree[i];
            bool isHover = seatBox.getGlobalBounds().contains(sf::Vector2f(static_cast<float>(mousePos.x), static_cast<float>(mousePos.y)));
            if (isHover) hoveredSeat = i;

            seatBox.setFillColor(available ? sf::Color(39, 174, 96) : sf::Color(192, 57, 43));
            seatBox.setOutlineThickness(isHover ? 2.5f : 1.0f);
            seatBox.setOutlineColor(isHover ? GuiTheme::AccentGold : sf::Color(20, 20, 25));
            mapWin.draw(seatBox);

            // Gối tựa đầu ghế
            sf::RectangleShape headrest(sf::Vector2f(8.0f, 34.0f));
            headrest.setPosition(sf::Vector2f(x + 30.0f, y + 6.0f));
            headrest.setFillColor(available ? sf::Color(46, 204, 113) : sf::Color(231, 76, 60));
            mapWin.draw(headrest);

            sf::Text seatNum(font, to_string(i), 12);
            seatNum.setFillColor(sf::Color::White);
            seatNum.setPosition(sf::Vector2f(x + 8.0f, y + 14.0f));
            mapWin.draw(seatNum);

            // Nếu người dùng click vào ghế trống -> Mở hộp thoại nhập Mã Vé để đặt luôn!
            if (isHover && mouseClicked && available) {
                string newTicketID;
                if (showInputDialog("DAT VE NHANH TREN SO DO GHE",
                                    "Nhap Ma Ve moi cho Ghe so " + to_string(i) + " (VD: TCK305):",
                                    newTicketID, font)) 
                {
                    if (ticketList.findTicketByID(newTicketID) != nullptr) {
                        statusBanner = "[LOI] Ma ve " + newTicketID + " da ton tai trong he thong!";
                    } else if (ticketList.isSeatAvailable(&stationList, &trainList, trainCode, carriageNo, i, fromSt, toSt)) {
                        ticketList.addTicket(newTicketID, trainCode, carriageNo, i, fromSt, toSt);
                        FileManager::saveTickets("tickets.txt", ticketList);
                        seatFree[i] = false; // Cập nhật ngay trên giao diện
                        statusBanner = "[THANH CONG] Da phat hanh ve " + newTicketID + " cho Ghe " + to_string(i) + "!";
                        outputLog = "-> [SEAT MAP BOOKING] Da dat ve " + newTicketID + " (Tau " + trainCode +
                                    ", Toa " + to_string(carriageNo) + ", Ghe " + to_string(i) + ")";
                    }
                }
            }
        }

        // Thanh trạng thái dưới đáy cửa sổ Seat Map
        sf::RectangleShape footerBox(sf::Vector2f(810.0f, 45.0f));
        footerBox.setPosition(sf::Vector2f(35.0f, 435.0f));
        footerBox.setFillColor(GuiTheme::CardBg);
        footerBox.setOutlineThickness(1.0f);
        footerBox.setOutlineColor(GuiTheme::Border);
        mapWin.draw(footerBox);

        string hoverInfo = (hoveredSeat != -1)
            ? ("Dang chon Ghe so " + to_string(hoveredSeat) + ": " + (seatFree[hoveredSeat] ? "CON TRONG (Click de dat)" : "DA CO KHACH DAT"))
            : statusBanner;
        sf::Text footerTxt(font, hoverInfo, 13);
        footerTxt.setFillColor(GuiTheme::AccentGold);
        footerTxt.setPosition(sf::Vector2f(50.0f, 448.0f));
        mapWin.draw(footerTxt);

        mapWin.display();
    }
}

// ==================== 3. GIAO DIỆN LAN TRUYỀN TRỄ & TÁI LẬP LỊCH ====================
void GuiMenu::applyDelayUI(StationList& stationList, TrainList& trainList, sf::Font& font, string& outputLog) {
    string trainCode, strStID, strDelay;
    if (showInputDialog("DELAY PROPAGATION", "1/3. Nhap Ma Tau bi su co tre (Vi du: SE1):", trainCode, font) &&
        showInputDialog("DELAY PROPAGATION", "2/3. Nhap ID Ga bat dau bi tre (Vi du: 102):", strStID, font) &&
        showInputDialog("DELAY PROPAGATION", "3/3. Nhap So phut tre (Vi du: 30):", strDelay, font)) 
    {
        try {
            int stID = stoi(strStID);
            int delayMins = stoi(strDelay);
            if (trainList.applyDelayAndReschedule(trainCode, stID, delayMins, &stationList)) {
                FileManager::saveTrains("trains.txt", trainList);
                outputLog = "=== KET QUA LAN TRUYEN TRE & TAI LAP LICH (EVENT-DRIVEN) ===\n\n";
                outputLog += " -> Da cong them " + to_string(delayMins) + " phut cho tau " + trainCode + " tu Ga ID " + to_string(stID) + ".\n";
                outputLog += " -> Da kich hoat bo lap lich theo su kien (xet Priority & maxTracks).\n\n";
                outputLog += "Cac xung dot vua duoc giai quyet:\n";
                for (int i = 0; i < trainList.getLogCount(); ++i) {
                    outputLog += "   + " + trainList.getLog(i) + "\n";
                }
            } else {
                outputLog = "-> [THAT BAI] Khong tim thay Tau " + trainCode + " hoac Ga " + strStID + " khong nam trong lich trinh!";
            }
        } catch (...) {
            outputLog = "-> [LOI] ID Ga va So phut tre phai la so nguyen!";
        }
    }
}

#pragma once

#include <string>

struct IntHash {
    unsigned long operator()(int key) const {
        return static_cast<unsigned long>(static_cast<unsigned int>(key));
    }
};

struct StringHash {
    unsigned long operator()(const std::string& key) const {
        unsigned long hash = 2166136261UL;
        for (std::string::size_type i = 0; i < key.size(); ++i) {
            hash ^= static_cast<unsigned char>(key[i]);
            hash *= 16777619UL;
        }
        return hash;
    }
};

template <typename K, typename V, typename HashFunc>
class HashMap {
private:
    struct Entry {
        K key;
        V value;
        Entry* next;

        Entry(const K& entryKey, const V& entryValue)
            : key(entryKey), value(entryValue), next(nullptr) {}
    };

    Entry** buckets;
    int capacity;
    int size;
    HashFunc hash;

    int bucketFor(const K& key, int bucketCount) const {
        return static_cast<int>(hash(key) % static_cast<unsigned long>(bucketCount));
    }

    void rehash(int newCapacity) {
        Entry** newBuckets = new Entry*[newCapacity];
        for (int i = 0; i < newCapacity; ++i) {
            newBuckets[i] = nullptr;
        }

        for (int i = 0; i < capacity; ++i) {
            Entry* entry = buckets[i];
            while (entry) {
                Entry* next = entry->next;
                const int index = bucketFor(entry->key, newCapacity);
                entry->next = newBuckets[index];
                newBuckets[index] = entry;
                entry = next;
            }
        }

        delete[] buckets;
        buckets = newBuckets;
        capacity = newCapacity;
    }

public:
    explicit HashMap(int initialCapacity = 16, const HashFunc& hashFunc = HashFunc())
        : buckets(nullptr), capacity(initialCapacity < 4 ? 4 : initialCapacity),
          size(0), hash(hashFunc) {
        buckets = new Entry*[capacity];
        for (int i = 0; i < capacity; ++i) {
            buckets[i] = nullptr;
        }
    }

    ~HashMap() {
        clear();
        delete[] buckets;
    }

    HashMap(const HashMap&) = delete;
    HashMap& operator=(const HashMap&) = delete;

    void clear() {
        for (int i = 0; i < capacity; ++i) {
            Entry* entry = buckets[i];
            while (entry) {
                Entry* next = entry->next;
                delete entry;
                entry = next;
            }
            buckets[i] = nullptr;
        }
        size = 0;
    }

    bool insert(const K& key, const V& value) {
        Entry* existing = findEntry(key);
        if (existing) {
            existing->value = value;
            return false;
        }

        if ((size + 1) * 4 > capacity * 3) {
            rehash(capacity * 2);
        }

        const int index = bucketFor(key, capacity);
        Entry* entry = new Entry(key, value);
        entry->next = buckets[index];
        buckets[index] = entry;
        ++size;
        return true;
    }

    V* find(const K& key) {
        Entry* entry = findEntry(key);
        return entry ? &entry->value : nullptr;
    }

    const V* find(const K& key) const {
        const Entry* entry = findEntry(key);
        return entry ? &entry->value : nullptr;
    }

    bool remove(const K& key) {
        const int index = bucketFor(key, capacity);
        Entry* previous = nullptr;
        Entry* entry = buckets[index];
        while (entry) {
            if (entry->key == key) {
                if (previous) {
                    previous->next = entry->next;
                } else {
                    buckets[index] = entry->next;
                }
                delete entry;
                --size;
                return true;
            }
            previous = entry;
            entry = entry->next;
        }
        return false;
    }

    int getSize() const {
        return size;
    }

private:
    Entry* findEntry(const K& key) {
        Entry* entry = buckets[bucketFor(key, capacity)];
        while (entry) {
            if (entry->key == key) {
                return entry;
            }
            entry = entry->next;
        }
        return nullptr;
    }

    const Entry* findEntry(const K& key) const {
        const Entry* entry = buckets[bucketFor(key, capacity)];
        while (entry) {
            if (entry->key == key) {
                return entry;
            }
            entry = entry->next;
        }
        return nullptr;
    }
};

    }

    LinkedList(const LinkedList&) = delete;
    LinkedList& operator=(const LinkedList&) = delete;

    void pushBack(const T& value) {
        Node* node = new Node(value);
        if (!tail) {
            head = tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
        ++size;
    }

    void clear() {
        while (head) {
            Node* node = head;
            head = head->next;
            delete node;
        }
        tail = nullptr;
        size = 0;
    }

    int getSize() const {
        return size;
    }

    Node* getHead() const {
        return head;
    }

    Node* getTail() const {
        return tail;
    }

    bool removeNode(Node* target) {
        if (!target) {
            return false;
        }
        Node* previous = nullptr;
        Node* current = head;
        while (current && current != target) {
            previous = current;
            current = current->next;
        }
        if (!current) {
            return false;
        }
        if (previous) {
            previous->next = current->next;
        } else {
            head = current->next;
        }
        if (tail == current) {
            tail = previous;
        }
        delete current;
        --size;
        return true;
    }
};

#pragma once

#include <stdexcept>

template <typename T>
struct MinHeapLess {
    bool operator()(const T& left, const T& right) const {
        return left < right;
    }
};

template <typename T, typename Comparator = MinHeapLess<T>>
class MinHeap {
private:
    T* items;
    int itemCount;
    int capacity;
    Comparator comesBefore;

    void grow() {
        const int newCapacity = capacity == 0 ? 8 : capacity * 2;
        T* expanded = new T[newCapacity];
        for (int i = 0; i < itemCount; ++i) {
            expanded[i] = items[i];
        }
        delete[] items;
        items = expanded;
        capacity = newCapacity;
    }

public:
    explicit MinHeap(const Comparator& comparator = Comparator())
        : items(nullptr), itemCount(0), capacity(0), comesBefore(comparator) {}

    ~MinHeap() {
        clear();
    }

    MinHeap(const MinHeap&) = delete;
    MinHeap& operator=(const MinHeap&) = delete;

    void clear() {
        delete[] items;
        items = nullptr;
        itemCount = 0;
        capacity = 0;
    }

    bool isEmpty() const {
        return itemCount == 0;
    }

    int size() const {
        return itemCount;
    }

    const T& top() const {
        if (isEmpty()) {
            throw std::out_of_range("MinHeap::top called on an empty heap");
        }
        return items[0];
    }

    void push(const T& value) {
        if (itemCount == capacity) {
            grow();
        }

        int index = itemCount++;
        items[index] = value;
        while (index > 0) {
            const int parent = (index - 1) / 2;
            if (!comesBefore(items[index], items[parent])) {
                break;
            }
            T temp = items[index];
            items[index] = items[parent];
            items[parent] = temp;
            index = parent;
        }
    }

    void pop() {
        if (isEmpty()) {
            return;
        }

        --itemCount;
        if (itemCount == 0) {
            return;
        }

        items[0] = items[itemCount];
        int index = 0;
        while (true) {
            const int left = index * 2 + 1;
            const int right = left + 1;
            int best = index;

            if (left < itemCount && comesBefore(items[left], items[best])) {
                best = left;
            }
            if (right < itemCount && comesBefore(items[right], items[best])) {
                best = right;
            }
            if (best == index) {
                break;
            }

            T temp = items[index];
            items[index] = items[best];
            items[best] = temp;
            index = best;
        }
    }
};

#pragma once

template <typename T>
class SegmentTree {
private:
    T* minimums;
    T* lazyValues;
    int segmentCount;

    void build(int node, int left, int right, const T& initialValue) {
        minimums[node] = initialValue;
        lazyValues[node] = T();
        if (left == right) {
            return;
        }
        const int middle = left + (right - left) / 2;
        build(node * 2, left, middle, initialValue);
        build(node * 2 + 1, middle + 1, right, initialValue);
    }

    void apply(int node, const T& value) {
        minimums[node] += value;
        lazyValues[node] += value;
    }

    void pushDown(int node) {
        if (lazyValues[node] == T()) {
            return;
        }
        apply(node * 2, lazyValues[node]);
        apply(node * 2 + 1, lazyValues[node]);
        lazyValues[node] = T();
    }

    void update(int node, int left, int right, int queryLeft, int queryRight,
                const T& value) {
        if (queryLeft <= left && right <= queryRight) {
            apply(node, value);
            return;
        }

        pushDown(node);
        const int middle = left + (right - left) / 2;
        if (queryLeft <= middle) {
            update(node * 2, left, middle, queryLeft, queryRight, value);
        }
        if (queryRight > middle) {
            update(node * 2 + 1, middle + 1, right, queryLeft, queryRight, value);
        }
        minimums[node] = minimums[node * 2] < minimums[node * 2 + 1]
            ? minimums[node * 2] : minimums[node * 2 + 1];
    }

    T query(int node, int left, int right, int queryLeft, int queryRight) {
        if (queryLeft <= left && right <= queryRight) {
            return minimums[node];
        }

        pushDown(node);
        const int middle = left + (right - left) / 2;
        bool hasResult = false;
        T result = T();

        if (queryLeft <= middle) {
            result = query(node * 2, left, middle, queryLeft, queryRight);
            hasResult = true;
        }
        if (queryRight > middle) {
            const T rightResult = query(node * 2 + 1, middle + 1, right,
                                        queryLeft, queryRight);
            if (!hasResult || rightResult < result) {
                result = rightResult;
            }
        }
        return result;
    }

public:
    SegmentTree() : minimums(nullptr), lazyValues(nullptr), segmentCount(0) {}

    explicit SegmentTree(int count, const T& initialValue = T())
        : minimums(nullptr), lazyValues(nullptr), segmentCount(0) {
        initialize(count, initialValue);
    }

    ~SegmentTree() {
        clear();
    }

    SegmentTree(const SegmentTree&) = delete;
    SegmentTree& operator=(const SegmentTree&) = delete;

    void clear() {
        delete[] minimums;
        delete[] lazyValues;
        minimums = nullptr;
        lazyValues = nullptr;
        segmentCount = 0;
    }

    void initialize(int count, const T& initialValue = T()) {
        clear();
        if (count <= 0) {
            return;
        }
        segmentCount = count;
        minimums = new T[4 * count];
        lazyValues = new T[4 * count];
        build(1, 0, count - 1, initialValue);
    }

    int size() const {
        return segmentCount;
    }

    T queryMin(int left, int right) {
        if (segmentCount == 0 || left < 0 || right >= segmentCount || left > right) {
            return T();
        }
        return query(1, 0, segmentCount - 1, left, right);
    }

    void updateRange(int left, int right, const T& value) {
        if (segmentCount == 0 || left < 0 || right >= segmentCount || left > right) {
            return;
        }
        update(1, 0, segmentCount - 1, left, right, value);
    }
};

void TicketList::addTicket(const string& ticketID, const string& trainCode, 
                           int carriageNo, int seatNo, int fromID, int toID) {
    TicketNode* newNode = new TicketNode(ticketID, trainCode, carriageNo, seatNo, fromID, toID);

        builtOccupancyRevision == occupancyRevision && occupancyTrainList &&
        builtTrainRevision == occupancyTrainList->getStructureRevision();

        ticketsByID.insert(ticketID, newNode);

bool TicketList::cancelTicket(const string& ticketID) {
    TicketNode* const* indexedTicket = ticketsByID.find(ticketID);
    if (!indexedTicket) return false;

    LinkedList<TicketNode*>::Node* target = tickets.getHead();
    while (target && target->data != *indexedTicket) {
        target = target->next;
    }
    if (!target) return false;

    const bool cacheWasCurrent =
        builtOccupancyRevision == occupancyRevision && occupancyTrainList &&
        builtTrainRevision == occupancyTrainList->getStructureRevision();
    TicketNode* ticket = target->data;
    if (cacheWasCurrent) updateSeatOccupancyForTicket(ticket, 1);
    tickets.removeNode(target);
    delete ticket;
    ticketsByID.remove(ticketID);
    for (LinkedList<TicketNode*>::Node* remaining = tickets.getHead();
         remaining; remaining = remaining->next) {
        if (remaining->data->getTicketID() == ticketID) {
            ticketsByID.insert(ticketID, remaining->data);
            break;
        }
    }
    ++occupancyRevision;
    if (cacheWasCurrent) builtOccupancyRevision = occupancyRevision;
    return true;
}

void TicketList::updateSeatOccupancyForTicket(const TicketNode* ticket,
                                               int delta) const {
    if (!ticket || !occupancyTrainList) return;
    TrainNode* train = occupancyTrainList->findTrainByCode(ticket->getTrainCode());
    if (!train) return;

    const int fromIndex = findScheduleIndex(train, ticket->getFromStationID());
    const int toIndex = findScheduleIndex(train, ticket->getToStationID());
    CarriageNode* carriage = train->findCarriage(ticket->getCarriageNo());
    if (carriage && fromIndex >= 0 && toIndex > fromIndex &&
        ticket->getSeatNo() >= 1 && ticket->getSeatNo() <= carriage->getTotalSeats()) {
        carriage->updateSeatSegments(ticket->getSeatNo(), fromIndex,
                                     toIndex - 1, delta);
    }
}

void TicketList::ensureSeatOccupancy(TrainList* trainList) const {
    if (!trainList ||
        (builtOccupancyRevision == occupancyRevision && occupancyTrainList == trainList &&
         builtTrainRevision == trainList->getStructureRevision())) {
        return;
    }

    // Reset each seat's route tree once, then replay tickets in one pass.
    for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
         trainNode; trainNode = trainNode->next) {
        TrainNode* train = trainNode->data;
        const int segmentCount = countScheduleStops(train) - 1;
        for (LinkedList<CarriageNode*>::Node* carriageNode =
                 train->getHeadCarriage(); carriageNode;
             carriageNode = carriageNode->next) {
            CarriageNode* carriage = carriageNode->data;
            carriage->configureSeatSegments(segmentCount);
        }
    }

    for (LinkedList<TicketNode*>::Node* ticketNode = tickets.getHead();
         ticketNode; ticketNode = ticketNode->next) {
        TicketNode* ticket = ticketNode->data;
        TrainNode* train = trainList->findTrainByCode(ticket->getTrainCode());
        if (!train) continue;

        const int fromIndex = findScheduleIndex(train, ticket->getFromStationID());
        const int toIndex = findScheduleIndex(train, ticket->getToStationID());
        CarriageNode* carriage = train->findCarriage(ticket->getCarriageNo());
        if (carriage && fromIndex >= 0 && toIndex > fromIndex &&
            ticket->getSeatNo() >= 1 && ticket->getSeatNo() <= carriage->getTotalSeats()) {
            carriage->updateSeatSegments(ticket->getSeatNo(), fromIndex, toIndex - 1, -1);
        }
    }

    occupancyTrainList = trainList;
    builtOccupancyRevision = occupancyRevision;
    builtTrainRevision = trainList->getStructureRevision();
}

bool TicketList::checkCarriageSeats(StationList* stationList, TrainList* trainList,
                                    const string& trainCode, int carriageNo,
                                    int newFromID, int newToID,
                                    bool seatFreeOut[], int maxSeats) const 
{
    if (!stationList || !trainList || !seatFreeOut ||
        newFromID == newToID || maxSeats <= 0) return false;

    StationNode* stNewFrom = stationList->findStationByID(newFromID);
    StationNode* stNewTo   = stationList->findStationByID(newToID);
    if (!stNewFrom || !stNewTo) return false;

    TrainNode* train = trainList->findTrainByCode(trainCode);
    if (!train) return false;

    const int fromIndex = findScheduleIndex(train, newFromID);
    const int toIndex = findScheduleIndex(train, newToID);
    CarriageNode* carriage = train->findCarriage(carriageNo);
    if (!carriage || fromIndex < 0 || toIndex <= fromIndex ||
        maxSeats > carriage->getTotalSeats()) return false;

    ensureSeatOccupancy(trainList);
    for (int seat = 1; seat <= maxSeats; ++seat) {
        seatFreeOut[seat] =
            carriage->isSeatAvailableOnSegments(seat, fromIndex, toIndex - 1);
    }

    return true;
}

bool TicketList::isSeatAvailable(StationList* stationList, TrainList* trainList, const string& trainCode, 
                                 int carriageNo, int seatNo, int newFromID, int newToID) const 
{
    if (!stationList || !trainList || newFromID == newToID) return false;

    StationNode* stNewFrom = stationList->findStationByID(newFromID);
    StationNode* stNewTo   = stationList->findStationByID(newToID);
    if (!stNewFrom || !stNewTo) return false;

    TrainNode* train = trainList->findTrainByCode(trainCode);
    if (!train) return false;

    const int fromIndex = findScheduleIndex(train, newFromID);
    const int toIndex = findScheduleIndex(train, newToID);

    if (!carriage || seatNo <= 0 || seatNo > carriage->getTotalSeats() ||
        fromIndex < 0 || toIndex <= fromIndex) return false;

    ensureSeatOccupancy(trainList);
    return carriage->isSeatAvailableOnSegments(seatNo, fromIndex, toIndex - 1);
}

    };

    void pushTrainEvents(MinHeap<ScheduleEvent, ScheduleEventBefore>& events,
                         TrainNode* train, int& sequence, int safetyHeadway) {
        LinkedList<StopScheduleNode*>::Node* stopNode = train->getHeadSchedule();
        while (stopNode) {
            StopScheduleNode* stop = stopNode->data;
            ScheduleEvent stationEvent;
            stationEvent.time = stop->getArrivalMinute();
            stationEvent.endTime = stop->getDepartureMinute() + safetyHeadway;
            stationEvent.type = 0;
            stationEvent.stationID = stop->getStationID();
            stationEvent.targetStationID = stop->getStationID();
            stationEvent.track = stop->getTrackUsed();
            stationEvent.sequence = sequence++;
            stationEvent.trainRevision = train->getScheduleRevision();
            stationEvent.train = train;
            stationEvent.stop = stop;
            stationEvent.trainCode = train->getTrainCode();
            if (stationEvent.endTime > stationEvent.time) {
                events.push(stationEvent);
            }

            if (stopNode->next) {
                StopScheduleNode* nextStop = stopNode->next->data;
                ScheduleEvent blockEvent;
                blockEvent.time = stop->getDepartureMinute();
                blockEvent.endTime = nextStop->getArrivalMinute() + safetyHeadway;
                blockEvent.type = 1;
                blockEvent.stationID = stop->getStationID();
                blockEvent.targetStationID = nextStop->getStationID();
                blockEvent.sequence = sequence++;
                blockEvent.trainRevision = train->getScheduleRevision();
                blockEvent.train = train;
                blockEvent.stop = stop;
                blockEvent.nextStop = nextStop;
                blockEvent.trainCode = train->getTrainCode();
                if (blockEvent.endTime > blockEvent.time) {
                    events.push(blockEvent);
                }
            }
            stopNode = stopNode->next;
        }
    }

    // KHÓA TÀI NGUYÊN (Sử dụng cho HashMap của bạn)
    struct ResourceKey {
        int type; // 0: Ray trong ga, 1: Khu gian
        int id1;  
        int id2;  

        ResourceKey(int t = 0, int a = 0, int b = 0) {
            type = t;
            if (type == 1 && a > b) {
                id1 = b;
                id2 = a;
            } else {
                id1 = a;
                id2 = b;
            }
        }

        // Toán tử == để HashMap kiểm tra trùng lặp khóa
        bool operator==(const ResourceKey& other) const {
            return type == other.type && id1 == other.id1 && id2 == other.id2;
        }
    };

    // HÀM BĂM (Functor) DÀNH CHO KHÓA TÀI NGUYÊN
    struct ResourceKeyHash {
        unsigned long operator()(const ResourceKey& k) const {
            unsigned long hash = 2166136261UL;
            hash = (hash ^ static_cast<unsigned long>(k.type + 1)) * 16777619UL;
            hash = (hash ^ static_cast<unsigned long>(k.id1)) * 16777619UL;
            hash = (hash ^ static_cast<unsigned long>(k.id2)) * 16777619UL;
            return hash;
        }
    };

    // Hàm phụ trợ trích xuất Key từ sự kiện
    ResourceKey getEventKey(const ScheduleEvent& ev) {
        if (ev.type == 0) {
            return ResourceKey(0, ev.stationID, ev.track);
        }
        return ResourceKey(1, ev.stationID, ev.targetStationID);
    }
}

TrainList::TrainList()
    : trains(), trainsByCode(), structureRevision(0), logStart(0), logCount(0) {}

TrainList::~TrainList() {
    clear();
}

void TrainList::clear() {
    LinkedList<TrainNode*>::Node* train = trains.getHead();
    while (train) {
        delete train->data;
        train = train->next;
    }
    trains.clear();
    trainsByCode.clear();
    ++structureRevision;
}

LinkedList<TrainNode*>::Node* TrainList::getHead() const {
    return trains.getHead();
}

void TrainList::addTrain(const string& code, int direction, int priority) {
    TrainNode* newNode = new TrainNode(code, direction, priority);
    trains.pushBack(newNode);
    if (!trainsByCode.find(code)) {
        trainsByCode.insert(code, newNode);
    }
    ++structureRevision;
}

TrainNode* TrainList::findTrainByCode(const string& code) const {
    TrainNode* const* train = trainsByCode.find(code);
    return train ? *train : nullptr;
}

void TrainList::addCarriageToTrain(const string& code, int carriageNo,
                                   const string& type, int seats) {
    TrainNode* train = findTrainByCode(code);
    if (train) {
        train->addCarriage(carriageNo, type, seats);
        ++structureRevision;
    }
}

void TrainList::addStopScheduleToTrain(const string& code, int stationID,
                                       int arrival, int departure, int track) {
    TrainNode* train = findTrainByCode(code);
    if (train) {
        train->addSchedule(stationID, arrival, departure, track);
        ++structureRevision;
    }
}

void TrainList::addConflictLog(const string& logMessage) {
    if (logCount < MAX_LOGS) {
        const int insertIndex = (logStart + logCount) % MAX_LOGS;
        conflictLogs[insertIndex] = logMessage;
        ++logCount;
    } else {
        conflictLogs[logStart] = logMessage;
        logStart = (logStart + 1) % MAX_LOGS;
    }
}

string TrainList::getLog(int index) const {
    if (index < 0 || index >= logCount) return "";
    return conflictLogs[(logStart + index) % MAX_LOGS];
}

int TrainList::findAvailableSidingTrack(int stationID, int arrival, int departure,
                                        int maxTracks, TrainNode* excludeTrain) const {
    if (maxTracks <= 1) return -1;

    const int HEADWAY = 10;
    for (int candidateTrack = 2; candidateTrack <= maxTracks; ++candidateTrack) {
        bool occupied = false;
        for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead();
             trainNode && !occupied; trainNode = trainNode->next) {
            TrainNode* train = trainNode->data;
            if (train == excludeTrain) continue;

            for (LinkedList<StopScheduleNode*>::Node* stopNode =
                     train->getHeadSchedule(); stopNode; stopNode = stopNode->next) {
                StopScheduleNode* stop = stopNode->data;
                if (stop->getStationID() != stationID ||
                    stop->getTrackUsed() != candidateTrack) continue;

                const int overlapStart = arrival > stop->getArrivalMinute()
                    ? arrival : stop->getArrivalMinute();
                const int stopEnd = stop->getDepartureMinute() + HEADWAY;
                const int requestedEnd = departure + HEADWAY;
                const int overlapEnd = requestedEnd < stopEnd
                    ? requestedEnd : stopEnd;
                if (overlapStart < overlapEnd) {
                    occupied = true;
                    break;
                }
            }
        }
        if (!occupied) return candidateTrack;
    }
    return -1;
}

void TrainList::resolveConflictAndScheduleSiding(StationList* stationList) {
    if (!stationList) return;

    const int SAFETY_HEADWAY = 10;
    MinHeap<ScheduleEvent, ScheduleEventBefore> events;
    int sequence = 0;
    for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead();
         trainNode; trainNode = trainNode->next) {
        TrainNode* train = trainNode->data;
        pushTrainEvents(events, train, sequence, SAFETY_HEADWAY);
    }

    HashMap<ResourceKey, ScheduleEvent, ResourceKeyHash> activeResources(32);

    while (!events.isEmpty()) {
        const ScheduleEvent current = events.top();
        events.pop();
        if (!isCurrentEvent(current)) {
            continue;
        }

        ResourceKey resKey = getEventKey(current);
        ScheduleEvent* occupiedEvent = activeResources.find(resKey);

        // 1. Giải phóng tài nguyên nếu tàu cũ đã rời đi hoặc đổi lịch
        if (occupiedEvent && (!isCurrentEvent(*occupiedEvent) || occupiedEvent->endTime <= current.time)) {
            activeResources.remove(resKey);
            occupiedEvent = nullptr;
        }

        // 2. Nếu tài nguyên trống, tàu hiện tại chiếm dụng
        if (!occupiedEvent) {
            activeResources.insert(resKey, current);
            continue;
        }

        // 3. Cùng một tàu kéo dài thời gian đỗ
        if (occupiedEvent->train == current.train) {
            if (current.endTime > occupiedEvent->endTime) {
                occupiedEvent->endTime = current.endTime;
            }
            continue;
        }

        // 4. Phát hiện xung đột, kiểm tra ưu tiên nhường đường
        const ScheduleEvent other = *occupiedEvent;
        const bool currentYields =
            current.train->getPriority() != other.train->getPriority()
                ? current.train->getPriority() > other.train->getPriority()
                : (current.time != other.time
                       ? current.time > other.time
                       : current.sequence > other.sequence);
        
        const ScheduleEvent& yielding = currentYields ? current : other;
        const ScheduleEvent& priority = currentYields ? other : current;
        TrainNode* yieldingTrain = yielding.train;
        TrainNode* priorityTrain = priority.train;
        bool changed = false;

        if (yielding.type == 0) {
            StationNode* station = stationList->findStationByID(yielding.stationID);
            const int trackCount = station ? station->getMaxTracks() : 1;
            const int siding = findAvailableSidingTrack(
                yielding.stationID, yielding.stop->getArrivalMinute(),
                yielding.stop->getDepartureMinute(), trackCount, yieldingTrain);

            if (siding != -1) {
                yielding.stop->setTrackUsed(siding);
                addConflictLog("[Ga " + to_string(yielding.stationID) + "] " +
                            yieldingTrain->getTrainCode() + " vao Ray " + to_string(siding));
                changed = true;
            } else {
                const int wait = priority.endTime - yielding.time;
                if (wait > 0) {
                    StopScheduleNode* previous = yieldingTrain->findPreviousStop(yielding.stop);
                    if (previous) {
                        previous->setDepartureMinute(previous->getDepartureMinute() + wait);
                    }
                    yieldingTrain->applyDelayPropagation(yielding.stationID, wait, true);
                    changed = true;
                }
            }
        } else { // type == 1 (Xung đột khu gian đường đơn)
        const int wait = priority.endTime - yielding.time;
        if (wait > 0) {
            
            StationNode* waitStation = stationList->findStationByID(yielding.stationID);
            const int waitTracks = waitStation ? waitStation->getMaxTracks() : 1;
            const int siding = findAvailableSidingTrack(
                yielding.stationID, 
                yielding.stop->getArrivalMinute(),
                yielding.stop->getDepartureMinute() + wait, // Khung giờ dự kiến đỗ mới
                waitTracks, 
                yieldingTrain
            );
            if (siding != -1) {
                yielding.stop->setTrackUsed(siding);
                yielding.stop->setDepartureMinute(yielding.stop->getDepartureMinute() + wait);
                yieldingTrain->applyDelayPropagation(yielding.targetStationID, wait, true);
            } else {
                StopScheduleNode* previous = yieldingTrain->findPreviousStop(yielding.stop);
                if (previous) {
                    previous->setDepartureMinute(previous->getDepartureMinute() + wait);
                    yieldingTrain->applyDelayPropagation(yielding.stationID, wait, true);
                } else {
                    yielding.stop->setDepartureMinute(yielding.stop->getDepartureMinute() + wait);
                    yieldingTrain->applyDelayPropagation(yielding.targetStationID, wait, true);
                }
            }
            addConflictLog("[Khu gian " + to_string(yielding.stationID) + "->" +
                        to_string(yielding.targetStationID) + "] " +
                        yieldingTrain->getTrainCode() + " cho " +
                        priorityTrain->getTrainCode() + " +" + to_string(wait) + "p" +
                        (siding != -1 ? " (doi sang Ray " + to_string(siding) + ")" : ""));
            changed = true;
        }
    }

        if (changed) {
            // Thay thế tàu bị loại ra khỏi tài nguyên nếu cần
            if (!currentYields) {
                activeResources.insert(resKey, current);
            }
            yieldingTrain->bumpScheduleRevision();
            pushTrainEvents(events, yieldingTrain, sequence, SAFETY_HEADWAY);
        }
    }
}

bool TrainList::applyDelayAndReschedule(const string& trainCode, int fromStationID,
                                        int delayMinutes, StationList* stationList) {
    TrainNode* train = findTrainByCode(trainCode);
    if (!train) return false;

    addConflictLog("[Su co] Tau " + trainCode + " bi tre +" +
                   to_string(delayMinutes) + "p tu Ga " +
                   to_string(fromStationID));
    const bool delayed =
        train->applyDelayPropagation(fromStationID, delayMinutes, true);
    if (delayed) {
        resolveConflictAndScheduleSiding(stationList);
    }
    return delayed;
}

#pragma once
#include <string>
#include "CarriageNode.h"
#include "StopScheduleNode.h"
#include "LinkedList.h"

struct Train {
    std::string trainCode;
    int direction; // 1: Nam, 0: Bac
    int priority;  // 1: Tau khach nhanh SE, 2: Tau thuong TN, 3: Tau hang
    LinkedList<Carriage*> carriages;
    LinkedList<StopSchedule*> schedule;
    unsigned long scheduleRevision;

public:
    static int inferPriorityFromCode(const std::string& code) {
        if (code.rfind("SE", 0) == 0) return 1; // Ưu tiên cao nhất: SE1, SE2, SE3, SE4...
        if (code.rfind("TN", 0) == 0) return 2; // Ưu tiên trung bình: TN1, TN2...
        return 3;                               // Ưu tiên thấp nhất: Tàu hàng (HH, GS...)
    }

    Train(const std::string& code, int dir, int prio = -1)
        : trainCode(code), direction(dir), 
          priority(prio > 0 ? prio : inferPriorityFromCode(code)),
          scheduleRevision(0) {}

    ~Train() {
        LinkedList<Carriage*>::Node* carriage = carriages.getHead();
        while (carriage) {
            delete carriage->data;
            carriage = carriage->next;
        }
        LinkedList<StopSchedule*>::Node* stop = schedule.getHead();
        while (stop) {
            delete stop->data;
            stop = stop->next;
        }
    }

    // Getters
    std::string getTrainCode() const { return trainCode; }
    int getDirection() const { return direction; }
    int getPriority() const { return priority; }
    LinkedList<Carriage*>::Node* getHeadCarriage() const { return carriages.getHead(); }
    LinkedList<StopSchedule*>::Node* getHeadSchedule() const { return schedule.getHead(); }
    LinkedList<StopSchedule*>::Node* getTailSchedule() const { return schedule.getTail(); }
    unsigned long getScheduleRevision() const { return scheduleRevision; }

    // Setters
    void setPriority(int prio) { priority = prio; }
    void bumpScheduleRevision() { ++scheduleRevision; }

    // Chèn toa xe độ phức tạp O(1)
    void addCarriage(int cNo, const std::string& type, int seats) {
        carriages.pushBack(new CarriageNode(cNo, type, seats));
    }

    // Chèn lịch dừng độ phức tạp O(1)
    void addSchedule(int stID, int arr, int dep, int track) {
        schedule.pushBack(new StopScheduleNode(stID, arr, dep, track));
    }

    // Tìm ga dừng liền trước một ga cụ thể trong hành trình của tàu
    StopScheduleNode* findPreviousStop(StopScheduleNode* targetStop) const {
        LinkedList<StopSchedule*>::Node* curr = schedule.getHead();
        if (!curr || curr->data == targetStop) return nullptr;
        while (curr && (!curr->next || curr->next->data != targetStop)) {
            curr = curr->next;
        }
        return curr ? curr->data : nullptr;
    }

    // Lan truyền dời giờ các ga tiếp theo khi xảy ra trễ
    bool applyDelayPropagation(int fromStationID, int delayMinutes, bool delayArrivalAtFirst = true) {
        LinkedList<StopSchedule*>::Node* curr = schedule.getHead();
        bool found = false;

        while (curr != nullptr) {
            StopScheduleNode* stop = curr->data;
            if (stop->getStationID() == fromStationID) {
                found = true;
                if (delayArrivalAtFirst) {
                    stop->setArrivalMinute(stop->getArrivalMinute() + delayMinutes);
                }
                stop->setDepartureMinute(stop->getDepartureMinute() + delayMinutes);
            } else if (found) {
                stop->setArrivalMinute(stop->getArrivalMinute() + delayMinutes);
                stop->setDepartureMinute(stop->getDepartureMinute() + delayMinutes);
            }
            curr = curr->next;
        }
        return found;
    }

    int getCarriageSeats(int carriageNo) const {
        CarriageNode* carriage = findCarriage(carriageNo);
        return carriage ? carriage->getTotalSeats() : 0;
    }

    CarriageNode* findCarriage(int carriageNo) const {
        LinkedList<Carriage*>::Node* current = carriages.getHead();
        while (current) {
            CarriageNode* carriage = current->data;
            if (carriage->getCarriageNo() == carriageNo) return carriage;
            current = current->next;
        }
        return nullptr;
    }
};

typedef Train TrainNode;

int parseCommandInteger(const string& value) {
    string::size_type parsed = 0;
    const int number = stoi(value, &parsed);
    if (parsed != value.size()) throw invalid_argument("So khong hop le");
    return number;
}

double parseCommandDouble(const string& value) {
    string::size_type parsed = 0;
    const double number = stod(value, &parsed);
    if (parsed != value.size()) throw invalid_argument("So thuc khong hop le");
    return number;
}

// ==================== STATIONS ====================
void FileManager::loadStations(const string& filename, StationList& stationList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string strID, name, strKm, strTracks;

        if (getline(ss, strID, '|') && getline(ss, name, '|') &&
            getline(ss, strKm, '|') && getline(ss, strTracks, '|')) {
            stationList.addStation(stoi(strID), name, stod(strKm), stoi(strTracks));
        }
    }
    inFile.close();
}

bool FileManager::saveStations(const string& filename, const StationList& stationList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return false;

    for (LinkedList<StationNode*>::Node* station = stationList.getHead();
         station != nullptr; station = station->next) {
        StationNode* st = station->data;
        outFile << st->getStationID() << "|"
                << st->getName() << "|"
                << st->getKmMarker() << "|"
                << st->getMaxTracks() << "\n";
    }
    outFile.close();
    return !outFile.fail();
}

// ==================== TRAINS ====================
void FileManager::loadTrains(const string& filename, TrainList& trainList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    TrainNode* currentTrain = nullptr;

    while (getline(inFile, line)) {
        if (line.empty()) continue;

        stringstream ss(line);
        string token;
        getline(ss, token, '|');

        if (token == "CARRIAGE") {
            if (currentTrain) {
                string strNo, type;
                if (getline(ss, strNo, '|') && getline(ss, type, '|')) {
                    currentTrain->addCarriage(stoi(strNo), type);
                }
            }
        } 
        else if (token == "STOP") {
            if (currentTrain) {
                string strStID, strArr, strDep, strTrack;
                if (getline(ss, strStID, '|') && getline(ss, strArr, '|') &&
                    getline(ss, strDep, '|') && getline(ss, strTrack, '|')) {
                    currentTrain->addSchedule(stoi(strStID), stoi(strArr), stoi(strDep), stoi(strTrack));
                }
            }
        } 
        else {
            // Định dạng dòng đầu: Mã Tàu | Hướng chạy [ | Độ ưu tiên ]
            string trainCode = token;
            string strDir, strPrio;

            if (getline(ss, strDir, '|')) {
                int prio = -1;
                if (getline(ss, strPrio, '|') && !strPrio.empty()) {
                    try {
                        prio = stoi(strPrio);
                    } catch (...) {
                        prio = -1;
                    }
                }

                trainList.addTrain(trainCode, stoi(strDir), prio);
                currentTrain = trainList.findTrainByCode(trainCode);
            }
        }
    }
    inFile.close();
}

bool FileManager::saveTrains(const string& filename, const TrainList& trainList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return false;

    for (LinkedList<TrainNode*>::Node* trainNode = trainList.getHead();
         trainNode != nullptr; trainNode = trainNode->next) {
        TrainNode* tr = trainNode->data;
        // Lưu kèm độ ưu tiên vào cột thứ 3: Mã Tàu | Hướng chạy | Độ ưu tiên
        outFile << tr->getTrainCode() << "|" 
                << tr->getDirection() << "|" 
                << tr->getPriority() << "\n";

        for (LinkedList<CarriageNode*>::Node* carriage = tr->getHeadCarriage();
             carriage != nullptr; carriage = carriage->next) {
            CarriageNode* c = carriage->data;
            outFile << "CARRIAGE|" << c->getCarriageNo() << "|" << c->getType() << "\n";
        }

        for (LinkedList<StopScheduleNode*>::Node* stop = tr->getHeadSchedule();
             stop != nullptr; stop = stop->next) {
            StopScheduleNode* s = stop->data;
            outFile << "STOP|" << s->getStationID() << "|" << s->getArrivalMinute() << "|" << s->getDepartureMinute() << "|" << s->getTrackUsed() << "\n";
        }
    }
    outFile.close();
    return !outFile.fail();
}

bool FileManager::processInputCommands(const string& filename,
                                       StationList& stationList,
                                       TrainList& trainList,
                                       string& resultLog) {
    ifstream inFile(filename);
    if (!inFile.is_open()) {
        resultLog = "Khong mo duoc file lenh " + filename;
        return false;
    }

    bool changedStations = false;
    bool changedTrains = false;
    string line;
    int lineNumber = 0;
    resultLog.clear();
    while (getline(inFile, line)) {
        ++lineNumber;
        if (!line.empty() && line.back() == '\r') line.pop_back();
        if (line.empty() || line[0] == '#') continue;

        string fields[8];
        int fieldCount = 0;
        stringstream parser(line);
        while (fieldCount < 8 && getline(parser, fields[fieldCount], '|')) ++fieldCount;
        for (int i = 0; i < fieldCount; ++i) {
            if (!fields[i].empty() && fields[i].back() == '\r') fields[i].pop_back();
        }
        const string& command = fields[0];
        bool success = false;
        try {
            // Mỗi lệnh chỉ được áp dụng sau khi kiểm tra toàn bộ điều kiện liên quan.
            if (command == "ADD_STATION" && fieldCount == 5) {
                success = stationList.addStation(parseCommandInteger(fields[1]), fields[2],
                    parseCommandDouble(fields[3]), parseCommandInteger(fields[4]));
                changedStations = changedStations || success;
            } else if (command == "UPDATE_STATION" && fieldCount == 5) {
                const int id = parseCommandInteger(fields[1]);
                StationNode* station = stationList.findStationByID(id);
                const int tracks = parseCommandInteger(fields[4]);
                const double km = parseCommandDouble(fields[3]);
                const bool routeSafe = (!station || !trainList.hasStationStop(id) ||
                    km == station->getKmMarker()) &&
                    (!station || km == station->getKmMarker() ||
                     !trainList.hasMaintenanceBlockForStation(id));
                success = routeSafe && !trainList.hasTrackAbove(id, tracks) &&
                    !trainList.hasMaintenanceTrackAbove(id, tracks) &&
                    stationList.updateStation(id, fields[2], km, tracks);
                changedStations = changedStations || success;
            } else if (command == "DELETE_STATION" && fieldCount == 2) {
                const int id = parseCommandInteger(fields[1]);
                success = !trainList.hasStationStop(id) &&
                          !trainList.hasMaintenanceForStation(id) &&
                          stationList.removeStation(id);
                changedStations = changedStations || success;
            } else if (command == "ADD_TRAIN" && fieldCount >= 3 && fieldCount <= 4) {
                const int priority = fieldCount == 4 ? parseCommandInteger(fields[3]) : -1;
                success = trainList.addTrain(fields[1], parseCommandInteger(fields[2]), priority);
                changedTrains = changedTrains || success;
            } else if (command == "UPDATE_TRAIN" && fieldCount == 4) {
                success = trainList.updateTrain(fields[1], parseCommandInteger(fields[2]),
                    parseCommandInteger(fields[3]), &stationList);
                changedTrains = changedTrains || success;
            } else if (command == "DELETE_TRAIN" && fieldCount == 2) {
                success = trainList.removeTrain(fields[1]);
                changedTrains = changedTrains || success;
            } else if (command == "ADD_CARRIAGE" && fieldCount == 4) {
                success = trainList.addCarriageToTrain(fields[1],
                    parseCommandInteger(fields[2]), fields[3]);
                changedTrains = changedTrains || success;
            } else if (command == "UPDATE_CARRIAGE" && fieldCount == 4) {
                success = trainList.updateCarriage(fields[1],
                    parseCommandInteger(fields[2]), fields[3]);
                changedTrains = changedTrains || success;
            } else if (command == "DELETE_CARRIAGE" && fieldCount == 3) {
                const int carriageNo = parseCommandInteger(fields[2]);
                success = trainList.removeCarriage(fields[1], carriageNo);
                changedTrains = changedTrains || success;
            } else if (command == "ADD_STOP" && fieldCount == 6) {
                success = trainList.addStopScheduleToTrain(fields[1], parseCommandInteger(fields[2]),
                    parseCommandInteger(fields[3]), parseCommandInteger(fields[4]),
                    parseCommandInteger(fields[5]), &stationList);
                changedTrains = changedTrains || success;
            } else if (command == "UPDATE_STOP" && fieldCount == 6) {
                success = trainList.updateStopSchedule(fields[1], parseCommandInteger(fields[2]),
                    parseCommandInteger(fields[3]), parseCommandInteger(fields[4]),
                    parseCommandInteger(fields[5]), &stationList);
                changedTrains = changedTrains || success;
            } else if (command == "DELETE_STOP" && fieldCount == 3) {
                const int stationID = parseCommandInteger(fields[2]);
                success = trainList.removeStopSchedule(fields[1], stationID);
                changedTrains = changedTrains || success;
            } else if ((command == "CLOSE_STATION" && fieldCount == 4) ||
                       (command == "OPEN_STATION" && fieldCount == 4) ||
                       (command == "CLOSE_BLOCK" && fieldCount == 5) ||
                       (command == "OPEN_BLOCK" && fieldCount == 5) ||
                       (command == "CLOSE_TRACK" && fieldCount == 5) ||
                       (command == "OPEN_TRACK" && fieldCount == 5) ||
                       (command == "REPAIR_STATION" && fieldCount == 4) ||
                       (command == "REPAIR_BLOCK" && fieldCount == 5) ||
                       (command == "REPAIR_TRACK" && fieldCount == 5)) {
                const bool opening = command.rfind("OPEN_", 0) == 0;
                const string resource = command.find("STATION") != string::npos
                    ? "STATION" : (command.find("BLOCK") != string::npos ? "BLOCK" : "TRACK");
                const int first = parseCommandInteger(fields[1]);
                const int second = resource == "STATION" ? 0 : parseCommandInteger(fields[2]);
                const int startField = resource == "STATION" ? 2 : 3;
                const int endField = resource == "STATION" ? 3 : 4;
                const bool stationsExist = stationList.findStationByID(first) &&
                    (resource != "BLOCK" || stationList.findStationByID(second));
                bool adjacentBlock = resource != "BLOCK";
                if (resource == "BLOCK" && stationsExist) {
                    StationNode* firstStation = stationList.findStationByID(first);
                    StationNode* secondStation = stationList.findStationByID(second);
                    adjacentBlock = firstStation->getKmMarker() != secondStation->getKmMarker();
                    const double lowKm = firstStation->getKmMarker() < secondStation->getKmMarker()
                        ? firstStation->getKmMarker() : secondStation->getKmMarker();
                    const double highKm = firstStation->getKmMarker() > secondStation->getKmMarker()
                        ? firstStation->getKmMarker() : secondStation->getKmMarker();
                    for (LinkedList<StationNode*>::Node* station = stationList.getHead();
                         station; station = station->next) {
                        const double km = station->data->getKmMarker();
                        if (km > lowKm && km < highKm) adjacentBlock = false;
                    }
                }
                const bool trackExists = resource != "TRACK" ||
                    (stationList.findStationByID(first) &&
                     second <= stationList.findStationByID(first)->getMaxTracks());
                const int startMinute = parseCommandInteger(fields[startField]);
                const int endMinute = parseCommandInteger(fields[endField]);
                success = stationsExist && adjacentBlock && trackExists &&
                    (opening
                        ? trainList.removeMaintenance(resource, first, second,
                                                      startMinute, endMinute)
                        : trainList.addMaintenance(resource, first, second,
                                                   startMinute, endMinute));
                changedTrains = changedTrains || success;
            }
        } catch (const exception&) {
            success = false;
        }

        resultLog += "Dong " + to_string(lineNumber) + ": " + command +
            (success ? " [OK]\n" : " [TU CHOI - sai du lieu, khong ton tai hoac vi pham rang buoc]\n");
    }

    bool saved = true;
    if (changedStations && !saveStations("stations.txt", stationList)) {
        resultLog += "Loi: khong ghi duoc stations.txt\n";
        saved = false;
    }
    if (!changedStations && !changedTrains && resultLog.empty()) {
        resultLog = "Khong co lenh nao trong commands.txt.";
    }
    trainList.resolveConflictAndScheduleSiding(&stationList);
    if (!saveTrains("trains.txt", trainList)) {
        resultLog += "Loi: khong ghi duoc trains.txt\n";
        saved = false;
    }
    return saved;
}

// Tạo hình chữ nhật bo góc bằng sf::ConvexShape
static sf::ConvexShape createRoundedRect(float width, float height, float radius, unsigned int cornerPoints = 6) {
    radius = std::min(radius, std::min(width, height) / 2.0f);
    sf::ConvexShape shape;
    shape.setPointCount(cornerPoints * 4);

    const float PI = 3.14159265359f;
    unsigned int idx = 0;

    sf::Vector2f centers[4] = {
        {width - radius, height - radius},
        {radius, height - radius},
        {radius, radius},
        {width - radius, radius}
    };

    for (int c = 0; c < 4; ++c) {
        float startAngle = c * (PI / 2.0f);
        for (unsigned int i = 0; i < cornerPoints; ++i) {
            float angle = startAngle + (static_cast<float>(i) / (cornerPoints - 1)) * (PI / 2.0f);
            shape.setPoint(idx++, {
                centers[c].x + radius * std::cos(angle),
                centers[c].y + radius * std::sin(angle)
            });
        }
    }
    shape.setOrigin({width / 2.0f, height / 2.0f});
    return shape;
}

// Thêm một hình chữ nhật (ghép từ 2 tam giác) vào VertexArray trong SFML 3.0
static void appendQuadAsTriangles(sf::VertexArray& va, float x, float y, float w, float h, const sf::Color& color) {
    sf::Vector2f p0(x, y), p1(x + w, y), p2(x + w, y + h), p3(x, y + h);
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p1, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p3, color});
}

// Vẽ đoạn thẳng có độ dày tùy chỉnh (dùng cho ghi rẽ nhánh đường ray phụ)
static void drawThickLine(sf::RenderWindow& window, sf::Vector2f p1, sf::Vector2f p2, float thickness, sf::Color color) {
    sf::Vector2f dir = p2 - p1;
    float length = std::sqrt(dir.x * dir.x + dir.y * dir.y);
    if (length <= 0.001f) return;

    sf::RectangleShape line({length, thickness});
    line.setOrigin({0.0f, thickness / 2.0f});
    line.setPosition(p1);
    float angleRad = std::atan2(dir.y, dir.x);
    line.setRotation(sf::radians(angleRad));
    line.setFillColor(color);
    window.draw(line);
}

// ==================== 3. HÀM TÍNH TỌA ĐỘ KM MƯỢT THEO THỜI GIAN THỰC ====================
static double calculateTrainKmSmooth(TrainNode* train, StationList* stationList, float currentMinute, int& trackUsedOut) {
    trackUsedOut = 0;
    LinkedList<StopScheduleNode*>::Node* firstStopNode = train->getHeadSchedule();
    StopScheduleNode* firstStop = firstStopNode ? firstStopNode->data : nullptr;
    if (!firstStop) return -1.0;

    LinkedList<StopScheduleNode*>::Node* lastStopNode = train->getTailSchedule();
    StopScheduleNode* lastStop = lastStopNode ? lastStopNode->data : nullptr;

    // 1. Chưa tới giờ xuất phát -> Tàu chờ sẵn tại Ga Đầu
    if (currentMinute < static_cast<float>(firstStop->getArrivalMinute())) {
        StationNode* st = stationList->findStationByID(firstStop->getStationID());
        if (st) {
            trackUsedOut = firstStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 2. Đã qua giờ về ga cuối -> Tàu đỗ tại Ga Cuối
    if (currentMinute > static_cast<float>(lastStop->getDepartureMinute())) {
        StationNode* st = stationList->findStationByID(lastStop->getStationID());
        if (st) {
            trackUsedOut = lastStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 3. Kiểm tra tàu đang đỗ tại ga hoặc đang di chuyển giữa 2 ga (Khu gian)
    for (LinkedList<StopScheduleNode*>::Node* stopNode = firstStopNode;
         stopNode != nullptr; stopNode = stopNode->next) {
        StopScheduleNode* curr = stopNode->data;
        float arr = static_cast<float>(curr->getArrivalMinute());
        float dep = static_cast<float>(curr->getDepartureMinute());

        if (currentMinute >= arr && currentMinute <= dep) {
            StationNode* st = stationList->findStationByID(curr->getStationID());
            if (st) {
                trackUsedOut = curr->getTrackUsed();
                return st->getKmMarker();
            }
        }

        StopScheduleNode* nextStop =
            stopNode->next ? stopNode->next->data : nullptr;
        if (nextStop != nullptr) {
            float nextArr = static_cast<float>(nextStop->getArrivalMinute());
            if (currentMinute > dep && currentMinute < nextArr) {
                StationNode* st1 = stationList->findStationByID(curr->getStationID());
                StationNode* st2 = stationList->findStationByID(nextStop->getStationID());
                if (st1 && st2 && nextArr > dep) {
                    trackUsedOut = 0; // 0: Đang chạy trên khu gian đơn
                    double ratio = static_cast<double>(currentMinute - dep) / static_cast<double>(nextArr - dep);
                    return st1->getKmMarker() + ratio * (st2->getKmMarker() - st1->getKmMarker());
                }
            }
        }
    }

    return -1.0;
}

// ==================== 4. VÒNG LẶP ĐỒ HỌA CHÍNH (GRAPHICAL SIMULATION ENGINE) ====================
void GraphicalSimulationEngine::runSFMLSimulation(StationList* stationList, TrainList* trainList) {
    if (!stationList || !trainList) return;

    sf::ContextSettings settings;
    settings.antiAliasingLevel = 8;
    sf::RenderWindow window(sf::VideoMode({WINDOW_WIDTH, WINDOW_HEIGHT}),
                            "PBL2: MO PHONG DIEU PHOI DUONG RAY & CHAY TAU (SFML GRAPHICS)",
                            sf::Style::Default, sf::State::Windowed, settings);
    window.setFramerateLimit(60);

    sf::Font font;
    if (!font.openFromFile("arial.ttf")) {
        cout << "[Canh bao] Khong tim thay font 'arial.ttf'!\n";
    }

    // Tìm mốc Km lớn nhất trên tuyến để chuẩn hóa tỷ lệ vẽ
    double maxKm = 100.0;
    for (LinkedList<StationNode*>::Node* station = stationList->getHead();
         station != nullptr; station = station->next) {
        StationNode* st = station->data;
        if (st->getKmMarker() > maxKm) maxKm = st->getKmMarker();
    }
    if (maxKm <= 0.0) maxKm = 100.0;

    const float marginX = 95.0f;
    const float trackY = 285.0f;
    const float trackWidth = WINDOW_WIDTH - 2.0f * marginX;
    const float sidingSpacingY = 44.0f;

    // Khởi tạo VertexArray cho Đường ray chính & Tà vẹt (Chỉ tốn 1 Draw Call mỗi khung hình)
    sf::VertexArray mainTrackVA(sf::PrimitiveType::Triangles);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 8.0f, trackWidth + 40.0f, 16.0f, Theme::BallastBed);
    for (float x = marginX - 10.0f; x <= marginX + trackWidth + 10.0f; x += 12.0f) {
        appendQuadAsTriangles(mainTrackVA, x, trackY - 7.0f, 4.0f, 14.0f, Theme::SleeperWood);
    }
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 3.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY + 1.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);

    // Mảng tĩnh quản lý hiệu ứng chuyển làn mượt (Smooth Y-Lerp) không dùng STL
    static const int MAX_SMOOTH_TRAINS = 64;
    TrainSmoothState smoothStates[MAX_SMOOTH_TRAINS];
    int smoothCount = 0;

    auto getSmoothY = [&](const string& code, float targetY, float dt) -> float {
        for (int i = 0; i < smoothCount; ++i) {
            if (smoothStates[i].code == code) {
                float lerpFactor = std::min(1.0f, dt * 8.5f);
                smoothStates[i].currentY += (targetY - smoothStates[i].currentY) * lerpFactor;
                return smoothStates[i].currentY;
            }
        }
        if (smoothCount < MAX_SMOOTH_TRAINS) {
            smoothStates[smoothCount].code = code;
            smoothStates[smoothCount].currentY = targetY;
            smoothCount++;
        }
        return targetY;
    };

    // Tự động lấy mốc thời gian bắt đầu sớm nhất của các đoàn tàu (trừ hao 15 phút chuẩn bị)
    int earliestMinute = std::numeric_limits<int>::max();
    int latestMinute = 0;
    for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
         trainNode != nullptr; trainNode = trainNode->next) {
        TrainNode* tr = trainNode->data;
        for (LinkedList<StopScheduleNode*>::Node* stop = tr->getHeadSchedule();
             stop; stop = stop->next) {
            earliestMinute = std::min(earliestMinute, stop->data->getArrivalMinute());
            latestMinute = std::max(latestMinute, stop->data->getDepartureMinute());
        }
    }
    if (earliestMinute == std::numeric_limits<int>::max()) earliestMinute = 0;
    const float endMinute = static_cast<float>(std::max(latestMinute, earliestMinute) + 60);
    float startMinute = static_cast<float>(std::max(0, earliestMinute - 15));
    float currentMinute = startMinute;

    float timeSpeed = 30.0f;
    bool isPaused = false;
    sf::Clock deltaClock;
    sf::Clock animClock;

    while (window.isOpen()) {
        while (const auto event = window.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                window.close();
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Space) {
                    isPaused = !isPaused;
                } else if (keyPressed->code == sf::Keyboard::Key::Up) {
                    timeSpeed = std::min(240.0f, timeSpeed + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Down) {
                    timeSpeed = std::max(5.0f, timeSpeed - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Right) {
                    currentMinute = std::min(endMinute, currentMinute + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Left) {
                    currentMinute = std::max(0.0f, currentMinute - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::R) {
                    currentMinute = startMinute;
                } else if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    window.close();
                }
            }
        }

        float dt = deltaClock.restart().asSeconds();
        float animTime = animClock.getElapsedTime().asSeconds();

        if (!isPaused) {
            currentMinute = std::min(endMinute, currentMinute + timeSpeed * dt);
        }

        window.clear(Theme::BgCanvas);

        // ==================== A. VẼ LƯỚI TỌA ĐỘ NỀN (SUBTLE GRID) ====================
        for (float gx = marginX; gx <= marginX + trackWidth; gx += 120.0f) {
            sf::RectangleShape vLine({1.0f, 390.0f});
            vLine.setPosition({gx, 105.0f});
            vLine.setFillColor(Theme::BgGridLine);
            window.draw(vLine);
        }

        // ==================== B. VẼ NHÀ GA, ĐƯỜNG RAY PHỤ & ĐÈN TÍN HIỆU ====================
        for (LinkedList<StationNode*>::Node* station = stationList->getHead();
             station != nullptr; station = station->next) {
            StationNode* st = station->data;
            float posX = marginX + static_cast<float>(st->getKmMarker() / maxKm) * trackWidth;
            int maxTracks = st->getMaxTracks();
            const int simulationMinute = static_cast<int>(currentMinute);
            const bool stationClosed = trainList->isUnderMaintenanceAt(
                "STATION", st->getStationID(), 0, simulationMinute);

            // Kiểm tra có đoàn tàu nào đang dừng đỗ tại ga này không để bật đèn tín hiệu Đỏ
            bool stationOccupied = false;
            for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
                 trainNode != nullptr; trainNode = trainNode->next) {
                TrainNode* tr = trainNode->data;
                for (LinkedList<StopScheduleNode*>::Node* stopNode =
                         tr->getHeadSchedule(); stopNode != nullptr;
                     stopNode = stopNode->next) {
                    StopScheduleNode* sc = stopNode->data;
                    if (sc->getStationID() == st->getStationID()) {
                        if (currentMinute >= static_cast<float>(sc->getArrivalMinute()) &&
                            (currentMinute <= static_cast<float>(sc->getDepartureMinute()) ||
                             (stopNode->next == nullptr &&
                              currentMinute <= endMinute))) {
                            stationOccupied = true;
                            break;
                        }
                    }
                }
                if (stationOccupied) break;
            }

            // 1. Trục gióng dọc tại tâm nhà ga
            float totalStationHeight = (maxTracks - 1) * sidingSpacingY + 75.0f;
            sf::RectangleShape guideLine({2.0f, totalStationHeight});
            guideLine.setPosition({posX - 1.0f, trackY - 35.0f});
            guideLine.setFillColor(stationClosed ? Theme::SignalRed : sf::Color(62, 70, 96, 110));
            window.draw(guideLine);

            // 2. Các đường ray phụ (Siding Tracks: từ Ray 2 đến maxTracks)
            for (int t = 2; t <= maxTracks; ++t) {
                float sidY = trackY + (t - 1) * sidingSpacingY;
                float prevY = trackY + (t - 2) * sidingSpacingY;
                float halfLen = 46.0f;
                const bool sidingClosed = stationClosed || trainList->isUnderMaintenanceAt(
                    "TRACK", st->getStationID(), t, simulationMinute);
                const sf::Color sidingColor = sidingClosed ? Theme::SignalRed : Theme::SidingRail;

                // Ghi rẽ nhánh trái & phải nối chéo lên ray phía trên
                drawThickLine(window, {posX - halfLen - 18.0f, prevY}, {posX - halfLen, sidY}, 3.0f, sidingColor);
                drawThickLine(window, {posX + halfLen, sidY}, {posX + halfLen + 18.0f, prevY}, 3.0f, sidingColor);

                // Tà vẹt trên ray phụ
                for (float sx = posX - halfLen; sx <= posX + halfLen; sx += 10.0f) {
                    sf::RectangleShape slp({3.0f, 10.0f});
                    slp.setOrigin({1.5f, 5.0f});
                    slp.setPosition({sx, sidY});
                    slp.setFillColor(sidingClosed ? Theme::SignalRed : Theme::SleeperWood);
                    window.draw(slp);
                }

                // Thanh ray phụ
                sf::RectangleShape sidingBar({halfLen * 2.0f, 3.5f});
                sidingBar.setOrigin({halfLen, 1.75f});
                sidingBar.setPosition({posX, sidY});
                sidingBar.setFillColor(sidingColor);
                window.draw(sidingBar);

                // Nhãn ký hiệu ray phụ (R2, R3...)
                sf::Text trkLabel(font, "R" + to_string(t), 12);
                trkLabel.setFillColor(sidingClosed ? Theme::SignalRed : Theme::TextSecondary);
                trkLabel.setPosition({posX - halfLen - 18.0f, sidY - 6.0f});
                window.draw(trkLabel);
            }

            // 3. Vùng Sân ga (Station Platform Badge)
            sf::ConvexShape platform = createRoundedRect(92.0f, 24.0f, 6.0f);
            platform.setPosition({posX, trackY - 58.0f});
            platform.setFillColor(stationClosed ? sf::Color(100, 28, 38) : Theme::StationPlatform);
            platform.setOutlineThickness(1.5f);
            platform.setOutlineColor(stationClosed ? Theme::SignalRed : Theme::HudBorder);
            window.draw(platform);

            sf::Text stTitle(font, st->getName(), 14);
            stTitle.setFillColor(Theme::AccentGold);
            sf::FloatRect tb = stTitle.getLocalBounds();
            stTitle.setPosition({posX - tb.size.x / 2.0f, trackY - 66.0f});
            window.draw(stTitle);

            sf::Text stSub(font, "ID:" + to_string(st->getStationID()) + " | " +
                           to_string(static_cast<int>(st->getKmMarker())) + "km (" + to_string(maxTracks) + "R)", 12);
            stSub.setFillColor(stationClosed ? Theme::SignalRed : Theme::TextSecondary);
            sf::FloatRect sb = stSub.getLocalBounds();
            stSub.setPosition({posX - sb.size.x / 2.0f, trackY - 42.0f});
            window.draw(stSub);

            // 4. Đèn tín hiệu nhà ga (Xanh / Đỏ nhấp nháy)
            sf::Color sigColor = stationClosed || stationOccupied
                ? Theme::SignalRed : Theme::SignalGreen;
            float pulse = stationClosed || stationOccupied
                ? (0.55f + 0.45f * std::sin(animTime * 7.0f)) : 1.0f;

            sf::CircleShape sigGlow(10.0f);
            sigGlow.setOrigin({10.0f, 10.0f});
            sigGlow.setPosition({posX + 38.0f, trackY - 22.0f});
            sigGlow.setFillColor(sf::Color(sigColor.r, sigColor.g, sigColor.b, static_cast<uint8_t>(90 * pulse)));
            window.draw(sigGlow);

            sf::CircleShape sigLamp(4.5f);
            sigLamp.setOrigin({4.5f, 4.5f});
            sigLamp.setPosition({posX + 38.0f, trackY - 22.0f});
            sigLamp.setFillColor(sf::Color(sigColor.r, sigColor.g, sigColor.b, static_cast<uint8_t>(255 * pulse)));
            sigLamp.setOutlineThickness(1.5f);
            sigLamp.setOutlineColor(sf::Color(20, 20, 25));
            window.draw(sigLamp);
        }

        // ==================== C. VẼ ĐƯỜNG RAY CHÍNH & CÁC TÂM GA ====================
        window.draw(mainTrackVA);

        for (LinkedList<StationNode*>::Node* station = stationList->getHead();
             station != nullptr; station = station->next) {
            StationNode* first = station->data;
            const int minute = static_cast<int>(currentMinute);
            const bool stationClosed = trainList->isUnderMaintenanceAt(
                "STATION", first->getStationID(), 0, minute);
            const bool mainTrackClosed = stationClosed ||
                trainList->isUnderMaintenanceAt("TRACK", first->getStationID(), 1, minute);
            const float firstX = marginX +
                static_cast<float>(first->getKmMarker() / maxKm) * trackWidth;
            if (mainTrackClosed) {
                drawThickLine(window, {firstX - 46.0f, trackY},
                              {firstX + 46.0f, trackY}, 5.0f, Theme::SignalRed);
            }
            for (LinkedList<StationNode*>::Node* other = stationList->getHead();
                 other != nullptr; other = other->next) {
                StationNode* second = other->data;
                if (first->getStationID() < second->getStationID() &&
                    trainList->isUnderMaintenanceAt("BLOCK", first->getStationID(),
                        second->getStationID(), minute)) {
                    const float secondX = marginX +
                        static_cast<float>(second->getKmMarker() / maxKm) * trackWidth;
                    drawThickLine(window, {firstX, trackY}, {secondX, trackY},
                                  5.0f, Theme::SignalRed);
                }
            }
        }

        for (LinkedList<StationNode*>::Node* station = stationList->getHead();
             station != nullptr; station = station->next) {
            StationNode* st = station->data;
            float posX = marginX + static_cast<float>(st->getKmMarker() / maxKm) * trackWidth;
            const bool stationClosed = trainList->isUnderMaintenanceAt(
                "STATION", st->getStationID(), 0, static_cast<int>(currentMinute));

            sf::CircleShape outerNode(9.0f);
            outerNode.setOrigin({9.0f, 9.0f});
            outerNode.setPosition({posX, trackY});
            outerNode.setFillColor(Theme::BgCanvas);
            outerNode.setOutlineThickness(2.5f);
            outerNode.setOutlineColor(stationClosed ? Theme::SignalRed : Theme::StationCore);
            window.draw(outerNode);

            sf::CircleShape innerNode(4.0f);
            innerNode.setOrigin({4.0f, 4.0f});
            innerNode.setPosition({posX, trackY});
            innerNode.setFillColor(stationClosed ? Theme::SignalRed : Theme::StationCore);
            window.draw(innerNode);
        }

        // ==================== D. VẼ CÁC ĐOÀN TÀU BO GÓC, ĐỔ BÓNG & NỘI SUY LERP ====================
        for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
             trainNode != nullptr; trainNode = trainNode->next) {
            TrainNode* train = trainNode->data;
            int trackUsed = 0;
            double currentKm = calculateTrainKmSmooth(train, stationList, currentMinute, trackUsed);

            if (currentKm >= 0.0) {
                float posX = marginX + static_cast<float>(currentKm / maxKm) * trackWidth;
                float targetY = trackY + (trackUsed > 1 ? (trackUsed - 1) * sidingSpacingY : 0.0f);
                float posY = getSmoothY(train->getTrainCode(), targetY, dt);

                if (trackUsed == 1) bodyColor = Theme::TrainMainStop;
                else if (trackUsed > 1) bodyColor = Theme::TrainSidingWait;

                if (trackUsed == 0) {

                    beam.setPointCount(3);
                    beam.setPoint(0, {posX + dirSign * 26.0f, posY});
                    beam.setPoint(1, {posX + dirSign * 72.0f, posY - 12.0f});
                    beam.setPoint(2, {posX + dirSign * 72.0f, posY + 12.0f});

                    window.draw(beam);

                trainShape.setPosition({posX, posY});
                trainShape.setFillColor(bodyColor);
                trainShape.setOutlineThickness(1.5f);
                trainShape.setOutlineColor(sf::Color(255, 255, 255, 210));
                window.draw(trainShape);

                // 4. Dải cửa sổ kính trên thân tàu
                sf::ConvexShape windowStrip = createRoundedRect(36.0f, 6.0f, 2.5f);
                windowStrip.setPosition({posX, posY - 3.5f});
                windowStrip.setFillColor(sf::Color(18, 22, 34, 200));
                window.draw(windowStrip);

                string badgeStr = code + " [P" + to_string(train->getPriority()) + "]";
                sf::Text trainLabel(font, badgeStr, 12);
                trainLabel.setFillColor(sf::Color(15, 18, 25));
                sf::FloatRect lb = trainLabel.getLocalBounds();
                trainLabel.setPosition({posX - lb.size.x / 2.0f, posY - 0.5f});
                window.draw(trainLabel);

                // Nhãn hướng chạy nổi phía trên đoàn tàu
                sf::Text dirTag(font, (train->getDirection() == 1 ? code + " ->" : "<- " + code), 13);
                dirTag.setFillColor(Theme::TextPrimary);
                sf::FloatRect db = dirTag.getLocalBounds();
                dirTag.setPosition({posX - db.size.x / 2.0f, posY - 26.0f});

        sf::ConvexShape topHud = createRoundedRect(WINDOW_WIDTH - 40.0f, 58.0f, 10.0f);
        topHud.setPosition({WINDOW_WIDTH / 2.0f, 38.0f});
        topHud.setFillColor(Theme::HudPanelBg);
        topHud.setOutlineThickness(1.5f);
        topHud.setOutlineColor(Theme::HudBorder);
        window.draw(topHud);

        const int absoluteMinute = static_cast<int>(currentMinute);
        const int day = absoluteMinute / 1440;
        int hours = (absoluteMinute / 60) % 24;
        int mins  = static_cast<int>(currentMinute) % 60;
        int secs  = static_cast<int>((currentMinute - std::floor(currentMinute)) * 60.0f);

        stringstream clockSS;
        if (day > 0) clockSS << "D+" << day << " ";
        clockSS << setfill('0') << setw(2) << hours << ":"
                << setfill('0') << setw(2) << mins << ":"
                << setfill('0') << setw(2) << secs;

        sf::Text clockText(font, "TIME: " + clockSS.str(), 20);
        clockText.setFillColor(Theme::AccentGold);
        clockText.setPosition({45.0f, 24.0f});
        window.draw(clockText);

        stringstream ctrlSS;
        ctrlSS << (isPaused ? "[TAM DUNG]" : "[DANG CHAY]")
               << "  |  TOC DO: " << static_cast<int>(timeSpeed) << "x"
               << "  |  [SPACE] Pause  [UP/DOWN] Toc do  [LEFT/RIGHT] Tua +/-15p  [R] Reset  [ESC] Thoat";
        sf::Text ctrlText(font, ctrlSS.str(), 13);
        ctrlText.setFillColor(Theme::TextPrimary);
        ctrlText.setPosition({255.0f, 28.0f});
        window.draw(ctrlText);

        // Chú giải màu sắc (Legend)
        auto drawLegendItem = [&](float lx, float ly, sf::Color col, const string& textStr) {
            sf::CircleShape dot(6.0f);
            dot.setPosition({lx, ly + 2.0f});

            sf::Text lbl(font, textStr, 12);
            lbl.setFillColor(Theme::TextSecondary);
            lbl.setPosition({lx + 18.0f, ly});
            window.draw(lbl);
        };
        drawLegendItem(45.0f, 76.0f, Theme::TrainRunning, "Chay khu gian (Cyan)");
        drawLegendItem(215.0f, 76.0f, Theme::TrainMainStop, "Do Ray chinh R1 (Amber)");
        drawLegendItem(400.0f, 76.0f, Theme::TrainSidingWait, "Tranh tau Ray phu R2..N (Purple)");
        drawLegendItem(635.0f, 76.0f, Theme::SignalRed, "Bao tri/ga co tau chiem");
        drawLegendItem(845.0f, 76.0f, Theme::SignalGreen, "Den Xanh: Ga thong thoang");

        // 2. Khung Nhật ký Giải quyết Xung đột (Đọc trực tiếp từ Circular Buffer của TrainList)
        float logW = 760.0f, logH = 220.0f;
        sf::ConvexShape logPanel = createRoundedRect(logW, logH, 10.0f);
        logPanel.setPosition({20.0f + logW / 2.0f, WINDOW_HEIGHT - 20.0f - logH / 2.0f});
        logPanel.setFillColor(Theme::HudPanelBg);
        logPanel.setOutlineThickness(1.5f);
        logPanel.setOutlineColor(Theme::HudBorder);
        window.draw(logPanel);

        sf::Text logTitle(font, "NHAT KY DIEU PHOI & GIAI QUYET XUNG DOT (MEET & PASS - CIRCULAR BUFFER LOGS)", 13);
        logTitle.setFillColor(Theme::AccentGold);
        logTitle.setPosition({38.0f, WINDOW_HEIGHT - logH - 8.0f});
        window.draw(logTitle);

        int totalLogs = trainList->getLogCount();
        if (totalLogs == 0) {
            sf::Text emptyLog(font, "-> He thong dang van hanh on dinh. Chua phat sinh xung dot moi hoac da giai toa xong.", 12);
            emptyLog.setFillColor(Theme::TextSecondary);
            emptyLog.setPosition({42.0f, WINDOW_HEIGHT - logH + 22.0f});
            window.draw(emptyLog);
        } else {
            float ly = WINDOW_HEIGHT - logH + 18.0f;
            for (int i = 0; i < totalLogs; ++i) {
                sf::Text logLine(font, ">> " + trainList->getLog(i), 12);
                logLine.setFillColor(Theme::TextPrimary);
                logLine.setPosition({42.0f, ly});
                window.draw(logLine);
                ly += 21.0f;
            }
        }

        // 3. Bảng Giám sát Trạng thái Đoàn tàu Thời gian thực (Bottom-Right Telemetry Panel)
        float teleW = 535.0f, teleH = 220.0f;
        float teleCenterX = WINDOW_WIDTH - 20.0f - teleW / 2.0f;
        sf::ConvexShape telePanel = createRoundedRect(teleW, teleH, 10.0f);
        telePanel.setPosition({teleCenterX, WINDOW_HEIGHT - 20.0f - teleH / 2.0f});
        telePanel.setFillColor(Theme::HudPanelBg);
        telePanel.setOutlineThickness(1.5f);
        telePanel.setOutlineColor(Theme::HudBorder);
        window.draw(telePanel);

        float teleLeft = WINDOW_WIDTH - 20.0f - teleW + 18.0f;
        sf::Text teleTitle(font, "BANG GIAM SAT TRANG THAI DOAN TAU (LIVE TELEMETRY)", 13);
        teleTitle.setFillColor(Theme::StationCore);
        teleTitle.setPosition({teleLeft, WINDOW_HEIGHT - teleH - 8.0f});
        window.draw(teleTitle);

        float ty = WINDOW_HEIGHT - teleH + 18.0f;
        for (LinkedList<TrainNode*>::Node* trainNode = trainList->getHead();
             trainNode != nullptr; trainNode = trainNode->next) {
            TrainNode* tr = trainNode->data;
            int trk = 0;
            double km = calculateTrainKmSmooth(tr, stationList, currentMinute, trk);
            string statusStr = (trk == 0) ? "Chay khu gian" : (trk == 1 ? "Do Ray Chinh (R1)" : "Tranh Ray Phu (R" + to_string(trk) + ")");

            stringstream rowSS;
            rowSS << left << setw(6) << tr->getTrainCode()
                  << " | Uu tien P" << tr->getPriority()
                  << " | " << (tr->getDirection() == 1 ? "Nam" : "Bac")
                  << " | Km " << setw(5) << static_cast<int>(std::max(0.0, km))
                  << " | " << statusStr;

            sf::Text rowTxt(font, rowSS.str(), 12);
            rowTxt.setFillColor(trk > 1 ? Theme::TrainSidingWait : (trk == 1 ? Theme::TrainMainStop : Theme::TrainRunning));
            rowTxt.setPosition({teleLeft, ty});
            window.draw(rowTxt);
            ty += 22.0f;
        }

        window.display();
    }

static string formatScheduleMinute(int totalMinutes) {
    const int day = totalMinutes / 1440;
    const int minuteOfDay = totalMinutes % 1440;
    stringstream formatted;
    if (day > 0) formatted << "D+" << day << " ";
    formatted << setfill('0') << setw(2) << minuteOfDay / 60 << ":"
              << setfill('0') << setw(2) << minuteOfDay % 60;
    return formatted.str();
}

// ==================== 1. HỘP THOẠI NHẬP LIỆU HIỆN ĐẠI (MODAL INPUT DIALOG) ====================
bool GuiMenu::showInputDialog(const string& title, const string& promptLabel, string& outputValue, sf::Font& font) {
    sf::RenderWindow dialog(sf::VideoMode({520, 230}), title, sf::Style::Titlebar | sf::Style::Close);
    dialog.setFramerateLimit(60);

    string inputText = "";

    sf::RectangleShape topBanner(sf::Vector2f(520.0f, 38.0f));
    topBanner.setFillColor(GuiTheme::CardBg);
    sf::Text bannerTitle(font, "NHAP THONG TIN HE THONG", 13);
    bannerTitle.setFillColor(GuiTheme::AccentGold);
    bannerTitle.setPosition(sf::Vector2f(20.0f, 10.0f));

    sf::Text promptText(font, promptLabel, 14);
    promptText.setFillColor(GuiTheme::TextLight);
    promptText.setPosition(sf::Vector2f(25.0f, 52.0f));

    sf::RectangleShape inputBox(sf::Vector2f(470.0f, 42.0f));
    inputBox.setPosition(sf::Vector2f(25.0f, 82.0f));
    inputBox.setFillColor(GuiTheme::BgMain);
    inputBox.setOutlineThickness(2.0f);
    inputBox.setOutlineColor(GuiTheme::AccentCyan);

    sf::Text typedText(font, "", 16);
    typedText.setFillColor(GuiTheme::AccentGold);
    typedText.setPosition(sf::Vector2f(35.0f, 92.0f));

    GuiButton btnConfirm;
    btnConfirm.init(75.0f, 150.0f, 175.0f, 40.0f, "XAC NHAN (ENTER)", font, 99, false);
    GuiButton btnCancel;
    btnCancel.init(270.0f, 150.0f, 175.0f, 40.0f, "HUY BO (ESC)", font, 98, true);

    sf::Clock blinkClock;

    while (dialog.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(dialog);
        bool mouseClicked = false;

        while (const auto event = dialog.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                dialog.close();
                return false;
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    dialog.close();
                    return false;
                }
            }
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
            if (const auto* textEntered = event->getIf<sf::Event::TextEntered>()) {
                uint32_t unicode = textEntered->unicode;
                if (unicode == 8) { // Backspace
                    if (!inputText.empty()) inputText.pop_back();
                } else if (unicode == 13) { // Enter
                    if (!inputText.empty()) {
                        outputValue = inputText;
                        dialog.close();
                        return true;
                    }
                } else if (unicode >= 32 && unicode <= 126) {
                    inputText += static_cast<char>(unicode);
                }
            }
        }

        btnConfirm.updateHover(mousePos);
        btnCancel.updateHover(mousePos);

        if (btnConfirm.isClicked(mouseClicked) && !inputText.empty()) {
            outputValue = inputText;
            dialog.close();
            return true;
        }
        if (btnCancel.isClicked(mouseClicked)) {
            dialog.close();
            return false;
        }

        bool showCursor = (static_cast<int>(blinkClock.getElapsedTime().asSeconds() * 2.0f) % 2 == 0);
        typedText.setString(inputText + (showCursor ? "_" : ""));

        dialog.clear(GuiTheme::PanelBg);
        dialog.draw(topBanner);
        dialog.draw(bannerTitle);
        dialog.draw(promptText);
        dialog.draw(inputBox);
        dialog.draw(typedText);
        btnConfirm.draw(dialog);
        btnCancel.draw(dialog);
        dialog.display();
    }
    return false;
}

// --- 2.1. VẼ BẢNG DANH SÁCH NHÀ GA (ACTION ID: 1) ---
void renderStationTable(sf::RenderWindow& window, sf::Font& font, const StationList& stationList, float startY, const string& filter) {

    float currentX = tableX;
    for (int i = 0; i < 5; ++i) {

        headerBox.setPosition(sf::Vector2f(currentX, startY));

        headerBox.setOutlineColor(GuiTheme::Border);

        txt.setPosition(sf::Vector2f(currentX + 10.0f, startY + 8.0f));

            cellTxt.setPosition(sf::Vector2f(currentX + 10.0f, currentY + 7.0f));
            window.draw(cellTxt);
            currentX += colWidths[c];
        }
        currentY += rowHeight;
        rowIndex++;
    }
}

// ==================== 4. GIAO DIỆN LAN TRUYỀN TRỄ & TÁI LẬP LỊCH ====================
void GuiMenu::applyDelayUI(StationList& stationList, TrainList& trainList, sf::Font& font, string& outputLog) {
    string trainCode, strStID, strDelay;
    if (showInputDialog("DELAY PROPAGATION", "1/3. Nhap Ma Tau bi su co tre (Vi du: SE1):", trainCode, font) &&
        showInputDialog("DELAY PROPAGATION", "2/3. Nhap ID Ga bat dau bi tre (Vi du: 102):", strStID, font) &&
        showInputDialog("DELAY PROPAGATION", "3/3. Nhap So phut tre (Vi du: 30):", strDelay, font)) 
    {
        try {
            int stID = stoi(strStID);
            int delayMins = stoi(strDelay);
            if (trainList.applyDelayAndReschedule(trainCode, stID, delayMins, &stationList)) {
                FileManager::saveTrains("trains.txt", trainList);
                outputLog = "=== KET QUA LAN TRUYEN TRE & TAI LAP LICH (EVENT-DRIVEN) ===\n\n";
                outputLog += " -> Da cong them " + to_string(delayMins) + " phut cho tau " + trainCode + " tu Ga ID " + to_string(stID) + ".\n";
                outputLog += " -> Da kich hoat bo lap lich theo su kien (xet Priority & maxTracks).\n\n";
                outputLog += "Cac xung dot vua duoc giai quyet:\n";
                for (int i = 0; i < trainList.getLogCount(); ++i) {
                    outputLog += "   + " + trainList.getLog(i) + "\n";
                }
            } else {
                outputLog = "-> [THAT BAI] Khong tim thay Tau " + trainCode + " hoac Ga " + strStID + " khong nam trong lich trinh!";
            }
        } catch (...) {
            outputLog = "-> [LOI] ID Ga va So phut tre phai la so nguyen!";
        }
    }
}

// Ghi thao tác bảo trì vào lịch sử lệnh để trạng thái được giữ sau khi khởi động lại.
static bool changeMaintenanceUI(bool opening, StationList& stationList, TrainList& trainList,
                                sf::Font& font, string& outputLog) {
    string resource, firstText, secondText, startText, endText;
    if (!GuiMenu::showInputDialog(opening ? "MO TAI NGUYEN" : "DONG TAI NGUYEN",
                                  "Loai tai nguyen (STATION, BLOCK, TRACK):",
                                  resource, font)) return false;
    transform(resource.begin(), resource.end(), resource.begin(),
              [](unsigned char ch) { return static_cast<char>(toupper(ch)); });
    if (resource != "STATION" && resource != "BLOCK" && resource != "TRACK") {
        outputLog = "-> [TU CHOI] Loai tai nguyen phai la STATION, BLOCK hoac TRACK.";
        return false;
    }

    if (!GuiMenu::showInputDialog("BAO TRI", "ID ga (hoac ga dau cua khu gian):",
                                  firstText, font)) return false;
    const bool isStation = resource == "STATION";
    if (!isStation && !GuiMenu::showInputDialog("BAO TRI",
            resource == "BLOCK" ? "ID ga ke tiep cua khu gian:" : "So ray tai ga:",
            secondText, font)) return false;
    if (!GuiMenu::showInputDialog("BAO TRI", "Phut bat dau (so phut tu moc ngay 0):",
                                  startText, font) ||
        !GuiMenu::showInputDialog("BAO TRI", "Phut ket thuc:", endText, font)) return false;

    try {
        size_t used = 0;
        const int first = stoi(firstText, &used);
        if (used != firstText.size()) throw invalid_argument("station");
        const int second = isStation ? 0 : stoi(secondText, &used);
        if (!isStation && used != secondText.size()) throw invalid_argument("resource");
        const int start = stoi(startText, &used);
        if (used != startText.size()) throw invalid_argument("start");
        const int end = stoi(endText, &used);
        if (used != endText.size()) throw invalid_argument("end");

        StationNode* firstStation = stationList.findStationByID(first);
        StationNode* secondStation = resource == "BLOCK"
            ? stationList.findStationByID(second) : nullptr;
        bool validResource = firstStation != nullptr;
        if (resource == "BLOCK") {
            validResource = validResource && secondStation && first != second &&
                firstStation->getKmMarker() != secondStation->getKmMarker();
            if (validResource) {
                const double lowKm = min(firstStation->getKmMarker(), secondStation->getKmMarker());
                const double highKm = max(firstStation->getKmMarker(), secondStation->getKmMarker());
                for (LinkedList<StationNode*>::Node* station = stationList.getHead();
                     station; station = station->next) {
                    if (station->data->getKmMarker() > lowKm &&
                        station->data->getKmMarker() < highKm) validResource = false;
                }
            }
        } else if (resource == "TRACK") {
            validResource = validResource && second >= 1 &&
                            second <= firstStation->getMaxTracks();
        }
        if (!validResource || start < 0 || end <= start) {
            outputLog = "-> [TU CHOI] Tai nguyen khong ton tai/khong hop le hoac khoang thoi gian sai.";
            return false;
        }

        const bool changed = opening
            ? trainList.removeMaintenance(resource, first, second, start, end)
            : trainList.addMaintenance(resource, first, second, start, end);
        if (!changed) {
            outputLog = "-> [TU CHOI] Khong tim thay cua so bao tri can mo.";
            return false;
        }

        trainList.resolveConflictAndScheduleSiding(&stationList);
        if (!FileManager::saveTrains("trains.txt", trainList)) {
            if (opening) trainList.addMaintenance(resource, first, second, start, end);
            else trainList.removeMaintenance(resource, first, second, start, end);
            trainList.resolveConflictAndScheduleSiding(&stationList);
            outputLog = "-> [LOI] Khong ghi duoc trains.txt; da khoi phuc trang thai bao tri.";
            return false;
        }

        const string operation = opening ? "OPEN_" : "CLOSE_";
        const string line = operation + resource + "|" + to_string(first) +
            (isStation ? "" : "|" + to_string(second)) + "|" +
            to_string(start) + "|" + to_string(end);
        ofstream commands("commands.txt", ios::app);
        if (commands) commands << line << '\n';
        commands.flush();
        if (!commands) {
            if (opening) trainList.addMaintenance(resource, first, second, start, end);
            else trainList.removeMaintenance(resource, first, second, start, end);
            trainList.resolveConflictAndScheduleSiding(&stationList);
            FileManager::saveTrains("trains.txt", trainList);
            outputLog = "-> [LOI] Khong ghi duoc commands.txt; da khoi phuc trang thai bao tri.";
            return false;
        }
        outputLog = opening
            ? "-> [THANH CONG] Da mo tai nguyen va cap nhat lai dieu phoi."
            : "-> [THANH CONG] Da dong tai nguyen, cap nhat dieu phoi va luu commands.txt.";
        return true;
    } catch (const exception&) {
        outputLog = "-> [LOI] ID va thoi gian phai la so nguyen hop le.";
        return false;
    }
}

    menuGroups[2].initGroup(startX, 0, btnW, mainH, "[3] BAO TRI & SU CO", font, 300);

    string outputLog =
        "HE THONG DIEU PHOI DUONG SAT DON (MEET & PASS CONTROL CENTER)\n"

        " * Thuat toan Uu tien Tau (Train Priority): SE (P1) > TN (P2) > Tau hang (P3).\n"

    string stationSearch, trainSearch, stopStationSearch;
    int searchFocus = 0;

    while (window.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(window);
        bool mouseClicked = false;

        while (const auto event = window.pollEvent()) {
            if (event->is<sf::Event::Closed>()) window.close();
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
            if (const auto* wheel = event->getIf<sf::Event::MouseWheelScrolled>()) {
                scrollOffsetY += wheel->delta * 28.0f;
                if (scrollOffsetY > 0.0f) scrollOffsetY = 0.0f;
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Down) scrollOffsetY -= 28.0f;
                if (keyPressed->code == sf::Keyboard::Key::Up) {
                    scrollOffsetY += 28.0f;
                    if (scrollOffsetY > 0.0f) scrollOffsetY = 0.0f;
                }
            }
            if (const auto* textEntered = event->getIf<sf::Event::TextEntered>()) {
                string* target = searchFocus == 1 ? &stationSearch :
                    (searchFocus == 2 ? &trainSearch :
                     (searchFocus == 3 ? &stopStationSearch : nullptr));
                if (target) {
                    if (textEntered->unicode == 8) {
                        if (!target->empty()) target->pop_back();
                    } else if (target->size() < 32 &&

                        target->push_back(static_cast<char>(textEntered->unicode));

        if (mouseClicked) {

        // Xử lý sự kiện Click Menu
        for (int g = 0; g < NUM_GROUPS; ++g) {

            if (menuGroups[g].mainCategoryBtn.isClicked(mouseClicked)) {

                    for (int k = 0; k < NUM_GROUPS; ++k) {

                        menuGroups[k].mainCategoryBtn.setSelected(false);

                    menuGroups[g].isExpanded = willExpand;

                        case 18: {

                                                    " * Ten Nha Ga     : " + st->getName() + "\n"
                                                    " * Moc Km         : " + to_string(static_cast<int>(st->getKmMarker())) + " km\n"

                            activeViewMode = 4; // Vẽ Bảng Đoàn Tàu

                            applyDelayUI(stationList, trainList, font, outputLog);

                        }
                        case 23: {
                            activeViewMode = 0;
                            changeMaintenanceUI(false, stationList, trainList, font, outputLog);
                            break;
                        }
                        case 24: {
                            activeViewMode = 0;
                            changeMaintenanceUI(true, stationList, trainList, font, outputLog);
                            break;
                        }
                        case 12: {
                            activeViewMode = 0;
                            GraphicalSimulationEngine::runSFMLSimulation(&stationList, &trainList);
                            outputLog = "-> Da dong cua so Mo phong 2D Running Engine va quay lai Trung tam Dieu hanh.";
                            break;
                        }
                        case 17: {
                            activeViewMode = 0;
                            trainList.resolveConflictAndScheduleSiding(&stationList);
                            FileManager::saveTrains("trains.txt", trainList);
                            outputLog = "=== HOAN TAT QUET XUNG DOT & TAI LAP LICH (EVENT-DRIVEN) ===\n\n";
                            if (trainList.getLogCount() == 0) {
                                outputLog += " -> Toan tuyen khong phat sinh xung dot moi.\n";
                            } else {
                                for (int i = 0; i < trainList.getLogCount(); ++i) {
                                    outputLog += " >> " + trainList.getLog(i) + "\n";
                                }
                            }
                            break;
                        }
                        case 13: {
                            activeViewMode = 0;
                            stationList.clear();
                            trainList.clear();
                            FileManager::loadStations("stations.txt", stationList);
                            FileManager::loadTrains("trains.txt", trainList);
                            string commandLog;
                            if (FileManager::processInputCommands("commands.txt", stationList,
                                    trainList, commandLog)) {
                                outputLog = "-> [RELOAD THANH CONG] Da nap du lieu va commands.txt.\n\n" +
                                            commandLog;
                            } else {
                                trainList.resolveConflictAndScheduleSiding(&stationList);
                                outputLog = "-> [LOI] " + commandLog;

                            break;

                        }
                    }
                }
            }
        }

        // ==================== VẼ GIAO DIỆN DASHBOARD HIỆN ĐẠI ====================
        window.clear(GuiTheme::BgMain);

        const int stCount = stationList.getSize();
        const int trCount = trainList.getSize();

        auto drawKPICard = [&](float x, float y, float w, float h, const string& title, const string& val, sf::Color accent) {
            sf::RectangleShape card(sf::Vector2f(w, h));
            card.setPosition(sf::Vector2f(x, y));
            card.setFillColor(GuiTheme::CardBg);
            card.setOutlineThickness(1.0f);
            card.setOutlineColor(GuiTheme::Border);
            window.draw(card);

            sf::RectangleShape topLine(sf::Vector2f(w, 3.5f));
            topLine.setPosition(sf::Vector2f(x, y));
            topLine.setFillColor(accent);
            window.draw(topLine);

            sf::Text lbl(font, title, 11);
            lbl.setFillColor(GuiTheme::TextMuted);
            lbl.setPosition(sf::Vector2f(x + 14.0f, y + 12.0f));
            window.draw(lbl);

            sf::Text vTxt(font, val, 20);
            vTxt.setFillColor(GuiTheme::TextLight);
            vTxt.setPosition(sf::Vector2f(x + 14.0f, y + 32.0f));
            window.draw(vTxt);
        };

        sf::Text appTitle(font, "RAILWAY CONTROL CENTER", 20);
        appTitle.setFillColor(GuiTheme::AccentGold);
        appTitle.setPosition(sf::Vector2f(startX, 24.0f));
        window.draw(appTitle);

        sf::Text appSub(font, "PBL2: Single-Track Meet & Pass Dispatcher", 12);
        appSub.setFillColor(GuiTheme::AccentCyan);
        appSub.setPosition(sf::Vector2f(startX, 52.0f));
        window.draw(appSub);

        drawKPICard(390.0f, 18.0f, 220.0f, 68.0f, "HA TANG NHA GA", to_string(stCount) + " Ga tren tuyen", GuiTheme::AccentGreen);
        drawKPICard(625.0f, 18.0f, 220.0f, 68.0f, "DOAN TAU DANG QUAN LY", to_string(trCount) + " Chuyen (P1-P3)", GuiTheme::AccentCyan);
        drawKPICard(860.0f, 18.0f, 220.0f, 68.0f, "LICH TRINH DANG QUAN LY", to_string(trCount) + " Doan tau", GuiTheme::AccentGold);
        drawKPICard(1095.0f, 18.0f, 240.0f, 68.0f, "LOG GIAI TOA XUNG DOT", to_string(trainList.getLogCount()) + " Su kien gan nhat", GuiTheme::AccentPurple);

        for (int g = 0; g < NUM_GROUPS; ++g) {
            menuGroups[g].mainCategoryBtn.draw(window);
            if (menuGroups[g].isExpanded) {
                for (int s = 0; s < menuGroups[g].subCount; ++s) {
                    menuGroups[g].subButtons[s].draw(window);
                }
            }
        }

        sf::RectangleShape outputBox(sf::Vector2f(945.0f, 445.0f));
        outputBox.setPosition(sf::Vector2f(390.0f, 105.0f));
        outputBox.setFillColor(GuiTheme::PanelBg);
        outputBox.setOutlineThickness(1.5f);
        outputBox.setOutlineColor(GuiTheme::Border);
        window.draw(outputBox);

        sf::View scrollView(sf::FloatRect({390.0f, 105.0f}, {945.0f, 445.0f}));
        scrollView.setViewport(sf::FloatRect(
            {390.0f / 1360.0f, 105.0f / 768.0f},
            {945.0f / 1360.0f, 445.0f / 768.0f}
        ));
        window.setView(scrollView);

        // Render giao diện theo chế độ view được chọn
        if (activeViewMode == 1) {
            renderStationTable(window, font, stationList, 150.0f + scrollOffsetY, stationSearch);
        } else if (activeViewMode == 4) {
            renderTrainTable(window, font, trainList, 150.0f + scrollOffsetY, trainSearch);
        } else if (activeViewMode == 5) {
            renderTrainDetailsPanel(window, font, trainList, stationList, 150.0f + scrollOffsetY,
                                    stopStationSearch);
        } else {
            sf::Text logText(font, outputLog, 13);
            logText.setFillColor(GuiTheme::TextLight);
            logText.setPosition(sf::Vector2f(410.0f, 120.0f + scrollOffsetY));
            window.draw(logText);
        }

        window.setView(defaultView);
        if (activeViewMode == 1 || activeViewMode == 4 || activeViewMode == 5) {
            const string filterLabel = activeViewMode == 1 ? "Tim ID ga: " :
                (activeViewMode == 4 ? "Tim ma tau: " : "Tim ID ga dung: ");
            const string& filter = activeViewMode == 1 ? stationSearch :
                (activeViewMode == 4 ? trainSearch : stopStationSearch);
            sf::RectangleShape searchBox({265.0f, 28.0f});
            searchBox.setPosition({1055.0f, 112.0f});
            searchBox.setFillColor(GuiTheme::BgMain);

#pragma once

#include <string>

struct IntHash {
    unsigned long operator()(int key) const {
        return static_cast<unsigned long>(static_cast<unsigned int>(key));
    }
};

struct StringHash {
    unsigned long operator()(const std::string& key) const {
        unsigned long hash = 2166136261UL;
        for (std::string::size_type i = 0; i < key.size(); ++i) {
            hash ^= static_cast<unsigned char>(key[i]);
            hash *= 16777619UL;
        }
        return hash;
    }
};

template <typename K, typename V, typename HashFunc>
class HashMap {
private:
    struct Entry {
        K key;
        V value;
        Entry* next;

        Entry(const K& entryKey, const V& entryValue)
            : key(entryKey), value(entryValue), next(nullptr) {}
    };

    Entry** buckets;
    int capacity;
    int size;
    HashFunc hash;

    int bucketFor(const K& key, int bucketCount) const {
        return static_cast<int>(hash(key) % static_cast<unsigned long>(bucketCount));
    }

    void rehash(int newCapacity) {
        Entry** newBuckets = new Entry*[newCapacity];
        for (int i = 0; i < newCapacity; ++i) {
            newBuckets[i] = nullptr;
        }

        for (int i = 0; i < capacity; ++i) {
            Entry* entry = buckets[i];
            while (entry) {
                Entry* next = entry->next;
                const int index = bucketFor(entry->key, newCapacity);
                entry->next = newBuckets[index];
                newBuckets[index] = entry;
                entry = next;
            }
        }

        delete[] buckets;
        buckets = newBuckets;
        capacity = newCapacity;
    }

public:
    explicit HashMap(int initialCapacity = 16, const HashFunc& hashFunc = HashFunc())
        : buckets(nullptr), capacity(initialCapacity < 4 ? 4 : initialCapacity),
          size(0), hash(hashFunc) {
        buckets = new Entry*[capacity];
        for (int i = 0; i < capacity; ++i) {
            buckets[i] = nullptr;
        }
    }

    ~HashMap() {
        clear();
        delete[] buckets;
    }

    HashMap(const HashMap&) = delete;
    HashMap& operator=(const HashMap&) = delete;

    void clear() {
        for (int i = 0; i < capacity; ++i) {
            Entry* entry = buckets[i];
            while (entry) {
                Entry* next = entry->next;
                delete entry;
                entry = next;
            }
            buckets[i] = nullptr;
        }
        size = 0;
    }

    bool insert(const K& key, const V& value) {
        Entry* existing = findEntry(key);
        if (existing) {
            existing->value = value;
            return false;
        }

        if ((size + 1) * 4 > capacity * 3) {
            rehash(capacity * 2);
        }

        const int index = bucketFor(key, capacity);
        Entry* entry = new Entry(key, value);
        entry->next = buckets[index];
        buckets[index] = entry;
        ++size;
        return true;
    }

    V* find(const K& key) {
        Entry* entry = findEntry(key);
        return entry ? &entry->value : nullptr;
    }

    const V* find(const K& key) const {
        const Entry* entry = findEntry(key);
        return entry ? &entry->value : nullptr;
    }

    bool remove(const K& key) {
        const int index = bucketFor(key, capacity);
        Entry* previous = nullptr;
        Entry* entry = buckets[index];
        while (entry) {
            if (entry->key == key) {
                if (previous) {
                    previous->next = entry->next;
                } else {
                    buckets[index] = entry->next;
                }
                delete entry;
                --size;
                return true;
            }
            previous = entry;
            entry = entry->next;
        }
        return false;
    }

    int getSize() const {
        return size;
    }

private:
    Entry* findEntry(const K& key) {
        Entry* entry = buckets[bucketFor(key, capacity)];
        while (entry) {
            if (entry->key == key) {
                return entry;
            }
            entry = entry->next;
        }
        return nullptr;
    }

    const Entry* findEntry(const K& key) const {
        const Entry* entry = buckets[bucketFor(key, capacity)];
        while (entry) {
            if (entry->key == key) {
                return entry;
            }
            entry = entry->next;
        }
        return nullptr;
    }
};

template <typename T>
struct Node {
    T data;
    Node<T>* next;

    explicit Node(const T& value) : data(value), next(nullptr) {}
};

template <typename T>
class LinkedList {
public:
    typedef ::Node<T> Node;

private:
    Node* head;
    Node* tail;
    int size;

public:
    LinkedList() : head(nullptr), tail(nullptr), size(0) {}

    ~LinkedList() {
        clear();
    }

    LinkedList(const LinkedList&) = delete;
    LinkedList& operator=(const LinkedList&) = delete;

    void pushBack(const T& value) {
        Node* node = new Node(value);
        if (!tail) {
            head = tail = node;
        } else {
            tail->next = node;
            tail = node;
        }
        ++size;
    }

    void clear() {
        while (head) {
            Node* node = head;
            head = head->next;
            delete node;
        }
        tail = nullptr;
        size = 0;
    }

    int getSize() const {
        return size;
    }

    Node* getHead() const {
        return head;
    }

    Node* getTail() const {
        return tail;
    }

    bool removeNode(Node* target) {
        if (!target) {
            return false;
        }
        Node* previous = nullptr;
        Node* current = head;
        while (current && current != target) {
            previous = current;
            current = current->next;
        }
        if (!current) {
            return false;
        }
        if (previous) {
            previous->next = current->next;
        } else {
            head = current->next;
        }
        if (tail == current) {
            tail = previous;
        }
        delete current;
        --size;
        return true;
    }
};

#pragma once

#include <stdexcept>

template <typename T>
struct MinHeapLess {
    bool operator()(const T& left, const T& right) const {
        return left < right;
    }
};

template <typename T, typename Comparator = MinHeapLess<T>>
class MinHeap {
private:
    T* items;
    int itemCount;
    int capacity;
    Comparator comesBefore;

    void grow() {
        const int newCapacity = capacity == 0 ? 8 : capacity * 2;
        T* expanded = new T[newCapacity];
        for (int i = 0; i < itemCount; ++i) {
            expanded[i] = items[i];
        }
        delete[] items;
        items = expanded;
        capacity = newCapacity;
    }

public:
    explicit MinHeap(const Comparator& comparator = Comparator())
        : items(nullptr), itemCount(0), capacity(0), comesBefore(comparator) {}

    ~MinHeap() {
        clear();
    }

    MinHeap(const MinHeap&) = delete;
    MinHeap& operator=(const MinHeap&) = delete;

    void clear() {
        delete[] items;
        items = nullptr;
        itemCount = 0;
        capacity = 0;
    }

    bool isEmpty() const {
        return itemCount == 0;
    }

    int size() const {
        return itemCount;
    }

    const T& top() const {
        if (isEmpty()) {
            throw std::out_of_range("MinHeap::top called on an empty heap");
        }
        return items[0];
    }

    void push(const T& value) {
        if (itemCount == capacity) {
            grow();
        }

        int index = itemCount++;
        items[index] = value;
        while (index > 0) {
            const int parent = (index - 1) / 2;
            if (!comesBefore(items[index], items[parent])) {
                break;
            }
            T temp = items[index];
            items[index] = items[parent];
            items[parent] = temp;
            index = parent;
        }
    }

    void pop() {
        if (isEmpty()) {
            return;
        }

        --itemCount;
        if (itemCount == 0) {
            return;
        }

        items[0] = items[itemCount];
        int index = 0;
        while (true) {
            const int left = index * 2 + 1;
            const int right = left + 1;
            int best = index;

            if (left < itemCount && comesBefore(items[left], items[best])) {
                best = left;
            }
            if (right < itemCount && comesBefore(items[right], items[best])) {
                best = right;
            }
            if (best == index) {
                break;
            }

            T temp = items[index];
            items[index] = items[best];
            items[best] = temp;
            index = best;
        }
    }
};

bool StationList::addStation(int id, const std::string& name, double km, int tracks) {
    if (id <= 0 || name.empty() || !std::isfinite(km) || km < 0 || tracks < 1 ||
        stationsByID.find(id)) return false;

    StationNode* newNode = new StationNode(id, name, km, tracks);

    stations.pushBack(newNode);
    stationsByID.insert(id, newNode);
    return true;
}

    return station ? *station : nullptr;
}

bool StationList::updateStation(int id, const std::string& name, double km, int tracks) {
    StationNode* station = findStationByID(id);
    if (!station || name.empty() || !std::isfinite(km) || km < 0 || tracks < 1) return false;
    station->name = name;
    station->kmMarker = km;
    station->setMaxTracks(tracks);
    return true;
}

bool StationList::removeStation(int id) {
    StationNode* station = findStationByID(id);
    if (!station) return false;
    LinkedList<Station*>::Node* current = stations.getHead();
    while (current && current->data != station) current = current->next;
    if (!current || !stations.removeNode(current)) return false;
    stationsByID.remove(id);
    delete station;
    return true;
}

    struct ScheduleEventBefore {
        bool operator()(const ScheduleEvent& left, const ScheduleEvent& right) const {
            if (left.time != right.time) return left.time < right.time;
            if (left.train->getPriority() != right.train->getPriority()) {
                return left.train->getPriority() < right.train->getPriority();
            }

    void pushTrainEvents(MinHeap<ScheduleEvent, ScheduleEventBefore>& events,
                        TrainNode* train, int& sequence, int safetyHeadway, int currentTime = 0) {

        while (stopNode) {

            stationEvent.track = stop->getTrackUsed();
            stationEvent.sequence = sequence++;
            stationEvent.trainRevision = train->getScheduleRevision();
            stationEvent.train = train;
            stationEvent.stop = stop;
            stationEvent.trainCode = train->getTrainCode();
            
            // Chỉ nạp nếu sự kiện kết thúc ở TƯƠNG LAI so với currentTime
            if (stationEvent.endTime > stationEvent.time && stationEvent.endTime > currentTime) {
                events.push(stationEvent);
            }

            // --- TẠO SỰ KIỆN KHU GIAN ---
            if (stopNode->next) {
                StopScheduleNode* nextStop = stopNode->next->data;
                ScheduleEvent blockEvent;
                blockEvent.time = stop->getDepartureMinute();
                blockEvent.endTime = nextStop->getArrivalMinute() + safetyHeadway;
                blockEvent.type = 1;
                blockEvent.stationID = stop->getStationID();
                blockEvent.targetStationID = nextStop->getStationID();
                blockEvent.sequence = sequence++;
                blockEvent.trainRevision = train->getScheduleRevision();
                blockEvent.train = train;
                blockEvent.stop = stop;
                blockEvent.nextStop = nextStop;
                blockEvent.trainCode = train->getTrainCode();
                
                // Chỉ nạp nếu sự kiện kết thúc ở TƯƠNG LAI so với currentTime
                if (blockEvent.endTime > blockEvent.time && blockEvent.endTime > currentTime) {
                    events.push(blockEvent);
                }
            }
            stopNode = stopNode->next;
        }
    }

    // KHÓA TÀI NGUYÊN (Sử dụng cho HashMap của bạn)
    struct ResourceKey {
        int type; // 0: Ray trong ga, 1: Khu gian
        int id1;  
        int id2;  

        ResourceKey(int t = 0, int a = 0, int b = 0) {
            type = t;
            if (type == 1 && a > b) {
                id1 = b;
                id2 = a;
            } else {
                id1 = a;
                id2 = b;
            }
        }

        // Toán tử == để HashMap kiểm tra trùng lặp khóa
        bool operator==(const ResourceKey& other) const {
            return type == other.type && id1 == other.id1 && id2 == other.id2;
        }
    };

    // HÀM BĂM (Functor) DÀNH CHO KHÓA TÀI NGUYÊN
    struct ResourceKeyHash {
        unsigned long operator()(const ResourceKey& k) const {
            unsigned long hash = 2166136261UL;
            hash = (hash ^ static_cast<unsigned long>(k.type + 1)) * 16777619UL;
            hash = (hash ^ static_cast<unsigned long>(k.id1)) * 16777619UL;
            hash = (hash ^ static_cast<unsigned long>(k.id2)) * 16777619UL;
            return hash;
        }
    };

    // Hàm phụ trợ trích xuất Key từ sự kiện
    ResourceKey getEventKey(const ScheduleEvent& ev) {
        if (ev.type == 0) {
            return ResourceKey(0, ev.stationID, ev.track);
        }
        return ResourceKey(1, ev.stationID, ev.targetStationID);
    }
}

TrainList::TrainList()
    : trains(), trainsByCode(), structureRevision(0), logStart(0), logCount(0),
      maintenanceWindows(), maintenanceCount(0) {}

TrainList::~TrainList() {
    clear();
}

void TrainList::clear() {
    LinkedList<TrainNode*>::Node* train = trains.getHead();
    while (train) {
        delete train->data;
        train = train->next;
    }
    trains.clear();
    trainsByCode.clear();
    maintenanceCount = 0;
    ++structureRevision;
}

LinkedList<TrainNode*>::Node* TrainList::getHead() const {
    return trains.getHead();
}

bool TrainList::addTrain(const string& code, int direction, int priority) {
    if (code.empty() || (direction != 0 && direction != 1) ||
        priority > 3 ||
        trainsByCode.find(code)) return false;
    TrainNode* newNode = new TrainNode(code, direction, priority);
    trains.pushBack(newNode);
    trainsByCode.insert(code, newNode);
    ++structureRevision;
    return true;
}

TrainNode* TrainList::findTrainByCode(const string& code) const {
    TrainNode* const* train = trainsByCode.find(code);
    return train ? *train : nullptr;
}

bool TrainList::updateTrain(const string& code, int direction, int priority,
                            const StationList* stationList) {
    TrainNode* train = findTrainByCode(code);
    if (!train || !stationList || (direction != 0 && direction != 1) || priority > 3) {
        return false;
    }
    // Chỉ đổi hướng khi thứ tự các ga hiện tại vẫn hợp lệ theo mốc km.
    StopScheduleNode* previous = nullptr;
    for (LinkedList<StopScheduleNode*>::Node* stopNode = train->getHeadSchedule();
         stopNode; stopNode = stopNode->next) {
        StopScheduleNode* current = stopNode->data;
        StationNode* currentStation = stationList->findStationByID(current->getStationID());
        StationNode* previousStation = previous
            ? stationList->findStationByID(previous->getStationID()) : nullptr;
        if (!currentStation || (previous &&
            (!previousStation ||
             (direction == 1
                ? currentStation->getKmMarker() <= previousStation->getKmMarker()
                : currentStation->getKmMarker() >= previousStation->getKmMarker())))) {
            return false;
        }
        previous = current;
    }
    train->setDirection(direction);
    train->setPriority(priority > 0 ? priority : TrainNode::inferPriorityFromCode(code));
    train->bumpScheduleRevision();
    ++structureRevision;
    return true;
}

bool TrainList::removeTrain(const string& code) {
    TrainNode* train = findTrainByCode(code);
    if (!train) return false;
    LinkedList<TrainNode*>::Node* node = trains.getHead();
    while (node && node->data != train) node = node->next;
    if (!node || !trains.removeNode(node)) return false;
    trainsByCode.remove(code);
    delete train;
    ++structureRevision;
    return true;
}

bool TrainList::addCarriageToTrain(const string& code, int carriageNo,
                                   const string& type) {
    TrainNode* train = findTrainByCode(code);
    if (!train || !train->addCarriage(carriageNo, type)) return false;
    ++structureRevision;
    return true;
}

bool TrainList::updateCarriage(const string& code, int carriageNo,
                               const string& type) {
    TrainNode* train = findTrainByCode(code);
    CarriageNode* carriage = train ? train->findCarriage(carriageNo) : nullptr;
    if (!carriage || type.empty()) return false;
    carriage->setType(type);
    ++structureRevision;
    return true;
}

bool TrainList::removeCarriage(const string& code, int carriageNo) {
    TrainNode* train = findTrainByCode(code);
    if (!train) return false;
    LinkedList<CarriageNode*>::Node* node = train->getHeadCarriage();
    while (node && node->data->getCarriageNo() != carriageNo) node = node->next;
    if (!node) return false;
    delete node->data;
    train->carriages.removeNode(node);
    ++structureRevision;
    return true;
}

bool TrainList::addStopScheduleToTrain(const string& code, int stationID,
                                       int arrival, int departure, int track,
                                       StationList* stationList) {
    TrainNode* train = findTrainByCode(code);
    StationNode* station = stationList ? stationList->findStationByID(stationID) : nullptr;
    LinkedList<StopScheduleNode*>::Node* tailNode =
        train ? train->getTailSchedule() : nullptr;
    StopScheduleNode* tail = tailNode ? tailNode->data : nullptr;
    StationNode* previousStation = tail && stationList
        ? stationList->findStationByID(tail->getStationID()) : nullptr;
    if (!train || !station || track > station->getMaxTracks() ||
        (tail && (arrival < tail->getDepartureMinute() || !previousStation ||
            (train->getDirection() == 1
                ? station->getKmMarker() <= previousStation->getKmMarker()
                : station->getKmMarker() >= previousStation->getKmMarker())))) return false;
    if (!train->addSchedule(stationID, arrival, departure, track)) return false;
    train->bumpScheduleRevision();
    ++structureRevision;
    // Kiểm tra và tái điều phối ngay để mọi nơi gọi API đều được xử lý xung đột.
    resolveConflictAndScheduleSiding(stationList);
    return true;
}

bool TrainList::updateStopSchedule(const string& code, int stationID, int arrival,
                                   int departure, int track,
                                   const StationList* stationList) {
    TrainNode* train = findTrainByCode(code);
    if (!train || arrival < 0 || departure < arrival || track < 1) return false;
    LinkedList<StopScheduleNode*>::Node* target = train->getHeadSchedule();
    while (target && target->data->getStationID() != stationID) target = target->next;
    StationNode* station = stationList ? stationList->findStationByID(stationID) : nullptr;
    if (!target || !station || track > station->getMaxTracks()) return false;

    StopScheduleNode* previous = train->findPreviousStop(target->data);
    StopScheduleNode* next = target->next ? target->next->data : nullptr;
    StationNode* previousStation = previous && stationList
        ? stationList->findStationByID(previous->getStationID()) : nullptr;
    StationNode* nextStation = next && stationList
        ? stationList->findStationByID(next->getStationID()) : nullptr;
    if ((previous && (arrival < previous->getDepartureMinute() || !previousStation ||
            (train->getDirection() == 1
                ? station->getKmMarker() <= previousStation->getKmMarker()
                : station->getKmMarker() >= previousStation->getKmMarker()))) ||
        (next && (departure > next->getArrivalMinute() || !nextStation ||
            (train->getDirection() == 1
                ? station->getKmMarker() >= nextStation->getKmMarker()
                : station->getKmMarker() <= nextStation->getKmMarker())))) return false;

    target->data->setArrivalMinute(arrival);
    target->data->setDepartureMinute(departure);
    target->data->setTrackUsed(track);
    train->bumpScheduleRevision();
    ++structureRevision;
    return true;
}

bool TrainList::removeStopSchedule(const string& code, int stationID) {
    TrainNode* train = findTrainByCode(code);
    if (!train || train->schedule.getSize() <= 2) return false;
    LinkedList<StopScheduleNode*>::Node* target = train->getHeadSchedule();
    while (target && target->data->getStationID() != stationID) target = target->next;
    if (!target) return false;
    delete target->data;
    train->schedule.removeNode(target);
    train->bumpScheduleRevision();
    ++structureRevision;
    return true;
}

bool TrainList::hasStationStop(int stationID) const {
    for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead(); trainNode; trainNode = trainNode->next) {
        for (LinkedList<StopScheduleNode*>::Node* stop = trainNode->data->getHeadSchedule(); stop; stop = stop->next) {
            if (stop->data->getStationID() == stationID) return true;
        }
    }
    return false;
}

bool TrainList::hasTrackAbove(int stationID, int track) const {
    for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead(); trainNode; trainNode = trainNode->next) {
        for (LinkedList<StopScheduleNode*>::Node* stop = trainNode->data->getHeadSchedule(); stop; stop = stop->next) {
            if (stop->data->getStationID() == stationID && stop->data->getTrackUsed() > track) return true;
        }
    }
    return false;
}

bool TrainList::hasMaintenanceForStation(int stationID) const {
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& window = maintenanceWindows[i];
        if ((window.resourceType == 0 || window.resourceType == 2) &&
            window.firstID == stationID) return true;
        if (window.resourceType == 1 &&
            (window.firstID == stationID || window.secondID == stationID)) return true;
    }
    return false;
}

bool TrainList::hasMaintenanceBlockForStation(int stationID) const {
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& window = maintenanceWindows[i];
        if (window.resourceType == 1 &&
            (window.firstID == stationID || window.secondID == stationID)) return true;
    }
    return false;
}

bool TrainList::hasMaintenanceTrackAbove(int stationID, int track) const {
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& window = maintenanceWindows[i];
        if (window.resourceType == 2 && window.firstID == stationID &&
            window.secondID > track) return true;
    }
    return false;
}

bool TrainList::addMaintenance(const string& resource, int firstID, int secondID,
                               int startMinute, int endMinute) {
    int type = -1;
    if (resource == "STATION") type = 0;
    else if (resource == "BLOCK") type = 1;
    else if (resource == "TRACK") type = 2;
    if (type < 0 || firstID <= 0 || startMinute < 0 ||
        endMinute <= startMinute ||
        (type == 2 && secondID < 1) ||
        (type == 1 && (secondID <= 0 || firstID == secondID))) return false;
    if (type == 1 && firstID > secondID) {
        const int temp = firstID;
        firstID = secondID;
        secondID = temp;
    }
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& existing = maintenanceWindows[i];
        if (existing.resourceType == type && existing.firstID == firstID &&
            existing.secondID == secondID && existing.startMinute == startMinute &&
            existing.endMinute == endMinute) return true;
    }
    if (maintenanceCount >= MAX_MAINTENANCE_WINDOWS) return false;
    maintenanceWindows[maintenanceCount++] =
        {type, firstID, secondID, startMinute, endMinute};
    return true;
}

bool TrainList::removeMaintenance(const string& resource, int firstID, int secondID,
                                 int startMinute, int endMinute) {
    int type = -1;
    if (resource == "STATION") type = 0;
    else if (resource == "BLOCK") type = 1;
    else if (resource == "TRACK") type = 2;
    if (type < 0) return false;
    if (type == 1 && firstID > secondID) {
        const int temp = firstID;
        firstID = secondID;
        secondID = temp;
    }
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& existing = maintenanceWindows[i];
        if (existing.resourceType == type && existing.firstID == firstID &&
            existing.secondID == secondID && existing.startMinute == startMinute &&
            existing.endMinute == endMinute) {
            for (int j = i + 1; j < maintenanceCount; ++j) {
                maintenanceWindows[j - 1] = maintenanceWindows[j];
            }
            --maintenanceCount;
            return true;
        }
    }
    return false;
}

bool TrainList::isUnderMaintenanceAt(const string& resource, int firstID,
                                     int secondID, int minute) const {
    int type = -1;
    if (resource == "STATION") type = 0;
    else if (resource == "BLOCK") type = 1;
    else if (resource == "TRACK") type = 2;
    if (type < 0) return false;
    if (type == 1 && firstID > secondID) {
        const int temp = firstID;
        firstID = secondID;
        secondID = temp;
    }
    for (int i = 0; i < maintenanceCount; ++i) {
        const MaintenanceWindow& repair = maintenanceWindows[i];
        if (repair.resourceType == type && repair.firstID == firstID &&
            repair.secondID == secondID && minute >= repair.startMinute &&
            minute < repair.endMinute) return true;
    }
    return false;
}

void TrainList::addConflictLog(const string& logMessage) {
    if (logCount < MAX_LOGS) {
        const int insertIndex = (logStart + logCount) % MAX_LOGS;
        conflictLogs[insertIndex] = logMessage;
        ++logCount;
    } else {
        conflictLogs[logStart] = logMessage;
        logStart = (logStart + 1) % MAX_LOGS;
    }
}

string TrainList::getLog(int index) const {
    if (index < 0 || index >= logCount) return "";
    return conflictLogs[(logStart + index) % MAX_LOGS];
}

int TrainList::findAvailableSidingTrack(int stationID, int arrival, int departure,
                                        int maxTracks, TrainNode* excludeTrain) const {
    if (maxTracks <= 1) return -1;

    const int HEADWAY = 10;
    for (int candidateTrack = 2; candidateTrack <= maxTracks; ++candidateTrack) {
        bool occupied = false;
        const int requestedEnd = departure + HEADWAY;
        for (int i = 0; i < maintenanceCount; ++i) {
            const MaintenanceWindow& repair = maintenanceWindows[i];
            if (repair.resourceType == 0 && repair.firstID == stationID &&
                arrival < repair.endMinute && repair.startMinute < requestedEnd) {
                occupied = true;
                break;
            }
            if (repair.resourceType == 2 && repair.firstID == stationID &&
                repair.secondID == candidateTrack &&
                arrival < repair.endMinute && repair.startMinute < requestedEnd) {
                occupied = true;
                break;
            }
        }
        if (occupied) continue;
        for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead();
             trainNode && !occupied; trainNode = trainNode->next) {
            TrainNode* train = trainNode->data;
            if (train == excludeTrain) continue;

            for (LinkedList<StopScheduleNode*>::Node* stopNode =
                     train->getHeadSchedule(); stopNode; stopNode = stopNode->next) {
                StopScheduleNode* stop = stopNode->data;
                if (stop->getStationID() != stationID ||
                    stop->getTrackUsed() != candidateTrack) continue;
                // Tàu kết thúc hành trình được xem là chiếm chỗ đỗ tại ga cuối.
                if (!stopNode->next) {
                    occupied = true;
                    break;
                }

                const int overlapStart = arrival > stop->getArrivalMinute()
                    ? arrival : stop->getArrivalMinute();
                const int stopEnd = stop->getDepartureMinute() + HEADWAY;
                const int overlapEnd = requestedEnd < stopEnd
                    ? requestedEnd : stopEnd;
                if (overlapStart < overlapEnd) {
                    occupied = true;
                    break;
                }
            }
        }
        if (!occupied) return candidateTrack;
    }
    return -1;
}

void TrainList::resolveConflictAndScheduleSiding(StationList* stationList) {
    if (!stationList) return;

    const int SAFETY_HEADWAY = 10;
    for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead();
         trainNode; trainNode = trainNode->next) {
        TrainNode* train = trainNode->data;
        LinkedList<StopScheduleNode*>::Node* lastStopNode = train->getTailSchedule();
        StopScheduleNode* lastStop = lastStopNode ? lastStopNode->data : nullptr;
        StationNode* station = lastStop
            ? stationList->findStationByID(lastStop->getStationID()) : nullptr;
        if (!lastStop || !station || station->getMaxTracks() < 2 ||
            lastStop->getTrackUsed() != 1) continue;
        // Ưu tiên dành ray phụ cho tàu đã kết thúc tuyến để không hiển thị đậu trên ray chính.
        const int parkingEnd = lastStop->getDepartureMinute() + 60;
        const int siding = findAvailableSidingTrack(lastStop->getStationID(),
            lastStop->getArrivalMinute(), parkingEnd, station->getMaxTracks(), train);
        if (siding > 1) {
            lastStop->setTrackUsed(siding);
            addConflictLog("[Ga ket thuc " + to_string(lastStop->getStationID()) +
                           "] Tau " + train->getTrainCode() + " dau Ray phu R" +
                           to_string(siding));
        }
    }

    MinHeap<ScheduleEvent, ScheduleEventBefore> events;
    int sequence = 0;
    for (LinkedList<TrainNode*>::Node* trainNode = trains.getHead();
         trainNode; trainNode = trainNode->next) {
        TrainNode* train = trainNode->data;
        pushTrainEvents(events, train, sequence, SAFETY_HEADWAY, 0);
    }

    HashMap<ResourceKey, ScheduleEvent, ResourceKeyHash> activeResources(32);

    while (!events.isEmpty()) {
        const ScheduleEvent current = events.top();
        events.pop();
        if (!isCurrentEvent(current)) {
            continue;
        }

        // Bảo trì được xử lý như một xung đột có thời hạn trước khi chiếm tài nguyên.
        const MaintenanceWindow* blockedByRepair = nullptr;
        int maintenanceEnd = 0;
        for (int i = 0; i < maintenanceCount; ++i) {
            const MaintenanceWindow& repair = maintenanceWindows[i];
            bool matches = false;
            if (repair.resourceType == 0 && current.type == 0) {
                matches = repair.firstID == current.stationID;
            } else if (repair.resourceType == 1 && current.type == 1) {
                const int first = min(current.stationID, current.targetStationID);
                const int second = max(current.stationID, current.targetStationID);
                matches = repair.firstID == first && repair.secondID == second;
            } else if (repair.resourceType == 2 && current.type == 0) {
                matches = repair.firstID == current.stationID &&
                          repair.secondID == current.track;
            }
            if (matches && current.time < repair.endMinute &&
                repair.startMinute < current.endTime &&
                (!blockedByRepair || repair.endMinute > maintenanceEnd)) {
                blockedByRepair = &repair;
                maintenanceEnd = repair.endMinute;
            }
        }

        if (blockedByRepair) {
            bool movedToAvailableTrack = false;
            if (current.type == 0 && blockedByRepair->resourceType == 2) {
                StationNode* station = stationList->findStationByID(current.stationID);
                if (station) {
                    for (int track = 1; track <= station->getMaxTracks(); ++track) {
                        if (track == current.track) continue;
                        bool underRepair = false;
                        for (int i = 0; i < maintenanceCount; ++i) {
                            const MaintenanceWindow& repair = maintenanceWindows[i];
                            if ((repair.resourceType == 0 && repair.firstID == current.stationID) ||
                                (repair.resourceType == 2 && repair.firstID == current.stationID &&
                                 repair.secondID == track &&
                                 current.time < repair.endMinute &&
                                 repair.startMinute < current.endTime)) {
                                underRepair = true;
                                break;
                            }
                        }
                        if (underRepair) continue;
                        if (track > 1 && findAvailableSidingTrack(current.stationID,
                                current.stop->getArrivalMinute(),
                                current.stop->getDepartureMinute(),
                                station->getMaxTracks(), current.train) != track) continue;
                        current.stop->setTrackUsed(track);
                        current.train->bumpScheduleRevision();
                        addConflictLog("[Bao tri] Tau " + current.train->getTrainCode() +
                            " doi sang Ray " + to_string(track) + " tai Ga " +
                            to_string(current.stationID));
                        pushTrainEvents(events, current.train, sequence,
                                        SAFETY_HEADWAY, current.time);
                        movedToAvailableTrack = true;
                        break;
                    }
                }
            }
            if (movedToAvailableTrack) continue;

            const int wait = maintenanceEnd - current.time;
            if (current.type == 0) {
                StopScheduleNode* previous = current.train->findPreviousStop(current.stop);
                if (previous) previous->setDepartureMinute(previous->getDepartureMinute() + wait);
                current.train->applyDelayPropagation(current.stationID, wait, true);
            } else {
                current.stop->setDepartureMinute(current.stop->getDepartureMinute() + wait);
                current.train->applyDelayPropagation(current.targetStationID, wait, true);
            }
            addConflictLog("[Bao tri] Tau " + current.train->getTrainCode() +
                " cho " + to_string(wait) + "p truoc khi qua tai nguyen dang sua");
            current.train->bumpScheduleRevision();
            pushTrainEvents(events, current.train, sequence, SAFETY_HEADWAY, current.time);
            continue;
        }

        ResourceKey resKey = getEventKey(current);
        ScheduleEvent* occupiedEvent = activeResources.find(resKey);

        // 1. Giải phóng tài nguyên nếu tàu cũ đã rời đi hoặc đổi lịch
        if (occupiedEvent && (!isCurrentEvent(*occupiedEvent) || occupiedEvent->endTime <= current.time)) {
            activeResources.remove(resKey);
            occupiedEvent = nullptr;
        }

        // 2. Nếu tài nguyên trống, tàu hiện tại chiếm dụng
        if (!occupiedEvent) {
            activeResources.insert(resKey, current);
            continue;
        }

        // 3. Cùng một tàu kéo dài thời gian đỗ
        if (occupiedEvent->train == current.train) {
            if (current.endTime > occupiedEvent->endTime) {
                occupiedEvent->endTime = current.endTime;
            }
            continue;
        }

        // 4. Phát hiện xung đột, kiểm tra ưu tiên nhường đường
        const ScheduleEvent other = *occupiedEvent;
        const bool currentYields =
            current.train->getPriority() != other.train->getPriority()
                ? current.train->getPriority() > other.train->getPriority()
                : (current.time != other.time
                       ? current.time > other.time
                       : current.sequence > other.sequence);
        
        const ScheduleEvent& yielding = currentYields ? current : other;
        const ScheduleEvent& priority = currentYields ? other : current;
        TrainNode* yieldingTrain = yielding.train;
        TrainNode* priorityTrain = priority.train;
        bool changed = false;

        if (yielding.type == 0) {
            StationNode* station = stationList->findStationByID(yielding.stationID);
            const int trackCount = station ? station->getMaxTracks() : 1;
            const int siding = findAvailableSidingTrack(
                yielding.stationID, yielding.stop->getArrivalMinute(),
                yielding.stop->getDepartureMinute(), trackCount, yieldingTrain);

            if (siding != -1) {
                yielding.stop->setTrackUsed(siding);
                addConflictLog("[Ga " + to_string(yielding.stationID) + "] " +
                            yieldingTrain->getTrainCode() + " vao Ray " + to_string(siding));
                changed = true;
            } else {
                const int wait = priority.endTime - yielding.time;
                if (wait > 0) {
                    StopScheduleNode* previous = yieldingTrain->findPreviousStop(yielding.stop);
                    if (previous) {
                        previous->setDepartureMinute(previous->getDepartureMinute() + wait);
                    }
                    yieldingTrain->applyDelayPropagation(yielding.stationID, wait, true);
                    changed = true;
                }
            }
        } else { // type == 1 (Xung đột khu gian đường đơn)
        const int wait = priority.endTime - yielding.time;
        if (wait > 0) {
            
            StationNode* waitStation = stationList->findStationByID(yielding.stationID);
            const int waitTracks = waitStation ? waitStation->getMaxTracks() : 1;
            const int siding = findAvailableSidingTrack(
                yielding.stationID, 
                yielding.stop->getArrivalMinute(),
                yielding.stop->getDepartureMinute() + wait, // Khung giờ dự kiến đỗ mới
                waitTracks, 
                yieldingTrain
            );
            if (siding != -1) {
                yielding.stop->setTrackUsed(siding);
                yielding.stop->setDepartureMinute(yielding.stop->getDepartureMinute() + wait);
                yieldingTrain->applyDelayPropagation(yielding.targetStationID, wait, true);
            } else {
                StopScheduleNode* previous = yieldingTrain->findPreviousStop(yielding.stop);
                if (previous) {
                    previous->setDepartureMinute(previous->getDepartureMinute() + wait);
                    yieldingTrain->applyDelayPropagation(yielding.stationID, wait, true);
                } else {
                    yielding.stop->setDepartureMinute(yielding.stop->getDepartureMinute() + wait);
                    yieldingTrain->applyDelayPropagation(yielding.targetStationID, wait, true);
                }
            }
            addConflictLog("[Khu gian " + to_string(yielding.stationID) + "->" +
                        to_string(yielding.targetStationID) + "] " +
                        yieldingTrain->getTrainCode() + " cho " +
                        priorityTrain->getTrainCode() + " +" + to_string(wait) + "p" +
                        (siding != -1 ? " (doi sang Ray " + to_string(siding) + ")" : ""));
            changed = true;
        }
    }

        if (changed) {
            // Thay thế tàu bị loại ra khỏi tài nguyên nếu cần
            if (!currentYields) {
                activeResources.insert(resKey, current);
            }
            yieldingTrain->bumpScheduleRevision();
            pushTrainEvents(events, yieldingTrain, sequence, SAFETY_HEADWAY, current.time);
        }
    }
}

bool TrainList::applyDelayAndReschedule(const string& trainCode, int fromStationID,
                                        int delayMinutes, StationList* stationList) {
    TrainNode* train = findTrainByCode(trainCode);
    if (!train) return false;

    addConflictLog("[Su co] Tau " + trainCode + " bi tre +" +
                   to_string(delayMinutes) + "p tu Ga " +
                   to_string(fromStationID));
    const bool delayed =
        train->applyDelayPropagation(fromStationID, delayMinutes, true);
    if (delayed) {
        resolveConflictAndScheduleSiding(stationList);
    }
    return delayed;
}

#pragma once
#include <string>
#include "CarriageNode.h"
#include "StopScheduleNode.h"
#include "LinkedList.h"

struct Train {
    std::string trainCode;
    int direction; // 1: Nam, 0: Bac
    int priority;  // 1: Tau khach nhanh SE, 2: Tau thuong TN, 3: Tau hang
    LinkedList<Carriage*> carriages;
    LinkedList<StopSchedule*> schedule;
    unsigned long scheduleRevision;

public:
    static int inferPriorityFromCode(const std::string& code) {
        if (code.rfind("SE", 0) == 0) return 1; // Ưu tiên cao nhất: SE1, SE2, SE3, SE4...
        if (code.rfind("TN", 0) == 0) return 2; // Ưu tiên trung bình: TN1, TN2...
        return 3;                               // Ưu tiên thấp nhất: Tàu hàng (HH, GS...)
    }

    Train(const std::string& code, int dir, int prio = -1)
        : trainCode(code), direction(dir), 
          priority(prio > 0 ? prio : inferPriorityFromCode(code)),
          scheduleRevision(0) {}

    ~Train() {
        LinkedList<Carriage*>::Node* carriage = carriages.getHead();
        while (carriage) {
            delete carriage->data;
            carriage = carriage->next;
        }
        LinkedList<StopSchedule*>::Node* stop = schedule.getHead();
        while (stop) {
            delete stop->data;
            stop = stop->next;
        }
    }

    // Getters
    std::string getTrainCode() const { return trainCode; }
    int getDirection() const { return direction; }
    int getPriority() const { return priority; }
    LinkedList<Carriage*>::Node* getHeadCarriage() const { return carriages.getHead(); }
    LinkedList<StopSchedule*>::Node* getHeadSchedule() const { return schedule.getHead(); }
    LinkedList<StopSchedule*>::Node* getTailSchedule() const { return schedule.getTail(); }
    unsigned long getScheduleRevision() const { return scheduleRevision; }

    // Setters
    void setPriority(int prio) { priority = prio; }
    void setDirection(int dir) { direction = dir; }
    void bumpScheduleRevision() { ++scheduleRevision; }

    // Chèn toa xe độ phức tạp O(1)
    CarriageNode* addCarriage(int cNo, const std::string& type) {
        if (cNo <= 0 || type.empty() || findCarriage(cNo)) return nullptr;
        CarriageNode* carriage = new CarriageNode(cNo, type);
        carriages.pushBack(carriage);
        return carriage;
    }

    // Chèn lịch dừng độ phức tạp O(1)
    StopScheduleNode* addSchedule(int stID, int arr, int dep, int track) {
        if (stID <= 0 || arr < 0 || dep < arr || track < 1) return nullptr;
        for (LinkedList<StopSchedule*>::Node* stop = schedule.getHead(); stop; stop = stop->next) {
            if (stop->data->getStationID() == stID) return nullptr;
        }
        StopScheduleNode* newStop = new StopScheduleNode(stID, arr, dep, track);
        schedule.pushBack(newStop);
        return newStop;
    }

    // Tìm ga dừng liền trước một ga cụ thể trong hành trình của tàu
    StopScheduleNode* findPreviousStop(StopScheduleNode* targetStop) const {
        LinkedList<StopSchedule*>::Node* curr = schedule.getHead();
        if (!curr || curr->data == targetStop) return nullptr;
        while (curr && (!curr->next || curr->next->data != targetStop)) {
            curr = curr->next;
        }
        return curr ? curr->data : nullptr;
    }

    // Lan truyền dời giờ các ga tiếp theo khi xảy ra trễ
    bool applyDelayPropagation(int fromStationID, int delayMinutes, bool delayArrivalAtFirst = true) {
        LinkedList<StopSchedule*>::Node* curr = schedule.getHead();
        bool found = false;

        while (curr != nullptr) {
            StopScheduleNode* stop = curr->data;
            if (stop->getStationID() == fromStationID) {
                found = true;
                if (delayArrivalAtFirst) {
                    stop->setArrivalMinute(stop->getArrivalMinute() + delayMinutes);
                }
                stop->setDepartureMinute(stop->getDepartureMinute() + delayMinutes);
            } else if (found) {
                stop->setArrivalMinute(stop->getArrivalMinute() + delayMinutes);
                stop->setDepartureMinute(stop->getDepartureMinute() + delayMinutes);
            }
            curr = curr->next;
        }
        return found;
    }

    CarriageNode* findCarriage(int carriageNo) const {
        LinkedList<Carriage*>::Node* current = carriages.getHead();
        while (current) {
            CarriageNode* carriage = current->data;
            if (carriage->getCarriageNo() == carriageNo) return carriage;
            current = current->next;
        }
        return nullptr;
    }
};

typedef Train TrainNode;

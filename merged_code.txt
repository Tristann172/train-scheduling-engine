#pragma once

#include <string>

class CarriageNode {
private:
    int carriageNo;
    std::string type;
    int totalSeats;
    CarriageNode* nextCarriage;

public:
    CarriageNode(int cNo, const std::string& cType, int seats)
        : carriageNo(cNo), type(cType), totalSeats(seats), nextCarriage(NULL) {}

    int getCarriageNo() const { return carriageNo; }
    std::string getType() const { return type; }
    int getTotalSeats() const { return totalSeats; }
    CarriageNode* getNextCarriage() const { return nextCarriage; }

    void setNextCarriage(CarriageNode* next) { nextCarriage = next; }
};

// ==================== STATIONS ====================
void FileManager::loadStations(const string& filename, StationList& stationList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string strID, name, strKm, strTracks;

        if (getline(ss, strID, '|') && getline(ss, name, '|') &&
            getline(ss, strKm, '|') && getline(ss, strTracks, '|')) {
            stationList.addStation(stoi(strID), name, stod(strKm), stoi(strTracks));
        }
    }
    inFile.close();
}

void FileManager::saveStations(const string& filename, const StationList& stationList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    StationNode* st = stationList.getHead();
    while (st != nullptr) {
        outFile << st->getStationID() << "|"
                << st->getName() << "|"
                << st->getKmMarker() << "|"
                << st->getNumberOfTracks() << "\n";
        st = st->getNextStation();
    }
    outFile.close();
}

// ==================== TRAINS ====================
void FileManager::loadTrains(const string& filename, TrainList& trainList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    TrainNode* currentTrain = nullptr;

    while (getline(inFile, line)) {
        if (line.empty()) continue;

        stringstream ss(line);
        string token;
        getline(ss, token, '|');

        if (token == "CARRIAGE") {
            if (currentTrain) {
                string strNo, type, strSeats;
                getline(ss, strNo, '|');
                getline(ss, type, '|');
                getline(ss, strSeats, '|');
                // Gọi đúng thuộc tính hoặc phương thức thêm toa xe hiện có
                if (currentTrain->getHeadCarriage() == nullptr) {
                    currentTrain->addCarriage(stoi(strNo), type, stoi(strSeats));
                } else {
                    currentTrain->addCarriage(stoi(strNo), type, stoi(strSeats));
                }
            }
        } 
        else if (token == "STOP") {
            if (currentTrain) {
                string strStID, strArr, strDep, strTrack;
                getline(ss, strStID, '|');
                getline(ss, strArr, '|');
                getline(ss, strDep, '|');
                getline(ss, strTrack, '|');
                currentTrain->addSchedule(stoi(strStID), stoi(strArr), stoi(strDep), stoi(strTrack));
            }
        } 
        else {
            // Đọc dòng đầu tiên của Tàu: Mã Tàu | Hướng chạy
            string trainCode = token;
            string strDir;
            if (getline(ss, strDir, '|')) {
                trainList.addTrain(trainCode, stoi(strDir));
                currentTrain = trainList.findTrainByCode(trainCode);
            }
        }
    }
    inFile.close();
}

void FileManager::saveTrains(const string& filename, const TrainList& trainList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    TrainNode* tr = trainList.getHead();
    while (tr != nullptr) {
        outFile << tr->getTrainCode() << "|" << tr->getDirection() << "\n";

        CarriageNode* c = tr->getHeadCarriage();
        while (c != nullptr) {
            outFile << "CARRIAGE|" << c->getCarriageNo() << "|" << c->getType() << "|" << c->getTotalSeats() << "\n";
            c = c->getNextCarriage();
        }

        StopScheduleNode* s = tr->getHeadSchedule();
        while (s != nullptr) {
            outFile << "STOP|" << s->getStationID() << "|" << s->getArrivalMinute() << "|" << s->getDepartureMinute() << "|" << s->getTrackUsed() << "\n";
            s = s->getNextStop();
        }

        tr = tr->getNextTrain();
    }
    outFile.close();
}

// ==================== TICKETS ====================
void FileManager::loadTickets(const string& filename, TicketList& ticketList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string tID, trCode, strCNo, strSNo, strFrom, strTo;

        if (getline(ss, tID, '|') && getline(ss, trCode, '|') &&
            getline(ss, strCNo, '|') && getline(ss, strSNo, '|') &&
            getline(ss, strFrom, '|') && getline(ss, strTo, '|')) {
            
            ticketList.addTicket(tID, trCode, stoi(strCNo), stoi(strSNo), stoi(strFrom), stoi(strTo));
        }
    }
    inFile.close();
}

void FileManager::saveTickets(const string& filename, const TicketList& ticketList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    TicketNode* tk = ticketList.getHead();
    while (tk != nullptr) {
        outFile << tk->getTicketID() << "|"
                << tk->getTrainCode() << "|"
                << tk->getCarriageNo() << "|"
                << tk->getSeatNo() << "|"
                << tk->getFromStationID() << "|"
                << tk->getToStationID() << "\n";
        tk = tk->getNextTicket();
    }
    outFile.close();
}

// Hàm tính vị trí Km của tàu chuẩn xác cho mọi mốc thời gian
static double calculateTrainKm(TrainNode* train, StationList* stationList, int currentMinute, int& trackUsedOut) {
    trackUsedOut = 0;
    StopScheduleNode* currStop = train->getHeadSchedule();
    if (!currStop) return -1.0;

    // Lấy thông tin ga đầu tiên và ga cuối cùng trong lịch trình
    StopScheduleNode* firstStop = currStop;
    StopScheduleNode* lastStop = currStop;
    while (lastStop->getNextStop() != nullptr) {
        lastStop = lastStop->getNextStop();
    }

    // 1. Trường hợp thời gian chưa đến giờ tàu xuất phát -> Cho tàu chờ sẵn tại Ga Đầu
    if (currentMinute < firstStop->getArrivalMinute()) {
        StationNode* st = stationList->findStationByID(firstStop->getStationID());
        if (st) {
            trackUsedOut = firstStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 2. Trường hợp thời gian đã qua giờ tàu về ga cuối -> Cho tàu đỗ tại Ga Cuối
    if (currentMinute > lastStop->getDepartureMinute()) {
        StationNode* st = stationList->findStationByID(lastStop->getStationID());
        if (st) {
            trackUsedOut = lastStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 3. Kiểm tra nếu tàu đang đỗ tại một ga trung gian
    currStop = train->getHeadSchedule();
    while (currStop != nullptr) {
        if (currentMinute >= currStop->getArrivalMinute() && currentMinute <= currStop->getDepartureMinute()) {
            StationNode* st = stationList->findStationByID(currStop->getStationID());
            if (st) {
                trackUsedOut = currStop->getTrackUsed();
                return st->getKmMarker();
            }
        }
        currStop = currStop->getNextStop();
    }

    // 4. Trường hợp tàu đang di chuyển giữa 2 ga (Khu gian)
    currStop = train->getHeadSchedule();
    while (currStop != nullptr && currStop->getNextStop() != nullptr) {
        StopScheduleNode* nextStop = currStop->getNextStop();

        if (currentMinute > currStop->getDepartureMinute() && currentMinute < nextStop->getArrivalMinute()) {
            StationNode* st1 = stationList->findStationByID(currStop->getStationID());
            StationNode* st2 = stationList->findStationByID(nextStop->getStationID());

            if (st1 && st2) {
                double km1 = st1->getKmMarker();
                double km2 = st2->getKmMarker();
                int t1 = currStop->getDepartureMinute();
                int t2 = nextStop->getArrivalMinute();

                double ratio = (double)(currentMinute - t1) / (t2 - t1);
                return km1 + ratio * (km2 - km1);
            }
        }
        currStop = currStop->getNextStop();
    }

    return -1.0;
}

void GraphicalSimulationEngine::runSFMLSimulation(StationList* stationList, TrainList* trainList) {
    if (!stationList || !trainList) return;

    sf::RenderWindow window(sf::VideoMode({WINDOW_WIDTH, WINDOW_HEIGHT}), 
                            "PBL2: MO PHONG DIEU PHOI DUONG RAY & CHAY TAU (SFML GRAPHICS)");
    window.setFramerateLimit(60);

    sf::Font font;
    if (!font.openFromFile("arial.ttf")) {
        cout << "[Coi chung] Khong tim thay font 'arial.ttf'!\n";
    }

    double maxKm = 800.0;
    StationNode* stLast = stationList->getHead();
    while (stLast && stLast->getNextStation()) stLast = stLast->getNextStation();
    if (stLast) maxKm = stLast->getKmMarker();

    float marginX = 100.0f;
    float trackY = 300.0f;
    float trackWidth = WINDOW_WIDTH - 2 * marginX;

    float currentMinute = 0.0f;
    float timeSpeed = 30.0f;
    sf::Clock deltaClock;

    while (window.isOpen()) {
        while (const auto event = window.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                window.close();
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Space) {
                    timeSpeed = (timeSpeed == 0.0f) ? 30.0f : 0.0f;
                }
            }
        }

        float dt = deltaClock.restart().asSeconds();
        currentMinute += timeSpeed * dt;
        if (currentMinute > 1440.0f) currentMinute = 0.0f;

        window.clear(sf::Color(30, 30, 35));

        // 1. Đồng hồ thời gian thực
        int hours = (int)currentMinute / 60;
        int mins = (int)currentMinute % 60;
        stringstream ss;
        ss << "PBL2 MO PHONG DIEU PHOI TAU HOA | THOI GIAN: "
           << setfill('0') << setw(2) << hours << ":" << setfill('0') << setw(2) << mins
           << " | (SPACE: Tam dung/Chay tiep)";

        sf::Text titleText(font, ss.str(), 20);
        titleText.setFillColor(sf::Color::White);
        titleText.setPosition(sf::Vector2f(marginX, 20.0f));
        window.draw(titleText);

        // 2. Đường ray chính (Main Track)
        sf::RectangleShape mainTrack(sf::Vector2f(trackWidth, 6.0f));
        mainTrack.setFillColor(sf::Color(150, 150, 150));
        mainTrack.setPosition(sf::Vector2f(marginX, trackY));
        window.draw(mainTrack);

        // 3. Hiển thị Nhà ga & Đường ray phụ
        StationNode* st = stationList->getHead();
        while (st != nullptr) {
            float posX = marginX + (float)(st->getKmMarker() / maxKm) * trackWidth;

            sf::CircleShape stationCircle(10.0f);
            stationCircle.setFillColor(sf::Color(46, 204, 113));
            stationCircle.setOrigin(sf::Vector2f(10.0f, 10.0f));
            stationCircle.setPosition(sf::Vector2f(posX, trackY + 3.0f));
            window.draw(stationCircle);

            sf::Text stName(font, st->getName() + "\n(" + to_string((int)st->getKmMarker()) + " km)", 12);
            stName.setFillColor(sf::Color(241, 196, 15));
            stName.setPosition(sf::Vector2f(posX - 25.0f, trackY + 20.0f));
            window.draw(stName);

            for (int t = 2; t <= st->getNumberOfTracks(); ++t) {
                float offsetY = (t - 1) * 35.0f;
                sf::RectangleShape sidingTrack(sf::Vector2f(80.0f, 4.0f));
                sidingTrack.setFillColor(sf::Color(100, 100, 200));
                sidingTrack.setOrigin(sf::Vector2f(40.0f, 2.0f));
                sidingTrack.setPosition(sf::Vector2f(posX, trackY + 3.0f + offsetY));
                window.draw(sidingTrack);
            }

            st = st->getNextStation();
        }

        // 4. Vẽ các đoàn tàu chuyển động
        TrainNode* train = trainList->getHead();
        while (train != nullptr) {
            int trackUsed = 0;
            double currentKm = calculateTrainKm(train, stationList, (int)currentMinute, trackUsed);

            if (currentKm >= 0.0) {
                float posX = marginX + (float)(currentKm / maxKm) * trackWidth;
                float posY = trackY;

                if (trackUsed > 1) {
                    posY += (trackUsed - 1) * 35.0f;
                }

                sf::RectangleShape trainRect(sf::Vector2f(45.0f, 20.0f));
                trainRect.setOrigin(sf::Vector2f(22.5f, 10.0f));
                trainRect.setPosition(sf::Vector2f(posX, posY + 3.0f));

                if (trackUsed == 0) {
                    trainRect.setFillColor(sf::Color(231, 76, 60));  // Đỏ: Chạy khu gian
                } else if (trackUsed == 1) {
                    trainRect.setFillColor(sf::Color(241, 196, 15)); // Vàng: Ray chính
                } else {
                    trainRect.setFillColor(sf::Color(155, 89, 182)); // Tím: Ray phụ
                }
                window.draw(trainRect);

                sf::Text trainText(font, train->getTrainCode(), 12);
                trainText.setFillColor(sf::Color::White);
                trainText.setPosition(sf::Vector2f(posX - 14.0f, posY - 25.0f));
                window.draw(trainText);
            }

            train = train->getNextTrain();
        }

        window.display();
    }
}

// Show Input Dialog Pop-up
bool GuiMenu::showInputDialog(const string& title, const string& promptLabel, string& outputValue, sf::Font& font) {
    sf::RenderWindow dialog(sf::VideoMode({480, 220}), title, sf::Style::Titlebar | sf::Style::Close);
    dialog.setFramerateLimit(60);

    string inputText = "";
    sf::Text promptText(font, promptLabel, 14);
    promptText.setFillColor(sf::Color::White);
    promptText.setPosition(sf::Vector2f(30.0f, 25.0f));

    sf::RectangleShape inputBox(sf::Vector2f(420.0f, 40.0f));
    inputBox.setPosition(sf::Vector2f(30.0f, 60.0f));
    inputBox.setFillColor(sf::Color(20, 20, 25));
    inputBox.setOutlineThickness(2.0f);
    inputBox.setOutlineColor(sf::Color(52, 152, 219));

    sf::Text typedText(font, "", 16);
    typedText.setFillColor(sf::Color(241, 196, 15));
    typedText.setPosition(sf::Vector2f(40.0f, 68.0f));

    GuiButton btnConfirm(135.0f, 130.0f, 210.0f, 40.0f, "XAC NHAN (ENTER)", font, 99);

    while (dialog.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(dialog);
        bool mouseClicked = false;

        while (const auto event = dialog.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                dialog.close();
                return false;
            }
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
            if (const auto* textEntered = event->getIf<sf::Event::TextEntered>()) {
                uint32_t unicode = textEntered->unicode;
                if (unicode == 8) {
                    if (!inputText.empty()) inputText.pop_back();
                } else if (unicode == 13) {
                    outputValue = inputText;
                    dialog.close();
                    return true;
                } else if (unicode >= 32 && unicode <= 126) {
                    inputText += static_cast<char>(unicode);
                }
            }
        }

        btnConfirm.updateHover(mousePos);
        if (btnConfirm.isClicked(mousePos, mouseClicked)) {
            outputValue = inputText;
            dialog.close();
            return true;
        }

        typedText.setString(inputText + "|");
        dialog.clear(sf::Color(35, 37, 42));
        dialog.draw(promptText);
        dialog.draw(inputBox);
        dialog.draw(typedText);
        btnConfirm.draw(dialog);
        dialog.display();
    }
    return false;
}

// SeatMap UI
void GuiMenu::showSeatMapUI(StationList& stationList, TrainList& trainList, TicketList& ticketList, sf::Font& font) {
    string trainCode, strCarriage, strFrom, strTo;
    if (!showInputDialog("PBL2: CHON TAU TRACUU SEAT MAP", "Nhap Ma Tau (Vi du: SE1):", trainCode, font) ||
        !showInputDialog("PBL2: CHON TOA TRACUU SEAT MAP", "Nhap So Toa (Vi du: 1):", strCarriage, font) ||
        !showInputDialog("PBL2: PHAN DOAN CHANG", "Nhap ID Ga Di (Vi du: 101):", strFrom, font) ||
        !showInputDialog("PBL2: PHAN DOAN CHANG", "Nhap ID Ga Den (Vi du: 105):", strTo, font)) return;

    int carriageNo = stoi(strCarriage), fromSt = stoi(strFrom), toSt = stoi(strTo);
    TrainNode* train = trainList.findTrainByCode(trainCode);
    if (!train) return;

    int totalSeats = train->getCarriageSeats(carriageNo);
    if (totalSeats == 0) return;

    sf::RenderWindow mapWin(sf::VideoMode({800, 500}), "PBL2: SO DO GHE NGOI (SEAT MAP MATRIX)", sf::Style::Titlebar | sf::Style::Close);
    mapWin.setFramerateLimit(60);

    while (mapWin.isOpen()) {
        while (const auto event = mapWin.pollEvent()) {
            if (event->is<sf::Event::Closed>()) mapWin.close();
        }
        mapWin.clear(sf::Color(30, 30, 35));

        sf::Text header(font, "SO DO GHE TOA " + to_string(carriageNo) + " - TAU " + trainCode + " (CHANG " + to_string(fromSt) + " -> " + to_string(toSt) + ")", 16);
        header.setFillColor(sf::Color(241, 196, 15));
        header.setPosition(sf::Vector2f(30.0f, 20.0f));
        mapWin.draw(header);

        for (int i = 1; i <= totalSeats; ++i) {
            int row = (i - 1) % 4, col = (i - 1) / 4;
            float x = 50.0f + col * 60.0f, y = 80.0f + row * 60.0f + (row >= 2 ? 20.0f : 0.0f);

            sf::RectangleShape seatBox(sf::Vector2f(48.0f, 48.0f));
            seatBox.setPosition(sf::Vector2f(x, y));

            bool available = ticketList.isSeatAvailable(&stationList, trainCode, carriageNo, i, fromSt, toSt);
            seatBox.setFillColor(available ? sf::Color(46, 204, 113) : sf::Color(231, 76, 60));
            mapWin.draw(seatBox);

            sf::Text seatNum(font, to_string(i), 12);
            seatNum.setFillColor(sf::Color::White);
            seatNum.setPosition(sf::Vector2f(x + 15.0f, y + 15.0f));
            mapWin.draw(seatNum);
        }
        mapWin.display();
    }
}

// Apply Delay UI
void GuiMenu::applyDelayUI(StationList& stationList, TrainList& trainList, sf::Font& font) {
    string trainCode, strStID, strDelay;
    if (showInputDialog("DELAY PROPAGATION", "Nhap Ma Tau bi tre (Vi du: SE1):", trainCode, font) &&

struct GuiButton {
    sf::RectangleShape shape;
    sf::Text text;
    int actionID;
    bool isHovered;
    bool isSelected;

    GuiButton(float x, float y, float width, float height, const std::string& label, sf::Font& font, int id) 
        : text(font, label, 12), actionID(id), isHovered(false), isSelected(false) 
    {
        shape.setPosition(sf::Vector2f(x, y));
        shape.setSize(sf::Vector2f(width, height));
        shape.setFillColor(sf::Color(41, 128, 185));
        shape.setOutlineThickness(1.5f);
        shape.setOutlineColor(sf::Color(52, 152, 219));

        text.setFillColor(sf::Color::White);
        sf::FloatRect textBounds = text.getLocalBounds();
        text.setPosition(sf::Vector2f(
            x + (width - textBounds.size.x) / 2.0f - textBounds.position.x,
            y + (height - textBounds.size.y) / 2.0f - textBounds.position.y
        ));
    }

    void updatePosition(float x, float y) {
        shape.setPosition(sf::Vector2f(x, y));
        sf::FloatRect textBounds = text.getLocalBounds();
        text.setPosition(sf::Vector2f(
            x + (shape.getSize().x - textBounds.size.x) / 2.0f - textBounds.position.x,
            y + (shape.getSize().y - textBounds.size.y) / 2.0f - textBounds.position.y
        ));
    }

    void updateHover(const sf::Vector2i& mousePos) {
        if (shape.getGlobalBounds().contains(sf::Vector2f(static_cast<float>(mousePos.x), static_cast<float>(mousePos.y)))) {
            isHovered = true;
            if (!isSelected) {
                shape.setFillColor(sf::Color(52, 152, 219));
                shape.setOutlineColor(sf::Color(241, 196, 15));
            }
        } else {
            isHovered = false;
            if (!isSelected) {
                shape.setFillColor(sf::Color(41, 128, 185));
                shape.setOutlineColor(sf::Color(52, 152, 219));
            }
        }
    }

    void setSelected(bool selected) {
        isSelected = selected;
        if (isSelected) {
            shape.setFillColor(sf::Color(39, 174, 96)); // Màu xanh lá khi được chọn
            shape.setOutlineColor(sf::Color(241, 196, 15));
        } else {
            shape.setFillColor(sf::Color(41, 128, 185));
            shape.setOutlineColor(sf::Color(52, 152, 219));
        }
    }

    bool isClicked(const sf::Vector2i& mousePos, bool mousePressed) const {
        return isHovered && mousePressed;
    }

    void draw(sf::RenderWindow& window) {
        window.draw(shape);
        window.draw(text);
    }
};

void TicketList::addTicket(const string& ticketID, const string& trainCode, 
                           int carriageNo, int seatNo, int fromID, int toID) {
    TicketNode* newNode = new TicketNode(ticketID, trainCode, carriageNo, seatNo, fromID, toID);
    if (!head) {
        head = newNode;
        return;
    }
    TicketNode* temp = head;
    while (temp->getNextTicket()) {
        temp = temp->getNextTicket();
    }
    temp->setNextTicket(newNode);
}

TicketNode* TicketList::findTicketByID(const string& ticketID) const {
    TicketNode* temp = head;
    while (temp != nullptr) {
        if (temp->getTicketID() == ticketID) return temp;
        temp = temp->getNextTicket();
    }
    return nullptr;
}

bool TicketList::cancelTicket(const string& ticketID) {
    if (!head) return false;

    if (head->getTicketID() == ticketID) {
        TicketNode* temp = head;
        head = head->getNextTicket();
        delete temp;
        return true;
    }

    TicketNode* curr = head;
    while (curr->getNextTicket() && curr->getNextTicket()->getTicketID() != ticketID) {
        curr = curr->getNextTicket();
    }

    if (curr->getNextTicket()) {
        TicketNode* temp = curr->getNextTicket();
        curr->setNextTicket(temp->getNextTicket());
        delete temp;
        return true;
    }
    return false;
}

// ==================== SEGMENT BOOKING (KIỂM TRA GIAO THOA KM) ====================
bool TicketList::isSeatAvailable(StationList* stationList, const string& trainCode, 
                                int carriageNo, int seatNo, int newFromID, int newToID) const 
{
    if (!stationList) return false;

    StationNode* stNewFrom = stationList->findStationByID(newFromID);
    StationNode* stNewTo = stationList->findStationByID(newToID);
    if (!stNewFrom || !stNewTo) return false;

    double newKmStart = min(stNewFrom->getKmMarker(), stNewTo->getKmMarker());
    double newKmEnd = max(stNewFrom->getKmMarker(), stNewTo->getKmMarker());

    TicketNode* curr = head;
    while (curr != nullptr) {
        if (curr->getTrainCode() == trainCode &&
            curr->getCarriageNo() == carriageNo &&
            curr->getSeatNo() == seatNo) 
        {
            StationNode* stExistFrom = stationList->findStationByID(curr->getFromStationID());
            StationNode* stExistTo = stationList->findStationByID(curr->getToStationID());

            if (stExistFrom && stExistTo) {
                double existKmStart = min(stExistFrom->getKmMarker(), stExistTo->getKmMarker());
                double existKmEnd = max(stExistFrom->getKmMarker(), stExistTo->getKmMarker());

                // Hai đoạn [A1, B1] và [A2, B2] giao thoa khi: max(A1, A2) < min(B1, B2)
                double overlapStart = max(newKmStart, existKmStart);
                double overlapEnd = min(newKmEnd, existKmEnd);

                if (overlapStart < overlapEnd) {
                    return false; // Đã bị chiếm trong đoạn này
                }
            }
        }
        curr = curr->getNextTicket();
    }
    return true; // Ghế còn trống
}

// ==================== THUẬT TOÁN TÁI LẬP LỊCH TRÁNH TÀU (MEET & PASS / SIDING) ====================
void TrainList::resolveConflictAndScheduleSiding(StationList* stationList) {
    for (TrainNode* t1 = head; t1 != nullptr; t1 = t1->getNextTrain()) {
        for (TrainNode* t2 = t1->getNextTrain(); t2 != nullptr; t2 = t2->getNextTrain()) {
            
            StopScheduleNode* s1 = t1->getHeadSchedule();
            while (s1 != nullptr) {
                StopScheduleNode* s2 = t2->getHeadSchedule();
                while (s2 != nullptr) {
                    
                    // Nếu 2 tàu cùng dừng đỗ tại cùng 1 Ga
                    if (s1->getStationID() == s2->getStationID()) {
                        int startOverlap = max(s1->getArrivalMinute(), s2->getArrivalMinute());
                        int endOverlap = min(s1->getDepartureMinute(), s2->getDepartureMinute());

                        // Nếu thời gian đỗ giao thoa trùng nhau và cả 2 cùng đăng ký đỗ tại Ray chính (Ray 1)
                        if (startOverlap < endOverlap && s1->getTrackUsed() == 1 && s2->getTrackUsed() == 1) {
                            // Tái lập lịch: Tự động chuyển Tàu 2 sang đỗ ở Ray 2 (Ray phụ) để tránh đụng độ
                            s2->setTrackUsed(2);
                        }
                    }
                    s2 = s2->getNextStop();
                }
                s1 = s1->getNextStop();
            }
        }
    }
}

// ==================== THUẬT TOÁN DELAY PROPAGATION TRỌN GÓI ====================
bool TrainList::applyDelayAndReschedule(const string& trainCode, int fromStationID, int delayMinutes, StationList* stationList) {
    TrainNode* train = findTrainByCode(trainCode);
    if (!train) return false;

    // BƯỚC 1: Dời giờ đến/đi cho tất cả các ga tiếp theo của tàu bị sự cố
    bool ok = train->applyDelayPropagation(fromStationID, delayMinutes);
    
    // BƯỚC 2: Tự động tái lập lịch tránh tàu ngay sau khi mốc thời gian bị thay đổi
    if (ok) {
        resolveConflictAndScheduleSiding(stationList);
    }
    return ok;
}

#pragma once

#include <string>
#include "CarriageNode.h"
#include "StopScheduleNode.h"

class TrainNode {
private:
    std::string trainCode;
    int direction; // 1: Nam, 0: Bac
    CarriageNode* headCarriage;
    StopScheduleNode* headSchedule;
    TrainNode* nextTrain;

public:
    TrainNode(const std::string& code, int dir)
        : trainCode(code), direction(dir), headCarriage(nullptr), headSchedule(nullptr), nextTrain(nullptr) {}

    ~TrainNode() {
        while (headCarriage) {
            CarriageNode* temp = headCarriage;
            headCarriage = headCarriage->getNextCarriage();
            delete temp;
        }
        while (headSchedule) {
            StopScheduleNode* temp = headSchedule;
            headSchedule = headSchedule->getNextStop();
            delete temp;
        }
    }

    // Getters
    std::string getTrainCode() const { return trainCode; }
    int getDirection() const { return direction; }
    CarriageNode* getHeadCarriage() const { return headCarriage; }
    StopScheduleNode* getHeadSchedule() const { return headSchedule; }
    TrainNode* getNextTrain() const { return nextTrain; }

    // Setters
    void setNextTrain(TrainNode* next) { nextTrain = next; }
    void setHeadCarriage(CarriageNode* head) { headCarriage = head; }
    void setHeadSchedule(StopScheduleNode* head) { headSchedule = head; }

    void addCarriage(int cNo, const std::string& type, int seats) {
        CarriageNode* newNode = new CarriageNode(cNo, type, seats);
        if (!headCarriage) {
            headCarriage = newNode;
            return;
        }
        CarriageNode* temp = headCarriage;
        while (temp->getNextCarriage()) temp = temp->getNextCarriage();
        temp->setNextCarriage(newNode);
    }

    void addSchedule(int stID, int arr, int dep, int track) {
        StopScheduleNode* newNode = new StopScheduleNode(stID, arr, dep, track);
        if (!headSchedule) {
            headSchedule = newNode;
            return;
        }
        StopScheduleNode* temp = headSchedule;
        while (temp->getNextStop()) temp = temp->getNextStop();
        temp->setNextStop(newNode);
    }

    // 1. Lan truyền dời giờ các ga tiếp theo khi xảy ra trễ
    bool applyDelayPropagation(int fromStationID, int delayMinutes) {
        StopScheduleNode* curr = headSchedule;
        bool found = false;

        while (curr != nullptr) {
            if (curr->getStationID() == fromStationID) {
                found = true;
            }
            if (found) {
                curr->setArrivalMinute(curr->getArrivalMinute() + delayMinutes);
                curr->setDepartureMinute(curr->getDepartureMinute() + delayMinutes);
            }
            curr = curr->getNextStop();
        }
        return found;
    }

    int getCarriageSeats(int carriageNo) const {
        CarriageNode* c = headCarriage;
        while (c) {
            if (c->getCarriageNo() == carriageNo) return c->getTotalSeats();
            c = c->getNextCarriage();
        }
        return 0;
    }
};

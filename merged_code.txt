#pragma once

#include <string>

class CarriageNode {
private:
    int carriageNo;
    std::string type;
    int totalSeats;
    CarriageNode* nextCarriage;

public:
    CarriageNode(int cNo, const std::string& cType, int seats)
        : carriageNo(cNo), type(cType), totalSeats(seats), nextCarriage(NULL) {}

    int getCarriageNo() const { return carriageNo; }
    std::string getType() const { return type; }
    int getTotalSeats() const { return totalSeats; }
    CarriageNode* getNextCarriage() const { return nextCarriage; }

    void setNextCarriage(CarriageNode* next) { nextCarriage = next; }
};

// ==================== STATIONS ====================
void FileManager::loadStations(const string& filename, StationList& stationList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string strID, name, strKm, strTracks;

        if (getline(ss, strID, '|') && getline(ss, name, '|') &&
            getline(ss, strKm, '|') && getline(ss, strTracks, '|')) {
            stationList.addStation(stoi(strID), name, stod(strKm), stoi(strTracks));
        }
    }
    inFile.close();
}

void FileManager::saveStations(const string& filename, const StationList& stationList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    StationNode* st = stationList.getHead();
    while (st != nullptr) {
        outFile << st->getStationID() << "|"
                << st->getName() << "|"
                << st->getKmMarker() << "|"
                << st->getMaxTracks() << "\n";
        st = st->getNextStation();
    }
    outFile.close();
}

// ==================== TRAINS ====================
void FileManager::loadTrains(const string& filename, TrainList& trainList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    TrainNode* currentTrain = nullptr;

    while (getline(inFile, line)) {
        if (line.empty()) continue;

        stringstream ss(line);
        string token;
        getline(ss, token, '|');

        if (token == "CARRIAGE") {
            if (currentTrain) {
                string strNo, type, strSeats;
                if (getline(ss, strNo, '|') && getline(ss, type, '|') && getline(ss, strSeats, '|')) {
                    currentTrain->addCarriage(stoi(strNo), type, stoi(strSeats));
                }
            }
        } 
        else if (token == "STOP") {
            if (currentTrain) {
                string strStID, strArr, strDep, strTrack;
                if (getline(ss, strStID, '|') && getline(ss, strArr, '|') &&
                    getline(ss, strDep, '|') && getline(ss, strTrack, '|')) {
                    currentTrain->addSchedule(stoi(strStID), stoi(strArr), stoi(strDep), stoi(strTrack));
                }
            }
        } 
        else {
            // Định dạng dòng đầu: Mã Tàu | Hướng chạy [ | Độ ưu tiên ]
            string trainCode = token;
            string strDir, strPrio;

            if (getline(ss, strDir, '|')) {
                int prio = -1; // -1: TrainNode sẽ tự nhận diện theo tiền tố mã tàu
                
                // Đọc cột thứ 3 (nếu file có lưu sẵn độ ưu tiên)
                if (getline(ss, strPrio, '|') && !strPrio.empty()) {
                    try {
                        prio = stoi(strPrio);
                    } catch (...) {
                        prio = -1;
                    }
                }

                trainList.addTrain(trainCode, stoi(strDir), prio);
                currentTrain = trainList.findTrainByCode(trainCode);
            }
        }
    }
    inFile.close();
}

void FileManager::saveTrains(const string& filename, const TrainList& trainList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    TrainNode* tr = trainList.getHead();
    while (tr != nullptr) {
        // Lưu kèm độ ưu tiên vào cột thứ 3: Mã Tàu | Hướng chạy | Độ ưu tiên
        outFile << tr->getTrainCode() << "|" 
                << tr->getDirection() << "|" 
                << tr->getPriority() << "\n";

        CarriageNode* c = tr->getHeadCarriage();
        while (c != nullptr) {
            outFile << "CARRIAGE|" << c->getCarriageNo() << "|" << c->getType() << "|" << c->getTotalSeats() << "\n";
            c = c->getNextCarriage();
        }

        StopScheduleNode* s = tr->getHeadSchedule();
        while (s != nullptr) {
            outFile << "STOP|" << s->getStationID() << "|" << s->getArrivalMinute() << "|" << s->getDepartureMinute() << "|" << s->getTrackUsed() << "\n";
            s = s->getNextStop();
        }

        tr = tr->getNextTrain();
    }
    outFile.close();
}

// ==================== TICKETS ====================
void FileManager::loadTickets(const string& filename, TicketList& ticketList) {
    ifstream inFile(filename);
    if (!inFile.is_open()) return;

    string line;
    while (getline(inFile, line)) {
        if (line.empty()) continue;
        stringstream ss(line);
        string tID, trCode, strCNo, strSNo, strFrom, strTo;

        if (getline(ss, tID, '|') && getline(ss, trCode, '|') &&
            getline(ss, strCNo, '|') && getline(ss, strSNo, '|') &&
            getline(ss, strFrom, '|') && getline(ss, strTo, '|')) {
            
            ticketList.addTicket(tID, trCode, stoi(strCNo), stoi(strSNo), stoi(strFrom), stoi(strTo));
        }
    }
    inFile.close();
}

void FileManager::saveTickets(const string& filename, const TicketList& ticketList) {
    ofstream outFile(filename);
    if (!outFile.is_open()) return;

    TicketNode* tk = ticketList.getHead();
    while (tk != nullptr) {
        outFile << tk->getTicketID() << "|"
                << tk->getTrainCode() << "|"
                << tk->getCarriageNo() << "|"
                << tk->getSeatNo() << "|"
                << tk->getFromStationID() << "|"
                << tk->getToStationID() << "\n";
        tk = tk->getNextTicket();
    }
    outFile.close();
}

// Cấu trúc lưu trạng thái tọa độ Y của từng tàu để nội suy Lerp (Không dùng std::map)

// ==================== 2. CÁC HÀM ĐỒ HỌA PHỤ TRỢ (SHAPES & VERTEX ARRAY) ====================

// Tạo hình chữ nhật bo góc bằng sf::ConvexShape
static sf::ConvexShape createRoundedRect(float width, float height, float radius, unsigned int cornerPoints = 6) {
    radius = std::min(radius, std::min(width, height) / 2.0f);
    sf::ConvexShape shape;
    shape.setPointCount(cornerPoints * 4);

    const float PI = 3.14159265359f;
    unsigned int idx = 0;

    sf::Vector2f centers[4] = {
        {width - radius, height - radius},
        {radius, height - radius},
        {radius, radius},
        {width - radius, radius}
    };

    for (int c = 0; c < 4; ++c) {
        float startAngle = c * (PI / 2.0f);
        for (unsigned int i = 0; i < cornerPoints; ++i) {
            float angle = startAngle + (static_cast<float>(i) / (cornerPoints - 1)) * (PI / 2.0f);
            shape.setPoint(idx++, {
                centers[c].x + radius * std::cos(angle),
                centers[c].y + radius * std::sin(angle)
            });
        }
    }
    shape.setOrigin({width / 2.0f, height / 2.0f});
    return shape;
}

// Thêm một hình chữ nhật (ghép từ 2 tam giác) vào VertexArray trong SFML 3.0
static void appendQuadAsTriangles(sf::VertexArray& va, float x, float y, float w, float h, const sf::Color& color) {
    sf::Vector2f p0(x, y), p1(x + w, y), p2(x + w, y + h), p3(x, y + h);
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p1, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p0, color});
    va.append(sf::Vertex{p2, color});
    va.append(sf::Vertex{p3, color});
}

// Vẽ đoạn thẳng có độ dày tùy chỉnh (dùng cho ghi rẽ nhánh đường ray phụ)
static void drawThickLine(sf::RenderWindow& window, sf::Vector2f p1, sf::Vector2f p2, float thickness, sf::Color color) {
    sf::Vector2f dir = p2 - p1;
    float length = std::sqrt(dir.x * dir.x + dir.y * dir.y);
    if (length <= 0.001f) return;

    sf::RectangleShape line({length, thickness});
    line.setOrigin({0.0f, thickness / 2.0f});
    line.setPosition(p1);
    float angleRad = std::atan2(dir.y, dir.x);
    line.setRotation(sf::radians(angleRad));
    line.setFillColor(color);
    window.draw(line);
}

// ==================== 3. HÀM TÍNH TỌA ĐỘ KM MƯỢT THEO THỜI GIAN THỰC ====================
static double calculateTrainKmSmooth(TrainNode* train, StationList* stationList, float currentMinute, int& trackUsedOut) {
    trackUsedOut = 0;
    StopScheduleNode* firstStop = train->getHeadSchedule();
    if (!firstStop) return -1.0;

    // Truy xuất trực tiếp ga cuối trong O(1) qua con trỏ tailSchedule
    StopScheduleNode* lastStop = train->getTailSchedule();
    if (!lastStop) {
        lastStop = firstStop;
        while (lastStop->getNextStop() != nullptr) {
            lastStop = lastStop->getNextStop();
        }
    }

    // 1. Chưa tới giờ xuất phát -> Tàu chờ sẵn tại Ga Đầu
    if (currentMinute < static_cast<float>(firstStop->getArrivalMinute())) {
        StationNode* st = stationList->findStationByID(firstStop->getStationID());
        if (st) {
            trackUsedOut = firstStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 2. Đã qua giờ về ga cuối -> Tàu đỗ tại Ga Cuối
    if (currentMinute > static_cast<float>(lastStop->getDepartureMinute())) {
        StationNode* st = stationList->findStationByID(lastStop->getStationID());
        if (st) {
            trackUsedOut = lastStop->getTrackUsed();
            return st->getKmMarker();
        }
    }

    // 3. Kiểm tra tàu đang đỗ tại ga hoặc đang di chuyển giữa 2 ga (Khu gian)
    for (StopScheduleNode* curr = firstStop; curr != nullptr; curr = curr->getNextStop()) {
        float arr = static_cast<float>(curr->getArrivalMinute());
        float dep = static_cast<float>(curr->getDepartureMinute());

        if (currentMinute >= arr && currentMinute <= dep) {
            StationNode* st = stationList->findStationByID(curr->getStationID());
            if (st) {
                trackUsedOut = curr->getTrackUsed();
                return st->getKmMarker();
            }
        }

        StopScheduleNode* nextStop = curr->getNextStop();
        if (nextStop != nullptr) {
            float nextArr = static_cast<float>(nextStop->getArrivalMinute());
            if (currentMinute > dep && currentMinute < nextArr) {
                StationNode* st1 = stationList->findStationByID(curr->getStationID());
                StationNode* st2 = stationList->findStationByID(nextStop->getStationID());
                if (st1 && st2 && nextArr > dep) {
                    trackUsedOut = 0; // 0: Đang chạy trên khu gian đơn
                    double ratio = static_cast<double>(currentMinute - dep) / static_cast<double>(nextArr - dep);
                    return st1->getKmMarker() + ratio * (st2->getKmMarker() - st1->getKmMarker());
                }
            }
        }
    }

    return -1.0;
}

// ==================== 4. VÒNG LẶP ĐỒ HỌA CHÍNH (GRAPHICAL SIMULATION ENGINE) ====================
void GraphicalSimulationEngine::runSFMLSimulation(StationList* stationList, TrainList* trainList) {
    if (!stationList || !trainList) return;

    sf::ContextSettings settings;
    settings.antiAliasingLevel = 8;
    sf::RenderWindow window(sf::VideoMode({WINDOW_WIDTH, WINDOW_HEIGHT}),
                            "PBL2: MO PHONG DIEU PHOI DUONG RAY & CHAY TAU (SFML GRAPHICS)",
                            sf::Style::Default, sf::State::Windowed, settings);
    window.setFramerateLimit(60);

    sf::Font font;
    if (!font.openFromFile("arial.ttf")) {
        cout << "[Canh bao] Khong tim thay font 'arial.ttf'!\n";
    }

    // Tìm mốc Km lớn nhất trên tuyến để chuẩn hóa tỷ lệ vẽ
    double maxKm = 100.0;
    for (StationNode* st = stationList->getHead(); st != nullptr; st = st->getNextStation()) {
        if (st->getKmMarker() > maxKm) maxKm = st->getKmMarker();
    }
    if (maxKm <= 0.0) maxKm = 100.0;

    const float marginX = 95.0f;
    const float trackY = 245.0f;
    const float trackWidth = WINDOW_WIDTH - 2.0f * marginX;
    const float sidingSpacingY = 44.0f;

    // Khởi tạo VertexArray cho Đường ray chính & Tà vẹt (Chỉ tốn 1 Draw Call mỗi khung hình)
    sf::VertexArray mainTrackVA(sf::PrimitiveType::Triangles);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 8.0f, trackWidth + 40.0f, 16.0f, Theme::BallastBed);
    for (float x = marginX - 10.0f; x <= marginX + trackWidth + 10.0f; x += 12.0f) {
        appendQuadAsTriangles(mainTrackVA, x, trackY - 7.0f, 4.0f, 14.0f, Theme::SleeperWood);
    }
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY - 3.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);
    appendQuadAsTriangles(mainTrackVA, marginX - 20.0f, trackY + 1.5f, trackWidth + 40.0f, 2.2f, Theme::MainRailSteel);

    // Mảng tĩnh quản lý hiệu ứng chuyển làn mượt (Smooth Y-Lerp) không dùng STL
    static const int MAX_SMOOTH_TRAINS = 64;
    TrainSmoothState smoothStates[MAX_SMOOTH_TRAINS];
    int smoothCount = 0;

    auto getSmoothY = [&](const string& code, float targetY, float dt) -> float {
        for (int i = 0; i < smoothCount; ++i) {
            if (smoothStates[i].code == code) {
                float lerpFactor = std::min(1.0f, dt * 8.5f);
                smoothStates[i].currentY += (targetY - smoothStates[i].currentY) * lerpFactor;
                return smoothStates[i].currentY;
            }
        }
        if (smoothCount < MAX_SMOOTH_TRAINS) {
            smoothStates[smoothCount].code = code;
            smoothStates[smoothCount].currentY = targetY;
            smoothCount++;
        }
        return targetY;
    };

    // Tự động lấy mốc thời gian bắt đầu sớm nhất của các đoàn tàu (trừ hao 15 phút chuẩn bị)
    int earliestMinute = 1440;
    for (TrainNode* tr = trainList->getHead(); tr != nullptr; tr = tr->getNextTrain()) {
        if (tr->getHeadSchedule()) {
            earliestMinute = std::min(earliestMinute, tr->getHeadSchedule()->getArrivalMinute());
        }
    }
    float startMinute = (earliestMinute < 1440) ? static_cast<float>(std::max(0, earliestMinute - 15)) : 0.0f;
    float currentMinute = startMinute;

    float timeSpeed = 30.0f;
    bool isPaused = false;
    sf::Clock deltaClock;
    sf::Clock animClock;

    while (window.isOpen()) {
        while (const auto event = window.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                window.close();
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Space) {
                    isPaused = !isPaused;
                } else if (keyPressed->code == sf::Keyboard::Key::Up) {
                    timeSpeed = std::min(240.0f, timeSpeed + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Down) {
                    timeSpeed = std::max(5.0f, timeSpeed - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Right) {
                    currentMinute = std::min(1439.0f, currentMinute + 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::Left) {
                    currentMinute = std::max(0.0f, currentMinute - 15.0f);
                } else if (keyPressed->code == sf::Keyboard::Key::R) {
                    currentMinute = startMinute;
                } else if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    window.close();
                }
            }
        }

        float dt = deltaClock.restart().asSeconds();
        float animTime = animClock.getElapsedTime().asSeconds();

        if (!isPaused) {
            currentMinute += timeSpeed * dt;
            if (currentMinute >= 1440.0f) currentMinute = 0.0f;
        }

        window.clear(Theme::BgCanvas);

        // ==================== A. VẼ LƯỚI TỌA ĐỘ NỀN (SUBTLE GRID) ====================
        for (float gx = marginX; gx <= marginX + trackWidth; gx += 120.0f) {
            sf::RectangleShape vLine({1.0f, 390.0f});
            vLine.setPosition({gx, 105.0f});
            vLine.setFillColor(Theme::BgGridLine);
            window.draw(vLine);
        }

        // ==================== B. VẼ NHÀ GA, ĐƯỜNG RAY PHỤ & ĐÈN TÍN HIỆU ====================
        for (StationNode* st = stationList->getHead(); st != nullptr; st = st->getNextStation()) {
            float posX = marginX + static_cast<float>(st->getKmMarker() / maxKm) * trackWidth;
            int maxTracks = st->getMaxTracks();

            // Kiểm tra có đoàn tàu nào đang dừng đỗ tại ga này không để bật đèn tín hiệu Đỏ
            bool stationOccupied = false;
            for (TrainNode* tr = trainList->getHead(); tr != nullptr; tr = tr->getNextTrain()) {
                for (StopScheduleNode* sc = tr->getHeadSchedule(); sc != nullptr; sc = sc->getNextStop()) {
                    if (sc->getStationID() == st->getStationID()) {
                        if (currentMinute >= static_cast<float>(sc->getArrivalMinute()) &&
                            currentMinute <= static_cast<float>(sc->getDepartureMinute())) {
                            stationOccupied = true;
                            break;
                        }
                    }
                }
                if (stationOccupied) break;
            }

            // 1. Trục gióng dọc tại tâm nhà ga
            float totalStationHeight = (maxTracks - 1) * sidingSpacingY + 75.0f;
            sf::RectangleShape guideLine({2.0f, totalStationHeight});
            guideLine.setPosition({posX - 1.0f, trackY - 35.0f});
            guideLine.setFillColor(sf::Color(62, 70, 96, 110));
            window.draw(guideLine);

            // 2. Các đường ray phụ (Siding Tracks: từ Ray 2 đến maxTracks)
            for (int t = 2; t <= maxTracks; ++t) {
                float sidY = trackY + (t - 1) * sidingSpacingY;
                float prevY = trackY + (t - 2) * sidingSpacingY;
                float halfLen = 46.0f;

                // Ghi rẽ nhánh trái & phải nối chéo lên ray phía trên
                drawThickLine(window, {posX - halfLen - 18.0f, prevY}, {posX - halfLen, sidY}, 3.0f, Theme::SidingRail);
                drawThickLine(window, {posX + halfLen, sidY}, {posX + halfLen + 18.0f, prevY}, 3.0f, Theme::SidingRail);

                // Tà vẹt trên ray phụ
                for (float sx = posX - halfLen; sx <= posX + halfLen; sx += 10.0f) {
                    sf::RectangleShape slp({3.0f, 10.0f});
                    slp.setOrigin({1.5f, 5.0f});
                    slp.setPosition({sx, sidY});
                    slp.setFillColor(Theme::SleeperWood);
                    window.draw(slp);
                }

                // Thanh ray phụ
                sf::RectangleShape sidingBar({halfLen * 2.0f, 3.5f});
                sidingBar.setOrigin({halfLen, 1.75f});
                sidingBar.setPosition({posX, sidY});
                sidingBar.setFillColor(Theme::SidingRail);
                window.draw(sidingBar);

                // Nhãn ký hiệu ray phụ (R2, R3...)
                sf::Text trkLabel(font, "R" + to_string(t), 10);
                trkLabel.setFillColor(Theme::TextSecondary);
                trkLabel.setPosition({posX - halfLen - 18.0f, sidY - 6.0f});

// ==================== 1. HỘP THOẠI NHẬP LIỆU HIỆN ĐẠI (MODAL INPUT DIALOG) ====================
bool GuiMenu::showInputDialog(const string& title, const string& promptLabel, string& outputValue, sf::Font& font) {
    sf::RenderWindow dialog(sf::VideoMode({520, 230}), title, sf::Style::Titlebar | sf::Style::Close);
    dialog.setFramerateLimit(60);

    string inputText = "";

    sf::RectangleShape topBanner(sf::Vector2f(520.0f, 38.0f));
    topBanner.setFillColor(GuiTheme::CardBg);
    sf::Text bannerTitle(font, "NHAP THONG TIN HE THONG", 13);
    bannerTitle.setFillColor(GuiTheme::AccentGold);
    bannerTitle.setPosition(sf::Vector2f(20.0f, 10.0f));

    sf::Text promptText(font, promptLabel, 14);
    promptText.setFillColor(GuiTheme::TextLight);
    promptText.setPosition(sf::Vector2f(25.0f, 52.0f));

    sf::RectangleShape inputBox(sf::Vector2f(470.0f, 42.0f));
    inputBox.setPosition(sf::Vector2f(25.0f, 82.0f));
    inputBox.setFillColor(GuiTheme::BgMain);
    inputBox.setOutlineThickness(2.0f);
    inputBox.setOutlineColor(GuiTheme::AccentCyan);

    sf::Text typedText(font, "", 16);
    typedText.setFillColor(GuiTheme::AccentGold);
    typedText.setPosition(sf::Vector2f(35.0f, 92.0f));

    GuiButton btnConfirm;
    btnConfirm.init(75.0f, 150.0f, 175.0f, 40.0f, "XAC NHAN (ENTER)", font, 99, false);
    GuiButton btnCancel;
    btnCancel.init(270.0f, 150.0f, 175.0f, 40.0f, "HUY BO (ESC)", font, 98, true);

    sf::Clock blinkClock;

    while (dialog.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(dialog);
        bool mouseClicked = false;

        while (const auto event = dialog.pollEvent()) {
            if (event->is<sf::Event::Closed>()) {
                dialog.close();
                return false;
            }
            if (const auto* keyPressed = event->getIf<sf::Event::KeyPressed>()) {
                if (keyPressed->code == sf::Keyboard::Key::Escape) {
                    dialog.close();
                    return false;
                }
            }
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
            if (const auto* textEntered = event->getIf<sf::Event::TextEntered>()) {
                uint32_t unicode = textEntered->unicode;
                if (unicode == 8) { // Backspace
                    if (!inputText.empty()) inputText.pop_back();
                } else if (unicode == 13) { // Enter
                    if (!inputText.empty()) {
                        outputValue = inputText;
                        dialog.close();
                        return true;
                    }
                } else if (unicode >= 32 && unicode <= 126) {
                    inputText += static_cast<char>(unicode);
                }
            }
        }

        btnConfirm.updateHover(mousePos);
        btnCancel.updateHover(mousePos);

        if (btnConfirm.isClicked(mouseClicked) && !inputText.empty()) {
            outputValue = inputText;
            dialog.close();
            return true;
        }
        if (btnCancel.isClicked(mouseClicked)) {
            dialog.close();
            return false;
        }

        bool showCursor = (static_cast<int>(blinkClock.getElapsedTime().asSeconds() * 2.0f) % 2 == 0);
        typedText.setString(inputText + (showCursor ? "_" : ""));

        dialog.clear(GuiTheme::PanelBg);
        dialog.draw(topBanner);
        dialog.draw(bannerTitle);
        dialog.draw(promptText);
        dialog.draw(inputBox);
        dialog.draw(typedText);
        btnConfirm.draw(dialog);
        btnCancel.draw(dialog);
        dialog.display();
    }
    return false;
}

// ==================== 2. SƠ ĐỒ GHẾ 2D TƯƠNG TÁC (CLICK ĐỂ ĐẶT VÉ TRỰC TIẾP) ====================
void GuiMenu::showSeatMapUI(StationList& stationList, TrainList& trainList, TicketList& ticketList, sf::Font& font, string& outputLog) {
    string trainCode, strCarriage, strFrom, strTo;
    if (!showInputDialog("SEAT MAP 2D", "1/4. Nhap Ma Tau (Vi du: SE1):", trainCode, font) ||
        !showInputDialog("SEAT MAP 2D", "2/4. Nhap So Toa (Vi du: 1):", strCarriage, font) ||
        !showInputDialog("SEAT MAP 2D", "3/4. Nhap ID Ga Di (Vi du: 101):", strFrom, font) ||
        !showInputDialog("SEAT MAP 2D", "4/4. Nhap ID Ga Den (Vi du: 105):", strTo, font)) return;

    int carriageNo = 0, fromSt = 0, toSt = 0;
    try {
        carriageNo = stoi(strCarriage);
        fromSt = stoi(strFrom);
        toSt = stoi(strTo);
    } catch (...) {
        outputLog = "-> [LOI NHAP LIEU] So toa va ID ga phai la so nguyen hop le!";
        return;
    }

    TrainNode* train = trainList.findTrainByCode(trainCode);
    if (!train) {
        outputLog = "-> [LOI TRA CUU] Khong tim thay doan tau co ma: " + trainCode;
        return;
    }

    int totalSeats = train->getCarriageSeats(carriageNo);
    if (totalSeats <= 0 || totalSeats > 128) {
        outputLog = "-> [LOI TOA XE] Toa so " + to_string(carriageNo) + " khong ton tai tren tau " + trainCode + "!";
        return;
    }

    // Gọi hàm checkCarriageSeats (O(M)) trả về bool để chặn ngay nếu sai hướng hoặc không qua ga
    bool seatFree[129];
    if (!ticketList.checkCarriageSeats(&stationList, &trainList, trainCode, carriageNo, fromSt, toSt, seatFree, totalSeats)) {
        outputLog = "-> [LOI HANH TRINH] Tau " + trainCode + " KHONG phuc vu chang Ga " + to_string(fromSt) +
                    " -> Ga " + to_string(toSt) + "\n   (Nguyen nhan: Nhap nguoc chieu chay cua tau hoac tau khong dung tai ga nay).";
        return;
    }

    StationNode* stFromNode = stationList.findStationByID(fromSt);
    StationNode* stToNode   = stationList.findStationByID(toSt);
    string routeDesc = (stFromNode ? stFromNode->getName() : to_string(fromSt)) + " -> " +
                       (stToNode ? stToNode->getName() : to_string(toSt));

    sf::RenderWindow mapWin(sf::VideoMode({880, 520}), "PBL2: SO DO GHE NGOI TUONG TAC (INTERACTIVE SEAT MAP 2D)", sf::Style::Titlebar | sf::Style::Close);
    mapWin.setFramerateLimit(60);

    string statusBanner = "MEO: Click chuot trai truc tiep vao o GHE XANH de dat ve nhanh cho ghe do!";

    while (mapWin.isOpen()) {
        sf::Vector2i mousePos = sf::Mouse::getPosition(mapWin);
        bool mouseClicked = false;

        while (const auto event = mapWin.pollEvent()) {
            if (event->is<sf::Event::Closed>()) mapWin.close();
            if (const auto* mouseBtn = event->getIf<sf::Event::MouseButtonPressed>()) {
                if (mouseBtn->button == sf::Mouse::Button::Left) mouseClicked = true;
            }
        }

        mapWin.clear(GuiTheme::BgMain);

        // Header thông tin toa tàu
        sf::Text header(font, "SO DO GHE TOA " + to_string(carriageNo) + " | TAU " + trainCode +
                        " (P" + to_string(train->getPriority()) + ") | CHANG: " + routeDesc, 16);
        header.setFillColor(GuiTheme::AccentGold);
        header.setPosition(sf::Vector2f(35.0f, 20.0f));
        mapWin.draw(header);

        // Chú giải (Legend)
        int freeCount = 0;
        for (int i = 1; i <= totalSeats; ++i) if (seatFree[i]) freeCount++;

        sf::Text subHeader(font, "Tong so ghe: " + to_string(totalSeats) +
                           "  |  Con trong (Xanh): " + to_string(freeCount) +
                           "  |  Da dat (Do): " + to_string(totalSeats - freeCount), 13);
        subHeader.setFillColor(GuiTheme::AccentCyan);
        subHeader.setPosition(sf::Vector2f(35.0f, 48.0f));
        mapWin.draw(subHeader);

        // Khung thân toa tàu (Coach Hull)
        sf::RectangleShape carriageHull(sf::Vector2f(810.0f, 330.0f));
        carriageHull.setPosition(sf::Vector2f(35.0f, 85.0f));
        carriageHull.setFillColor(GuiTheme::PanelBg);
        carriageHull.setOutlineThickness(2.0f);
        carriageHull.setOutlineColor(GuiTheme::Border);
        mapWin.draw(carriageHull);

        // Lối đi giữa toa (Center Aisle)
        sf::RectangleShape aisle(sf::Vector2f(790.0f, 26.0f));
        aisle.setPosition(sf::Vector2f(45.0f, 237.0f));
        aisle.setFillColor(GuiTheme::BgMain);
        mapWin.draw(aisle);

        sf::Text aisleText(font, "--- LOI DI GIUA TOA TAU (AISLE) ---", 11);
        aisleText.setFillColor(GuiTheme::TextMuted);
        aisleText.setPosition(sf::Vector2f(330.0f, 243.0f));
        mapWin.draw(aisleText);

        // Vẽ ma trận ghế và bắt sự kiện click đặt vé trực tiếp
        int hoveredSeat = -1;
        for (int i = 1; i <= totalSeats; ++i) {
            int row = (i - 1) % 4;
            int col = (i - 1) / 4;
            float x = 55.0f + col * 48.0f;
            float y = 105.0f + row * 62.0f + (row >= 2 ? 32.0f : 0.0f);

            sf::RectangleShape seatBox(sf::Vector2f(40.0f, 46.0f));
            seatBox.setPosition(sf::Vector2f(x, y));

            bool available = seatFree[i];
            bool isHover = seatBox.getGlobalBounds().contains(sf::Vector2f(static_cast<float>(mousePos.x), static_cast<float>(mousePos.y)));
            if (isHover) hoveredSeat = i;

            seatBox.setFillColor(available ? sf::Color(39, 174, 96) : sf::Color(192, 57, 43));
            seatBox.setOutlineThickness(isHover ? 2.5f : 1.0f);
            seatBox.setOutlineColor(isHover ? GuiTheme::AccentGold : sf::Color(20, 20, 25));
            mapWin.draw(seatBox);

            // Gối tựa đầu ghế
            sf::RectangleShape headrest(sf::Vector2f(8.0f, 34.0f));
            headrest.setPosition(sf::Vector2f(x + 30.0f, y + 6.0f));
            headrest.setFillColor(available ? sf::Color(46, 204, 113) : sf::Color(231, 76, 60));
            mapWin.draw(headrest);

            sf::Text seatNum(font, to_string(i), 12);
            seatNum.setFillColor(sf::Color::White);
            seatNum.setPosition(sf::Vector2f(x + 8.0f, y + 14.0f));
            mapWin.draw(seatNum);

            // Nếu người dùng click vào ghế trống -> Mở hộp thoại nhập Mã Vé để đặt luôn!
            if (isHover && mouseClicked && available) {
                string newTicketID;
                if (showInputDialog("DAT VE NHANH TREN SO DO GHE",
                                    "Nhap Ma Ve moi cho Ghe so " + to_string(i) + " (VD: TCK305):",
                                    newTicketID, font)) 
                {
                    if (ticketList.findTicketByID(newTicketID) != nullptr) {
                        statusBanner = "[LOI] Ma ve " + newTicketID + " da ton tai trong he thong!";
                    } else if (ticketList.isSeatAvailable(&stationList, &trainList, trainCode, carriageNo, i, fromSt, toSt)) {
                        ticketList.addTicket(newTicketID, trainCode, carriageNo, i, fromSt, toSt);
                        FileManager::saveTickets("tickets.txt", ticketList);
                        seatFree[i] = false; // Cập nhật ngay trên giao diện
                        statusBanner = "[THANH CONG] Da phat hanh ve " + newTicketID + " cho Ghe " + to_string(i) + "!";
                        outputLog = "-> [SEAT MAP BOOKING] Da dat ve " + newTicketID + " (Tau " + trainCode +
                                    ", Toa " + to_string(carriageNo) + ", Ghe " + to_string(i) + ")";
                    }
                }
            }
        }

        // Thanh trạng thái dưới đáy cửa sổ Seat Map
        sf::RectangleShape footerBox(sf::Vector2f(810.0f, 45.0f));
        footerBox.setPosition(sf::Vector2f(35.0f, 435.0f));
        footerBox.setFillColor(GuiTheme::CardBg);
        footerBox.setOutlineThickness(1.0f);
        footerBox.setOutlineColor(GuiTheme::Border);
        mapWin.draw(footerBox);

        string hoverInfo = (hoveredSeat != -1)
            ? ("Dang chon Ghe so " + to_string(hoveredSeat) + ": " + (seatFree[hoveredSeat] ? "CON TRONG (Click de dat)" : "DA CO KHACH DAT"))
            : statusBanner;
        sf::Text footerTxt(font, hoverInfo, 13);
        footerTxt.setFillColor(GuiTheme::AccentGold);
        footerTxt.setPosition(sf::Vector2f(50.0f, 448.0f));
        mapWin.draw(footerTxt);

        mapWin.display();
    }
}

// ==================== 3. GIAO DIỆN LAN TRUYỀN TRỄ & TÁI LẬP LỊCH ====================
void GuiMenu::applyDelayUI(StationList& stationList, TrainList& trainList, sf::Font& font, string& outputLog) {
    string trainCode, strStID, strDelay;
    if (showInputDialog("DELAY PROPAGATION", "1/3. Nhap Ma Tau bi su co tre (Vi du: SE1):", trainCode, font) &&
        showInputDialog("DELAY PROPAGATION", "2/3. Nhap ID Ga bat dau bi tre (Vi du: 102):", strStID, font) &&
        showInputDialog("DELAY PROPAGATION", "3/3. Nhap So phut tre (Vi du: 30):", strDelay, font)) 
    {
        try {
            int stID = stoi(strStID);
            int delayMins = stoi(strDelay);
            if (trainList.applyDelayAndReschedule(trainCode, stID, delayMins, &stationList)) {
                FileManager::saveTrains("trains.txt", trainList);
                outputLog = "=== KET QUA LAN TRUYEN TRE & TAI LAP LICH (MULTI-PASS) ===\n\n";
                outputLog += " -> Da cong them " + to_string(delayMins) + " phut cho tau " + trainCode + " tu Ga ID " + to_string(stID) + ".\n";
                outputLog += " -> Da kich hoat thuat toan Multi-pass quet toan tuyen (xet Priority & maxTracks).\n\n";
                outputLog += "Cac xung dot vua duoc giai quyet:\n";
                for (int i = 0; i < trainList.getLogCount(); ++i) {
                    outputLog += "   + " + trainList.getLog(i) + "\n";
                }
            } else {
                outputLog = "-> [THAT BAI] Khong tim thay Tau " + trainCode + " hoac Ga " + strStID + " khong nam trong lich trinh!";
            }
        } catch (...) {
            outputLog = "-> [LOI] ID Ga va So phut tre phai la so nguyen!";
        }
    }
}

struct GuiButton {
    sf::RectangleShape shape;
    sf::RectangleShape accentBar; // Thanh chỉ báo màu bên trái nút
    sf::Text text;
    int actionID;
    bool isHovered;
    bool isSelected;
    bool isSubButton;

    // Constructor mặc định phục vụ cấp phát mảng tĩnh
    GuiButton() : text(*(new sf::Font()), "", 12), actionID(-1), isHovered(false), isSelected(false), isSubButton(false) {}

    void init(float x, float y, float width, float height, const std::string& label, sf::Font& font, int id, bool sub = false) {
        actionID = id;
        isHovered = false;
        isSelected = false;
        isSubButton = sub;

        shape.setPosition(sf::Vector2f(x, y));
        shape.setSize(sf::Vector2f(width, height));
        shape.setFillColor(isSubButton ? GuiTheme::BtnSub : GuiTheme::BtnMain);
        shape.setOutlineThickness(1.0f);
        shape.setOutlineColor(GuiTheme::Border);

        accentBar.setPosition(sf::Vector2f(x, y));
        accentBar.setSize(sf::Vector2f(4.0f, height));
        accentBar.setFillColor(isSubButton ? GuiTheme::AccentCyan : GuiTheme::AccentGold);

        text.setFont(font);
        text.setString(label);
        text.setCharacterSize(isSubButton ? 12 : 13);
        text.setFillColor(GuiTheme::TextLight);

        updatePosition(x, y);
    }

    void updatePosition(float x, float y) {
        shape.setPosition(sf::Vector2f(x, y));
        accentBar.setPosition(sf::Vector2f(x, y));
        sf::FloatRect textBounds = text.getLocalBounds();
        // Căn lề trái có khoảng đệm hiện đại thay vì căn giữa đơn điệu
        text.setPosition(sf::Vector2f(
            x + 14.0f,
            y + (shape.getSize().y - textBounds.size.y) / 2.0f - textBounds.position.y
        ));
    }

    void updateHover(const sf::Vector2i& mousePos) {
        sf::Vector2f mp(static_cast<float>(mousePos.x), static_cast<float>(mousePos.y));
        isHovered = shape.getGlobalBounds().contains(mp);

        if (isHovered) {
            shape.setFillColor(isSubButton ? GuiTheme::BtnSubHover : GuiTheme::BtnMainHover);
            shape.setOutlineColor(isSubButton ? GuiTheme::AccentCyan : GuiTheme::AccentGold);
        } else {
            if (isSelected && !isSubButton) {
                shape.setFillColor(GuiTheme::BtnMainActive);
                shape.setOutlineColor(GuiTheme::AccentGold);
            } else {
                shape.setFillColor(isSubButton ? GuiTheme::BtnSub : GuiTheme::BtnMain);
                shape.setOutlineColor(GuiTheme::Border);
            }
        }
    }

    void setSelected(bool selected) {
        isSelected = selected;
        updateHover({-1, -1});
    }

    bool isClicked(bool mousePressed) const {
        return isHovered && mousePressed;
    }

    void draw(sf::RenderWindow& window) const {
        window.draw(shape);
        if (isSelected || isHovered) {
            window.draw(accentBar);
        }
        window.draw(text);
    }
};

bool TicketList::cancelTicket(const string& ticketID) {
    if (!head) return false;

    // Trường hợp vé cần hủy nằm ngay đầu danh sách
    if (head->getTicketID() == ticketID) {
        TicketNode* temp = head;
        head = head->getNextTicket();
        if (!head) tail = nullptr; // Danh sách trở thành rỗng
        delete temp;
        return true;
    }

    TicketNode* curr = head;
    while (curr->getNextTicket() && curr->getNextTicket()->getTicketID() != ticketID) {
        curr = curr->getNextTicket();
    }

    // Tìm thấy vé ở giữa hoặc cuối danh sách
    if (curr->getNextTicket()) {
        TicketNode* temp = curr->getNextTicket();
        curr->setNextTicket(temp->getNextTicket());
        if (temp == tail) {
            tail = curr; // Cập nhật lại tail nếu hủy vé cuối cùng
        }
        delete temp;
        return true;
    }
    return false;
}

bool TicketList::checkCarriageSeats(StationList* stationList, TrainList* trainList,
                                    const string& trainCode, int carriageNo,
                                    int newFromID, int newToID,
                                    bool seatFreeOut[], int maxSeats) const 
{
    if (!stationList || !trainList || newFromID == newToID || maxSeats <= 0) return false;

    StationNode* stNewFrom = stationList->findStationByID(newFromID);
    StationNode* stNewTo   = stationList->findStationByID(newToID);
    if (!stNewFrom || !stNewTo) return false;

    TrainNode* train = trainList->findTrainByCode(trainCode);
    if (!train) return false;

    bool foundFrom = false;
    bool validDirection = false;
    StopScheduleNode* stop = train->getHeadSchedule();
    while (stop != nullptr) {
        if (stop->getStationID() == newFromID) {
            foundFrom = true;
        } else if (stop->getStationID() == newToID && foundFrom) {
            validDirection = true;
            break;
        }
        stop = stop->getNextStop();
    }

    if (!validDirection) {
        return false; 
    }

    for (int i = 1; i <= maxSeats; ++i) {
        seatFreeOut[i] = true;
    }

    double newKmStart = min(stNewFrom->getKmMarker(), stNewTo->getKmMarker());
    double newKmEnd   = max(stNewFrom->getKmMarker(), stNewTo->getKmMarker());

    TicketNode* curr = head;
    while (curr != nullptr) {
        int sNo = curr->getSeatNo();

        if (curr->getTrainCode() == trainCode &&
            curr->getCarriageNo() == carriageNo &&
            sNo >= 1 && sNo <= maxSeats && seatFreeOut[sNo]) 
        {
            StationNode* stExistFrom = stationList->findStationByID(curr->getFromStationID());
            StationNode* stExistTo   = stationList->findStationByID(curr->getToStationID());

            if (stExistFrom && stExistTo) {
                double existKmStart = min(stExistFrom->getKmMarker(), stExistTo->getKmMarker());
                double existKmEnd   = max(stExistFrom->getKmMarker(), stExistTo->getKmMarker());

                if (max(newKmStart, existKmStart) < min(newKmEnd, existKmEnd)) {
                    seatFreeOut[sNo] = false;
                }
            }
        }
        curr = curr->getNextTicket();
    }

    return true;
}

// ==================== KIỂM TRA ĐƠN LẺ KHI ĐẶT VÉ (SEGMENT BOOKING) ====================
bool TicketList::isSeatAvailable(StationList* stationList, TrainList* trainList, const string& trainCode, 
                                 int carriageNo, int seatNo, int newFromID, int newToID) const 
{
    if (!stationList || !trainList || newFromID == newToID) return false;

    StationNode* stNewFrom = stationList->findStationByID(newFromID);
    StationNode* stNewTo   = stationList->findStationByID(newToID);
    if (!stNewFrom || !stNewTo) return false;

    TrainNode* train = trainList->findTrainByCode(trainCode);
    if (!train) return false;

    int maxSeats = train->getCarriageSeats(carriageNo);
    if (maxSeats == 0 || seatNo <= 0 || seatNo > maxSeats) return false;

    bool foundFrom = false;
    bool validDirection = false;
    StopScheduleNode* stop = train->getHeadSchedule();
    while (stop != nullptr) {
        if (stop->getStationID() == newFromID) {
            foundFrom = true;
        } else if (stop->getStationID() == newToID && foundFrom) {
            validDirection = true;
            break;
        }
        stop = stop->getNextStop();
    }
    if (!validDirection) return false;

    double newKmStart = min(stNewFrom->getKmMarker(), stNewTo->getKmMarker());
    double newKmEnd   = max(stNewFrom->getKmMarker(), stNewTo->getKmMarker());

    TicketNode* curr = head;
    while (curr != nullptr) {
        if (curr->getTrainCode() == trainCode &&
            curr->getCarriageNo() == carriageNo &&
            curr->getSeatNo() == seatNo) 
        {
            StationNode* stExistFrom = stationList->findStationByID(curr->getFromStationID());
            StationNode* stExistTo   = stationList->findStationByID(curr->getToStationID());

            if (stExistFrom && stExistTo) {
                double existKmStart = min(stExistFrom->getKmMarker(), stExistTo->getKmMarker());
                double existKmEnd   = max(stExistFrom->getKmMarker(), stExistTo->getKmMarker());

                if (max(newKmStart, existKmStart) < min(newKmEnd, existKmEnd)) {
                    return false;
                }
            }
        }
        curr = curr->getNextTicket();
    }
    return true;
}

static string formatTimeHHMM(int totalMinutes) {
    int h = (totalMinutes / 60) % 24;
    int m = totalMinutes % 60;
    stringstream ss;
    ss << setfill('0') << setw(2) << h << ":" << setfill('0') << setw(2) << m;
    return ss.str();
}

TrainList::TrainList() : head(nullptr), tail(nullptr), logStart(0), logCount(0) {}

TrainList::~TrainList() {
    clear();
}

void TrainList::clear() {
    while (head != nullptr) {
        TrainNode* temp = head;
        head = head->getNextTrain();
        delete temp;
    }
    head = nullptr;
    tail = nullptr;
}

TrainNode* TrainList::getHead() const { return head; }

void TrainList::addTrain(const string& code, int direction, int priority) {
    TrainNode* newNode = new TrainNode(code, direction, priority);
    if (!head) {
        head = tail = newNode;
        return;
    }
    tail->setNextTrain(newNode);
    tail = newNode;
}

TrainNode* TrainList::findTrainByCode(const string& code) const {
    TrainNode* temp = head;
    while (temp != nullptr) {
        if (temp->getTrainCode() == code) return temp;
        temp = temp->getNextTrain();
    }
    return nullptr;
}

void TrainList::addCarriageToTrain(const string& code, int cNo, const string& type, int seats) {
    TrainNode* train = findTrainByCode(code);
    if (train) train->addCarriage(cNo, type, seats);
}

void TrainList::addStopScheduleToTrain(const string& code, int stID, int arr, int dep, int track) {
    TrainNode* train = findTrainByCode(code);
    if (train) train->addSchedule(stID, arr, dep, track);
}

void TrainList::addConflictLog(const std::string& logMsg) {
    if (logCount < MAX_LOGS) {
        // Chưa đầy mảng: ghi tiếp vào vị trí kế
        int insertIdx = (logStart + logCount) % MAX_LOGS;
        conflictLogs[insertIdx] = logMsg;
        logCount++;
    } else {
        // Đã đầy: ghi đè lên phần tử cũ nhất và dịch vị trí bắt đầu sang phải
        conflictLogs[logStart] = logMsg;
        logStart = (logStart + 1) % MAX_LOGS;
    }
}

std::string TrainList::getLog(int index) const {
    if (index < 0 || index >= logCount) return "";
    int actualIdx = (logStart + index) % MAX_LOGS;
    return conflictLogs[actualIdx];
}

// Tìm ray phụ trống từ ray 2 đến maxTracks tại ga stationID
int TrainList::findAvailableSidingTrack(int stationID, int arr, int dep, int maxTracks, TrainNode* excludeTrain) const {
    if (maxTracks <= 1) return -1;

    const int HEADWAY = 10; // Khoảng đệm an toàn

    for (int candidateTrack = 2; candidateTrack <= maxTracks; ++candidateTrack) {
        bool occupied = false;
        for (TrainNode* tr = head; tr != nullptr; tr = tr->getNextTrain()) {
            if (tr == excludeTrain) continue;

            for (StopScheduleNode* st = tr->getHeadSchedule(); st != nullptr; st = st->getNextStop()) {
                if (st->getStationID() == stationID && st->getTrackUsed() == candidateTrack) {
                    
                    // Dùng std::max và std::min có cộng HEADWAY
                    int overlapStart = std::max(arr, st->getArrivalMinute());
                    int overlapEnd   = std::min(dep + HEADWAY, st->getDepartureMinute() + HEADWAY);

                    if (overlapStart < overlapEnd) {
                        occupied = true;
                        break;
                    }
                }
            }
            if (occupied) break;
        }
        if (!occupied) return candidateTrack;
    }
    return -1;
}

// ==================== THUẬT TOÁN TÁI LẬP LỊCH TRÁNH TÀU ĐA LƯỢT ====================
void TrainList::resolveConflictAndScheduleSiding(StationList* stationList) {
    if (!stationList) return;

    const int SAFETY_HEADWAY = 10; // Khoảng cách an toàn tối thiểu (10 phút)
    const int MAX_PASSES = 10;     // Giới hạn vòng lặp giải tỏa domino
    int passCount = 0;
    bool conflictFound = false;

    do {
        conflictFound = false;
        passCount++;

        for (TrainNode* t1 = head; t1 != nullptr; t1 = t1->getNextTrain()) {
            for (TrainNode* t2 = t1->getNextTrain(); t2 != nullptr; t2 = t2->getNextTrain()) {

                StopScheduleNode* s1 = t1->getHeadSchedule();
                while (s1 != nullptr) {
                    StopScheduleNode* s2 = t2->getHeadSchedule();

                    while (s2 != nullptr) {
                        // =========================================================================
                        // TRƯỜNG HỢP 1: XUNG ĐỘT CÙNG GA CÙNG THỜI ĐIỂM & CÙNG RAY (Station Conflict)
                        // =========================================================================
                        if (s1->getStationID() == s2->getStationID() && s1->getTrackUsed() == s2->getTrackUsed()) {
                            
                            // Sử dụng std::max / std::min có cộng khoảng đệm an toàn
                            int startOverlap = std::max(s1->getArrivalMinute(), s2->getArrivalMinute());
                            int endOverlap   = std::min(s1->getDepartureMinute() + SAFETY_HEADWAY, 
                                                        s2->getDepartureMinute() + SAFETY_HEADWAY);

                            // Phát hiện cấn giờ an toàn tại ga
                            if (startOverlap < endOverlap) {
                                // Phân định tàu nhường: Priority trước, FCFS sau
                                bool t2Yields = true;
                                if (t1->getPriority() != t2->getPriority()) {
                                    t2Yields = (t2->getPriority() > t1->getPriority());
                                } else {
                                    t2Yields = (s2->getArrivalMinute() >= s1->getArrivalMinute());
                                }

                                TrainNode* yieldTrain = t2Yields ? t2 : t1;
                                TrainNode* prioTrain  = t2Yields ? t1 : t2;
                                StopScheduleNode* yieldStop = t2Yields ? s2 : s1;
                                StopScheduleNode* prioStop  = t2Yields ? s1 : s2;

                                StationNode* currStation = stationList->findStationByID(yieldStop->getStationID());
                                int maxTracks = currStation ? currStation->getMaxTracks() : 1;

                                // Dò ray phụ trống tại ga hiện tại
                                int freeSiding = findAvailableSidingTrack(
                                    yieldStop->getStationID(), yieldStop->getArrivalMinute(),
                                    yieldStop->getDepartureMinute(), maxTracks, yieldTrain
                                );

                                if (freeSiding != -1) {
                                    // Ga có ray phụ -> Dạt vào ray phụ an toàn
                                    yieldStop->setTrackUsed(freeSiding);
                                    conflictFound = true;
                                    addConflictLog("[Ga " + std::to_string(yieldStop->getStationID()) + "] " +
                                                   yieldTrain->getTrainCode() + " (P" + std::to_string(yieldTrain->getPriority()) +
                                                   ") vao Ray " + std::to_string(freeSiding) + " nhuong " + prioTrain->getTrainCode());
                                } else {
                                    // Hết ray phụ hoặc ga chỉ có 1 ray -> Đẩy lùi lịch trình về ga trước
                                    StopScheduleNode* prevStop = yieldTrain->findPreviousStop(yieldStop);
                                    int extraWait = std::max(0, prioStop->getDepartureMinute() - yieldStop->getArrivalMinute() + SAFETY_HEADWAY);

                                    if (extraWait > 0) {
                                        if (prevStop != nullptr) {
                                            // Lùi giờ khởi hành tại ga trước
                                            prevStop->setDepartureMinute(prevStop->getDepartureMinute() + extraWait);
                                            
                                            StationNode* prevSt = stationList->findStationByID(prevStop->getStationID());
                                            int prevMaxTracks = prevSt ? prevSt->getMaxTracks() : 1;
                                            
                                            // Kiểm tra ray trống tại ga trước trong khung giờ mới
                                            int prevSiding = findAvailableSidingTrack(
                                                prevStop->getStationID(), prevStop->getArrivalMinute(),
                                                prevStop->getDepartureMinute(), prevMaxTracks, yieldTrain
                                            );
                                            
                                            if (prevSiding != -1) {
                                                prevStop->setTrackUsed(prevSiding); // Có ray phụ -> vào ray phụ
                                            }
                                            // Lan truyền trễ
                                            yieldTrain->applyDelayPropagation(yieldStop->getStationID(), extraWait, true);
                                            addConflictLog("[Ga 1-Ray " + std::to_string(yieldStop->getStationID()) + "] Giu " +
                                                           yieldTrain->getTrainCode() + " tai Ga " + std::to_string(prevStop->getStationID()) +
                                                           " +" + std::to_string(extraWait) + "p nhuong " + prioTrain->getTrainCode());
                                        } else {
                                            // Trường hợp biên: Tàu đang ở ga xuất phát đầu tiên, không thể lùi được nữa
                                            yieldTrain->applyDelayPropagation(yieldStop->getStationID(), extraWait, true);
                                            addConflictLog("[Ga Dau " + std::to_string(yieldStop->getStationID()) + "] Lui khoi hanh " +
                                                           yieldTrain->getTrainCode() + " +" + std::to_string(extraWait) + "p");
                                        }
                                        conflictFound = true; // Kích hoạt Multi-pass lượt kế tiếp
                                    }
                                }
                            }
                        }

                        // =========================================================================
                        // TRƯỜNG HỢP 2: XUNG ĐỘT KHU GIAN ĐƯỜNG ĐƠN (Block Section Conflict)
                        // =========================================================================
                        StopScheduleNode* nextS1 = s1->getNextStop();
                        StopScheduleNode* nextS2 = s2->getNextStop();

                        if (nextS1 != nullptr && nextS2 != nullptr) {
                            bool isHeadOn  = (s1->getStationID() == nextS2->getStationID() && nextS1->getStationID() == s2->getStationID());
                            bool isRearEnd = (s1->getStationID() == s2->getStationID() && nextS1->getStationID() == nextS2->getStationID());

                            if (isHeadOn || isRearEnd) {
                                int t1_start = s1->getDepartureMinute();
                                int t1_end   = nextS1->getArrivalMinute();
                                int t2_start = s2->getDepartureMinute();
                                int t2_end   = nextS2->getArrivalMinute();

                                // Kiểm tra 2 tàu có cùng xuất hiện trên khu gian không
                                int trackOverlapStart = std::max(t1_start, t2_start);
                                int trackOverlapEnd   = std::min(t1_end, t2_end);

                                if (trackOverlapStart < trackOverlapEnd) {
                                    // Phân định tàu nhường
                                    bool t2Yields = true;
                                    if (t1->getPriority() != t2->getPriority()) {
                                        t2Yields = (t2->getPriority() > t1->getPriority());
                                    } else {
                                        t2Yields = (t2_start >= t1_start);
                                    }

                                    TrainNode* yieldTrain = t2Yields ? t2 : t1;
                                    TrainNode* prioTrain  = t2Yields ? t1 : t2;
                                    StopScheduleNode* yieldStop = t2Yields ? s2 : s1;
                                    StopScheduleNode* yieldNext = t2Yields ? nextS2 : nextS1;
                                    
                                    int prioEnd  = t2Yields ? t1_end : t2_end;
                                    int yieldDep = t2Yields ? t2_start : t1_start;

                                    int extraWaitTime = std::max(0, prioEnd - yieldDep + SAFETY_HEADWAY);

                                    if (extraWaitTime > 0) {
                                        // Cộng giờ chờ tại ga trước khi vào khu gian
                                        yieldStop->setDepartureMinute(yieldStop->getDepartureMinute() + extraWaitTime);

                                        StationNode* waitSt = stationList->findStationByID(yieldStop->getStationID());
                                        int maxTracks = waitSt ? waitSt->getMaxTracks() : 1;
                                        
                                        // Tìm ray phụ trong thời gian chờ
                                        int freeSiding = findAvailableSidingTrack(
                                            yieldStop->getStationID(), yieldStop->getArrivalMinute(),
                                            yieldStop->getDepartureMinute(), maxTracks, yieldTrain
                                        );
                                        
                                        if (freeSiding != -1) yieldStop->setTrackUsed(freeSiding);

                                        // Lan truyền trễ
                                        yieldTrain->applyDelayPropagation(yieldNext->getStationID(), extraWaitTime, true);

                                        conflictFound = true;
                                        addConflictLog("[Khu gian " + std::to_string(yieldStop->getStationID()) + "->" +
                                                       std::to_string(yieldNext->getStationID()) + "] " + yieldTrain->getTrainCode() +
                                                       " (P" + std::to_string(yieldTrain->getPriority()) + ") cho " +
                                                       prioTrain->getTrainCode() + " +" + std::to_string(extraWaitTime) +
                                                       "p (Xuat phat: " + formatTimeHHMM(yieldStop->getDepartureMinute()) + ")");
                                    }
                                }
                            }
                        }

                        s2 = s2->getNextStop();
                    }
                    s1 = s1->getNextStop();
                }
            }
        }
    } while (conflictFound && passCount < MAX_PASSES);
}

// ==================== THUẬT TOÁN DELAY PROPAGATION TRỌN GÓI ====================
bool TrainList::applyDelayAndReschedule(const string& trainCode, int fromStationID, int delayMinutes, StationList* stationList) {
    TrainNode* train = findTrainByCode(trainCode);
    if (!train) return false;

    addConflictLog("[Su co] Tau " + trainCode + " bi tre +" + to_string(delayMinutes) + "p tu Ga " + to_string(fromStationID));
    bool ok = train->applyDelayPropagation(fromStationID, delayMinutes, true);

    if (ok) {
        resolveConflictAndScheduleSiding(stationList);
    }
    return ok;
}

#pragma once
#include <string>
#include "CarriageNode.h"
#include "StopScheduleNode.h"

class TrainNode {
private:
    std::string trainCode;
    int direction; // 1: Nam, 0: Bac
    int priority;  // 1: Tau khach nhanh SE, 2: Tau thuong TN, 3: Tau hang
    CarriageNode* headCarriage;
    CarriageNode* tailCarriage;
    StopScheduleNode* headSchedule;
    StopScheduleNode* tailSchedule;
    TrainNode* nextTrain;

public:
    static int inferPriorityFromCode(const std::string& code) {
        if (code.rfind("SE", 0) == 0) return 1; // Ưu tiên cao nhất: SE1, SE2, SE3, SE4...
        if (code.rfind("TN", 0) == 0) return 2; // Ưu tiên trung bình: TN1, TN2...
        return 3;                               // Ưu tiên thấp nhất: Tàu hàng (HH, GS...)
    }

    TrainNode(const std::string& code, int dir, int prio = -1)
        : trainCode(code), direction(dir), 
          priority(prio > 0 ? prio : inferPriorityFromCode(code)),
          headCarriage(nullptr), tailCarriage(nullptr),
          headSchedule(nullptr), tailSchedule(nullptr), nextTrain(nullptr) {}

    ~TrainNode() {
        while (headCarriage) {
            CarriageNode* temp = headCarriage;
            headCarriage = headCarriage->getNextCarriage();
            delete temp;
        }
        while (headSchedule) {
            StopScheduleNode* temp = headSchedule;
            headSchedule = headSchedule->getNextStop();
            delete temp;
        }
    }

    // Getters
    std::string getTrainCode() const { return trainCode; }
    int getDirection() const { return direction; }
    int getPriority() const { return priority; }
    CarriageNode* getHeadCarriage() const { return headCarriage; }
    StopScheduleNode* getHeadSchedule() const { return headSchedule; }
    StopScheduleNode* getTailSchedule() const { return tailSchedule; }
    TrainNode* getNextTrain() const { return nextTrain; }

    // Setters
    void setPriority(int prio) { priority = prio; }
    void setNextTrain(TrainNode* next) { nextTrain = next; }

    // Chèn toa xe độ phức tạp O(1)
    void addCarriage(int cNo, const std::string& type, int seats) {
        CarriageNode* newNode = new CarriageNode(cNo, type, seats);
        if (!headCarriage) {
            headCarriage = tailCarriage = newNode;
            return;
        }
        tailCarriage->setNextCarriage(newNode);
        tailCarriage = newNode;
    }

    // Chèn lịch dừng độ phức tạp O(1)
    void addSchedule(int stID, int arr, int dep, int track) {
        StopScheduleNode* newNode = new StopScheduleNode(stID, arr, dep, track);
        if (!headSchedule) {
            headSchedule = tailSchedule = newNode;
            return;
        }
        tailSchedule->setNextStop(newNode);
        tailSchedule = newNode;
    }

    // Tìm ga dừng liền trước một ga cụ thể trong hành trình của tàu
    StopScheduleNode* findPreviousStop(StopScheduleNode* targetStop) const {
        if (!headSchedule || headSchedule == targetStop) return nullptr;
        StopScheduleNode* curr = headSchedule;
        while (curr && curr->getNextStop() != targetStop) {
            curr = curr->getNextStop();
        }
        return curr;
    }

    // Lan truyền dời giờ các ga tiếp theo khi xảy ra trễ
    bool applyDelayPropagation(int fromStationID, int delayMinutes, bool delayArrivalAtFirst = true) {
        StopScheduleNode* curr = headSchedule;
        bool found = false;

        while (curr != nullptr) {
            if (curr->getStationID() == fromStationID) {
                found = true;
                if (delayArrivalAtFirst) {
                    curr->setArrivalMinute(curr->getArrivalMinute() + delayMinutes);
                }
                curr->setDepartureMinute(curr->getDepartureMinute() + delayMinutes);
            } else if (found) {
                curr->setArrivalMinute(curr->getArrivalMinute() + delayMinutes);
                curr->setDepartureMinute(curr->getDepartureMinute() + delayMinutes);
            }
            curr = curr->getNextStop();
        }
        return found;
    }

    int getCarriageSeats(int carriageNo) const {
        CarriageNode* c = headCarriage;
        while (c) {
            if (c->getCarriageNo() == carriageNo) return c->getTotalSeats();
            c = c->getNextCarriage();
        }
        return 0;
    }
};
